struct LogicVal {
    uint64_t aval; // Data word
    uint64_t bval; // Control word: 0=known, 1=unknown/Z
    uint32_t width;

    LogicVal(uint32_t w = 1, uint64_t a = 0, uint64_t b = 0)
        : aval(a), bval(b), width(w) {
        mask();
    }

    static LogicVal make_x(uint32_t w) {
        uint64_t m = (w >= 64) ? ~0ULL : ((1ULL << w) - 1);
        return LogicVal(w, m, m);
    }

    static LogicVal make_known(uint32_t w, uint64_t val) {
        return LogicVal(w, val, 0);
    }

    void mask() {
        uint64_t m = (width >= 64) ? ~0ULL : ((1ULL << width) - 1);
        aval &= m;
        bval &= m;
    }

    bool is_unknown() const {
        return bval != 0;
    }

    bool to_bool() const {
        if (is_unknown()) return false;
        return aval != 0;
    }

    bool operator==(const LogicVal& o) const {
        return (aval == o.aval) && (bval == o.bval) && (width == o.width);
    }

    bool operator!=(const LogicVal& o) const {
        return !(*this == o);
    }

    // 4-state bitwise NOT (~)
    LogicVal bitwise_not() const {
        uint64_t new_a = (~aval) & (~bval);
        return LogicVal(width, new_a, bval);
    }

    // 4-state bitwise AND (&)
    static LogicVal bitwise_and(const LogicVal& l, const LogicVal& r) {
        uint32_t max_w = std::max(l.width, r.width);
        uint64_t new_a = l.aval & r.aval;
        uint64_t l_zero = (~l.aval) & (~l.bval);
        uint64_t r_zero = (~r.aval) & (~r.bval);
        uint64_t any_zero = l_zero | r_zero;
        uint64_t new_b = (l.bval | r.bval) & (~any_zero);
        return LogicVal(max_w, new_a, new_b);
    }

    // 4-state bitwise OR (|)
    static LogicVal bitwise_or(const LogicVal& l, const LogicVal& r) {
        uint32_t max_w = std::max(l.width, r.width);
        uint64_t new_a = l.aval | r.aval;
        uint64_t l_one = l.aval & (~l.bval);
        uint64_t r_one = r.aval & (~r.bval);
        uint64_t any_one = l_one | r_one;
        uint64_t new_b = (l.bval | r.bval) & (~any_one);
        return LogicVal(max_w, new_a, new_b);
    }

    // 4-state bitwise XOR (^)
    static LogicVal bitwise_xor(const LogicVal& l, const LogicVal& r) {
        uint32_t max_w = std::max(l.width, r.width);
        uint64_t new_b = l.bval | r.bval;
        uint64_t new_a = (l.aval ^ r.aval) & (~new_b);
        return LogicVal(max_w, new_a, new_b);
    }

    // Logical Left Shift (<<)
    static LogicVal shift_left(const LogicVal& l, const LogicVal& r) {
        if (r.is_unknown()) return make_x(l.width);
        uint64_t sh = r.aval;
        if (sh >= l.width) return LogicVal(l.width, 0, 0);
        return LogicVal(l.width, l.aval << sh, l.bval << sh);
    }

    // Logical Right Shift (>>)
    static LogicVal shift_right(const LogicVal& l, const LogicVal& r) {
        if (r.is_unknown()) return make_x(l.width);
        uint64_t sh = r.aval;
        if (sh >= l.width) return LogicVal(l.width, 0, 0);
        return LogicVal(l.width, l.aval >> sh, l.bval >> sh);
    }

    // 4-state Equality (==)
    static LogicVal logical_eq(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval == r.aval) ? 1 : 0);
    }

    // 4-state Inequality (!=)
    static LogicVal logical_neq(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval != r.aval) ? 1 : 0);
    }

    // Relational Operators (<, <=, >, >=)
    static LogicVal rel_lt(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval < r.aval) ? 1 : 0);
    }

    static LogicVal rel_le(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval <= r.aval) ? 1 : 0);
    }

    static LogicVal rel_gt(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval > r.aval) ? 1 : 0);
    }

    static LogicVal rel_ge(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval >= r.aval) ? 1 : 0);
    }

    std::string to_bin_string() const {
        std::string s;
        for (int i = (int)width - 1; i >= 0; --i) {
            bool a = (aval >> i) & 1;
            bool b = (bval >> i) & 1;
            if (!b) {
                s += (a ? '1' : '0');
            } else {
                s += (a ? 'x' : 'z');
            }
        }
        return s;
    }
};

struct NumberNode : public ExpressionNode {
    uint64_t value;
    uint32_t width;
    bool is_unknown = false;

    explicit NumberNode(uint64_t val, uint32_t w = 32, bool unk = false)
        : value(val), width(w), is_unknown(unk) {}

    std::unique_ptr<ExpressionNode> clone() const override {
        return std::make_unique<NumberNode>(value, width, is_unknown);
    }
};

struct IdentifierNode : public ExpressionNode {
    std::string name;

    explicit IdentifierNode(std::string n) : name(std::move(n)) {}

    std::unique_ptr<ExpressionNode> clone() const override {
        return std::make_unique<IdentifierNode>(name);
    }
};

struct IndexSelectNode : public ExpressionNode {
    std::string name;
    std::unique_ptr<ExpressionNode> msb_expr;
    std::unique_ptr<ExpressionNode> lsb_expr;
    bool is_part_select = false;

    IndexSelectNode(std::string n, std::unique_ptr<ExpressionNode> idx)
        : name(std::move(n)), msb_expr(std::move(idx)), lsb_expr(nullptr), is_part_select(false) {}

    IndexSelectNode(std::string n, std::unique_ptr<ExpressionNode> msb, std::unique_ptr<ExpressionNode> lsb)
        : name(std::move(n)), msb_expr(std::move(msb)), lsb_expr(std::move(lsb)), is_part_select(true) {}

    std::unique_ptr<ExpressionNode> clone() const override {
        if (!is_part_select) {
            return std::make_unique<IndexSelectNode>(name, msb_expr ? msb_expr->clone() : nullptr);
        } else {
            return std::make_unique<IndexSelectNode>(
                name,
                msb_expr ? msb_expr->clone() : nullptr,
                lsb_expr ? lsb_expr->clone() : nullptr
            );
        }
    }
};

struct UnaryOpNode : public ExpressionNode {
    char op;
    std::unique_ptr<ExpressionNode> operand;

    UnaryOpNode(char o, std::unique_ptr<ExpressionNode> opnd)
        : op(o), operand(std::move(opnd)) {}

    std::unique_ptr<ExpressionNode> clone() const override {
        return std::make_unique<UnaryOpNode>(op, operand ? operand->clone() : nullptr);
    }
};

struct BinaryOpNode : public ExpressionNode {
    std::string op;
    std::unique_ptr<ExpressionNode> lhs;
    std::unique_ptr<ExpressionNode> rhs;

    BinaryOpNode(std::string o, std::unique_ptr<ExpressionNode> l, std::unique_ptr<ExpressionNode> r)
        : op(std::move(o)), lhs(std::move(l)), rhs(std::move(r)) {}

    std::unique_ptr<ExpressionNode> clone() const override {
        return std::make_unique<BinaryOpNode>(
            op,
            lhs ? lhs->clone() : nullptr,
            rhs ? rhs->clone() : nullptr
        );
    }
};

struct TernaryOpNode : public ExpressionNode {
    std::unique_ptr<ExpressionNode> condition;
    std::unique_ptr<ExpressionNode> true_expr;
    std::unique_ptr<ExpressionNode> false_expr;

    TernaryOpNode(std::unique_ptr<ExpressionNode> c,
                  std::unique_ptr<ExpressionNode> t,
                  std::unique_ptr<ExpressionNode> f)
        : condition(std::move(c)), true_expr(std::move(t)), false_expr(std::move(f)) {}

    std::unique_ptr<ExpressionNode> clone() const override {
        return std::make_unique<TernaryOpNode>(
            condition ? condition->clone() : nullptr,
            true_expr ? true_expr->clone() : nullptr,
            false_expr ? false_expr->clone() : nullptr
        );
    }
};

struct LogicVal {
    uint64_t aval; // Data word
    uint64_t bval; // Control word: 0=known, 1=unknown/Z
    uint32_t width;

    LogicVal(uint32_t w = 1, uint64_t a = 0, uint64_t b = 0)
        : aval(a), bval(b), width(w) {
        mask();
    }

    static LogicVal make_x(uint32_t w) {
        uint64_t m = (w >= 64) ? ~0ULL : ((1ULL << w) - 1);
        return LogicVal(w, m, m);
    }

    static LogicVal make_known(uint32_t w, uint64_t val) {
        return LogicVal(w, val, 0);
    }

    void mask() {
        uint64_t m = (width >= 64) ? ~0ULL : ((1ULL << width) - 1);
        aval &= m;
        bval &= m;
    }

    bool is_unknown() const {
        return bval != 0;
    }

    bool to_bool() const {
        if (is_unknown()) return false;
        return aval != 0;
    }

    bool operator==(const LogicVal& o) const {
        return (aval == o.aval) && (bval == o.bval) && (width == o.width);
    }

    bool operator!=(const LogicVal& o) const {
        return !(*this == o);
    }

    // 4-state bitwise NOT (~)
    LogicVal bitwise_not() const {
        // Known 0 -> 1, Known 1 -> 0, Unknown stays Unknown
        uint64_t new_a = (~aval) & (~bval);
        return LogicVal(width, new_a, bval);
    }

    // 4-state bitwise AND (&): 0 & X = 0, 1 & X = X
    static LogicVal bitwise_and(const LogicVal& l, const LogicVal& r) {
        uint32_t max_w = std::max(l.width, r.width);
        uint64_t new_a = l.aval & r.aval;
        // Result is unknown where an operand is unknown unless the other operand is known 0
        uint64_t l_zero = (~l.aval) & (~l.bval);
        uint64_t r_zero = (~r.aval) & (~r.bval);
        uint64_t any_zero = l_zero | r_zero;
        uint64_t new_b = (l.bval | r.bval) & (~any_zero);
        return LogicVal(max_w, new_a, new_b);
    }

    // 4-state bitwise OR (|): 1 | X = 1, 0 | X = X
    static LogicVal bitwise_or(const LogicVal& l, const LogicVal& r) {
        uint32_t max_w = std::max(l.width, r.width);
        uint64_t new_a = l.aval | r.aval;
        // Result is unknown where an operand is unknown unless the other operand is known 1
        uint64_t l_one = l.aval & (~l.bval);
        uint64_t r_one = r.aval & (~r.bval);
        uint64_t any_one = l_one | r_one;
        uint64_t new_b = (l.bval | r.bval) & (~any_one);
        return LogicVal(max_w, new_a, new_b);
    }

    // 4-state bitwise XOR (^): any unknown bit produces unknown
    static LogicVal bitwise_xor(const LogicVal& l, const LogicVal& r) {
        uint32_t max_w = std::max(l.width, r.width);
        uint64_t new_b = l.bval | r.bval;
        uint64_t new_a = (l.aval ^ r.aval) & (~new_b);
        return LogicVal(max_w, new_a, new_b);
    }

    // Logical Left Shift (<<)
    static LogicVal shift_left(const LogicVal& l, const LogicVal& r) {
        if (r.is_unknown()) return make_x(l.width);
        uint64_t sh = r.aval;
        if (sh >= l.width) return LogicVal(l.width, 0, 0);
        return LogicVal(l.width, l.aval << sh, l.bval << sh);
    }

    // Logical Right Shift (>>)
    static LogicVal shift_right(const LogicVal& l, const LogicVal& r) {
        if (r.is_unknown()) return make_x(l.width);
        uint64_t sh = r.aval;
        if (sh >= l.width) return LogicVal(l.width, 0, 0);
        return LogicVal(l.width, l.aval >> sh, l.bval >> sh);
    }

    // 4-state Equality (==)
    static LogicVal logical_eq(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval == r.aval) ? 1 : 0);
    }

    // 4-state Inequality (!=)
    static LogicVal logical_neq(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval != r.aval) ? 1 : 0);
    }

    // Relational Operators (<, <=, >, >=)
    static LogicVal rel_lt(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval < r.aval) ? 1 : 0);
    }

    static LogicVal rel_le(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval <= r.aval) ? 1 : 0);
    }

    static LogicVal rel_gt(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval > r.aval) ? 1 : 0);
    }

    static LogicVal rel_ge(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval >= r.aval) ? 1 : 0);
    }

    std::string to_bin_string() const {
        std::string s;
        for (int i = (int)width - 1; i >= 0; --i) {
            bool a = (aval >> i) & 1;
            bool b = (bval >> i) & 1;
            if (!b) {
                s += (a ? '1' : '0');
            } else {
                s += (a ? 'x' : 'z');
            }
        }
        return s;
    }
};

void Engine::propagate_comb() {
    delta_cycles_count++;
    bool changed = true;
    size_t iters = 0;
    while (changed && iters < 100) {
        changed = false;

        for (auto& bind : slice_bindings) {
            LogicVal src_val = evaluator.read_slice(bind.src);
            LogicVal dst_val = evaluator.read_slice(bind.dst);
            if (src_val != dst_val) {
                evaluator.write_slice(bind.dst, src_val);
                changed = true;
            }
        }

        for (auto& g : gates) {
            LogicVal res = evaluator.eval_expr(g.in_exprs[0], g.scope);
            for (size_t i = 1; i < g.in_exprs.size(); ++i) {
                LogicVal next_in = evaluator.eval_expr(g.in_exprs[i], g.scope);
                if (g.type == GateType::AND) res = LogicVal::bitwise_and(res, next_in);
                else if (g.type == GateType::OR) res = LogicVal::bitwise_or(res, next_in);
                else if (g.type == GateType::XOR) res = LogicVal::bitwise_xor(res, next_in);
                else if (g.type == GateType::NAND) res = LogicVal::bitwise_and(res, next_in);
                else if (g.type == GateType::NOR) res = LogicVal::bitwise_or(res, next_in);
                else if (g.type == GateType::XNOR) res = LogicVal::bitwise_xor(res, next_in);
            }
            if (g.type == GateType::NAND || g.type == GateType::NOR || g.type == GateType::XNOR || g.type == GateType::NOT) {
                res = res.bitwise_not();
            }
            LogicVal cur_out = evaluator.read_slice(g.out_target);
            if (cur_out != res) {
                evaluator.write_slice(g.out_target, res);
                changed = true;
            }
        }

        for (auto& ca : continuous_assigns) {
            LogicVal val = evaluator.eval_expr(ca.rhs, ca.scope);
            LogicVal cur = evaluator.read_slice(ca.target);
            if (cur != val) {
                evaluator.write_slice(ca.target, val);
                changed = true;
            }
        }

        for (auto& bp : processes) {
            if (bp.proc->type != ProcessType::COMBINATIONAL) continue;
            for (auto& stmt : bp.proc->body) {
                evaluator.execute_statement(stmt.get(), bp);
            }
        }

        for (const auto& n : netlist) {
            if (n.modified) {
                changed = true;
                break;
            }
        }
        iters++;
    }
}

void Engine::evaluate_sequential_blocks() {
    for (auto& bp : processes) {
        if (bp.proc->type != ProcessType::SEQUENTIAL) continue;

        bool triggered = false;
        for (auto& sens : bp.proc->sensitivity_list) {
            std::string scoped_sig = bp.scope + "." + sens.signal_name;
            size_t net_id = net_map.at(scoped_sig);
            const auto& net = netlist[net_id];

            if (sens.edge == EdgeType::POSEDGE && net.is_posedge()) { triggered = true; break; }
            else if (sens.edge == EdgeType::NEGEDGE && net.is_negedge()) { triggered = true; break; }
        }

        if (triggered) {
            for (auto& stmt : bp.proc->body) evaluator.execute_statement(stmt.get(), bp);
        }
    }
}

void Engine::commit_nba_updates() {
    if (nba_queue.empty() && nba_mem_queue.empty()) return;

    nba_commits_count++;
    for (const auto& update : nba_queue) netlist[update.net_id].set(update.val);
    nba_queue.clear();

    for (const auto& mem_up : nba_mem_queue) {
        if (memory_map.find(mem_up.mem_scoped_name) != memory_map.end()) {
            auto& mem = memory_map.at(mem_up.mem_scoped_name);
            if (mem_up.addr < mem.depth) mem.words[mem_up.addr] = mem_up.val;
        }
    }
    nba_mem_queue.clear();
    propagate_comb();
}

void Engine::step_process(size_t pid) {
    auto& bp = processes[pid];
    if (bp.completed) return;

    while (bp.pc < bp.proc->body.size()) {
        auto stmt = bp.proc->body[bp.pc].get();

        if (auto delay = dynamic_cast<DelayStatementNode*>(stmt)) {
            bp.pc++;
            uint64_t wake_time = sim_time + delay->delay;
            time_wheel.push({wake_time, (uint32_t)pid, event_order_counter++});
            return;
        }

        if (auto df = dynamic_cast<DumpFileStatementNode*>(stmt)) {
            vcd_dumper.enable(df->filename);
            bp.pc++;
            continue;
        }

        if (dynamic_cast<DumpVarsStatementNode*>(stmt)) {
            vcd_dumper.init_header(netlist, sim_time);
            bp.pc++;
            continue;
        }

        if (auto if_stmt = dynamic_cast<IfStatementNode*>(stmt)) {
            evaluator.execute_statement(if_stmt, bp);
            if (!simulation_active) return;
            bp.pc++;
            continue;
        }

        if (auto case_stmt = dynamic_cast<CaseStatementNode*>(stmt)) {
            evaluator.execute_statement(case_stmt, bp);
            if (!simulation_active) return;
            bp.pc++;
            continue;
        }

        evaluator.execute_statement(stmt, bp);
        if (!simulation_active) return;

        propagate_comb();
        evaluate_sequential_blocks();
        commit_nba_updates();
        vcd_dumper.dump_sample(netlist, sim_time);
        evaluator.check_monitor();

        for (auto& n : netlist) n.modified = false;
        bp.pc++;
    }

    bp.completed = true;
}

void Engine::run(std::ostream* debug_out) {
    if (debug_out) {
        *debug_out << "[PHASE 4: Scheduler Initialization]\n";
        *debug_out << "  -> Hierarchical Nets Elaborated : " << netlist.size() << "\n";
        *debug_out << "  -> Slice Interconnects Bound    : " << slice_bindings.size() << "\n";
        *debug_out << "  -> Built-in Gates Instantiated  : " << gates.size() << "\n";
        *debug_out << "  -> Continuous Assigns Bound     : " << continuous_assigns.size() << "\n";
        *debug_out << "  -> Memory Arrays Allocated      : " << memory_map.size() << "\n";
        *debug_out << "  -> Concurrent Processes Bound   : " << processes.size() << "\n";
        *debug_out << "[PHASE 5: Simulation Runtime (Discrete-Event Scheduler)]\n";
        *debug_out << "--------------------------------------------------------\n";
    }

    propagate_comb();

    for (size_t i = 0; i < processes.size(); ++i) {
        if (processes[i].proc->type == ProcessType::INITIAL) {
            step_process(i);
        }
    }

    while (simulation_active && !time_wheel.empty()) {
        events_dispatched++;
        SimEvent ev = time_wheel.top();
        time_wheel.pop();

        sim_time = ev.time;
        step_process(ev.process_id);
    }

    if (vcd_dumper.is_enabled()) {
        std::string dumped_file = vcd_dumper.get_filename();
        vcd_dumper.close();
        std::cout << "[CAS VCD] Trace successfully dumped to: " << dumped_file << "\n";
    }

    std::cout << "--------------------------------------------------------\n";
    std::cout << "[CAS SUMMARY] Simulation completed cleanly at T = " << sim_time << " ns.\n" << std::flush;
}

EngineStats Engine::get_stats() {
    EngineStats s;
    s.net_count = netlist.size();
    s.net_cap = netlist.capacity();
    s.proc_count = processes.size();
    s.proc_cap = processes.capacity();
    s.slice_count = slice_bindings.size();
    s.slice_cap = slice_bindings.capacity();
    s.gate_count = gates.size();
    s.gate_cap = gates.capacity();
    s.ca_count = continuous_assigns.size();
    s.ca_cap = continuous_assigns.capacity();
    s.ram_count = memory_map.size();

    size_t words = 0;
    for (const auto& kv : memory_map) {
        words += kv.second.words.capacity();
    }
    s.ram_total_words = words;
    s.net_map_keys = net_map.size();
    s.net_map_buckets = net_map.bucket_count();
    s.net_map_load = net_map.load_factor();
    s.events_dispatched = events_dispatched;
    s.delta_cycles = delta_cycles_count;
    s.nba_commits = nba_commits_count;
    return s;

Lexer::Lexer(const std::string& filename, const std::string& source, const std::string& base_dir, std::ostream* debug_stream)
    : current_dir(base_dir), debug_out(debug_stream) {
    file_stack.push_back({filename, source, 0, 1});
    if (debug_out) {
        *debug_out << "  [Source Accessed] #1: '" << filename << "' (" << source.size() << " bytes)\n";
    }
}

char Lexer::peek() const {
    if (file_stack.empty()) return '\0';
    const auto& cur = file_stack.back();
    if (cur.cursor >= cur.src.size()) return '\0';
    return cur.src[cur.cursor];
}

char Lexer::advance() {
    if (file_stack.empty()) return '\0';
    auto& cur = file_stack.back();
    if (cur.cursor >= cur.src.size()) return '\0';
    char c = cur.src[cur.cursor++];
    if (c == '\n') cur.line++;
    return c;
}

std::string Lexer::get_current_location() const {
    if (file_stack.empty()) return "EOF:0";
    return file_stack.back().filename + ":" + std::to_string(file_stack.back().line);
}

void Lexer::handle_include() {
    skip_whitespace_and_comments();
    char q = advance();
    if (q != '"' && q != '<') {
        throw std::runtime_error(get_current_location() + ": Expected '\"' after `include");
    }
    char close_q = (q == '"') ? '"' : '>';
    std::string inc_path = "";
    while (peek() != close_q && peek() != '\0' && peek() != '\n') {
        inc_path += advance();
    }
    if (peek() == close_q) advance();

    std::string parent_file = file_stack.back().filename;
    size_t parent_line = file_stack.back().line;

    std::string full_path = current_dir.empty() ? inc_path : (current_dir + "/" + inc_path);
    std::ifstream f(full_path);
    if (!f.is_open()) {
        f.open(inc_path);
        if (!f.is_open()) {
            throw std::runtime_error(get_current_location() + ": Cannot open `include file: " + inc_path);
        }
    }

    std::stringstream ss;
    ss << f.rdbuf();
    std::string content = ss.str();

    include_depth++;
    if (debug_out) {
        std::string indent(include_depth * 2, ' ');
        *debug_out << "  " << indent << "â””â”€â”€ [Preprocessor `include] '" << inc_path 
                   << "' (included by " << parent_file << ":" << parent_line 
                   << ", " << content.size() << " bytes)\n";
    }

    file_stack.push_back({inc_path, content, 0, 1});
}

void Lexer::skip_whitespace_and_comments() {
    while (!file_stack.empty()) {
        auto& cur = file_stack.back();
        if (cur.cursor >= cur.src.size()) {
            if (debug_out && file_stack.size() > 1) {
                std::string indent(include_depth * 2, ' ');
                *debug_out << "  " << indent << "â””â”€â”€ [EOF Reached] Returning to '" 
                           << file_stack[file_stack.size() - 2].filename << "'\n";
            }
            if (include_depth > 0) include_depth--;
            file_stack.pop_back();
            continue;
        }

        char c = peek();
        if (std::isspace(c)) {
            advance();
        } else if (c == '/' && cur.cursor + 1 < cur.src.size() && cur.src[cur.cursor + 1] == '/') {
            while (peek() != '\n' && peek() != '\0') advance();
        } else if (c == '/' && cur.cursor + 1 < cur.src.size() && cur.src[cur.cursor + 1] == '*') {
            advance(); advance();
            while (!(peek() == '*' && cur.cursor + 1 < cur.src.size() && cur.src[cur.cursor + 1] == '/')) {
                if (peek() == '\0') break;
                advance();
            }
            if (peek() != '\0') { advance(); advance(); }
        } else if (c == '`') {
            size_t saved_cursor = cur.cursor;
            advance(); // consume '`'
            std::string directive = "";
            while (std::isalpha(peek())) directive += advance();
            if (directive == "include") {
                handle_include();
            } else {
                cur.cursor = saved_cursor;
                break;
            }
        } else {
            break;
        }
    }
}

Token Lexer::next_token() {
    skip_whitespace_and_comments();
    if (file_stack.empty()) return {TokenType::TOKEN_EOF, "", 0, 0, "", 0};

    const std::string cur_file = file_stack.back().filename;
    const size_t cur_line = file_stack.back().line;
    char c = peek();

    // Delimiters
    if (c == '(') { advance(); return {TokenType::TOKEN_LPAREN, "(", 0, 0, cur_file, cur_line}; }
    if (c == ')') { advance(); return {TokenType::TOKEN_RPAREN, ")", 0, 0, cur_file, cur_line}; }
    if (c == '[') { advance(); return {TokenType::TOKEN_LBRACKET, "[", 0, 0, cur_file, cur_line}; }
    if (c == ']') { advance(); return {TokenType::TOKEN_RBRACKET, "]", 0, 0, cur_file, cur_line}; }
    if (c == '{') { advance(); return {TokenType::TOKEN_LBRACE, "{", 0, 0, cur_file, cur_line}; }
    if (c == '}') { advance(); return {TokenType::TOKEN_RBRACE, "}", 0, 0, cur_file, cur_line}; }
    if (c == ';') { advance(); return {TokenType::TOKEN_SEMICOLON, ";", 0, 0, cur_file, cur_line}; }
    if (c == ',') { advance(); return {TokenType::TOKEN_COMMA, ",", 0, 0, cur_file, cur_line}; }
    if (c == '.') { advance(); return {TokenType::TOKEN_DOT, ".", 0, 0, cur_file, cur_line}; }
    if (c == '#') { advance(); return {TokenType::TOKEN_HASH, "#", 0, 0, cur_file, cur_line}; }
    if (c == '@') { advance(); return {TokenType::TOKEN_AT, "@", 0, 0, cur_file, cur_line}; }
    if (c == '?') { advance(); return {TokenType::TOKEN_QUESTION, "?", 0, 0, cur_file, cur_line}; }
    if (c == ':') { advance(); return {TokenType::TOKEN_COLON, ":", 0, 0, cur_file, cur_line}; }
    if (c == '~') { advance(); return {TokenType::TOKEN_TILDE, "~", 0, 0, cur_file, cur_line}; }
    if (c == '^') { advance(); return {TokenType::TOKEN_XOR, "^", 0, 0, cur_file, cur_line}; }
    if (c == '+') { advance(); return {TokenType::TOKEN_PLUS, "+", 0, 0, cur_file, cur_line}; }
    if (c == '-') { advance(); return {TokenType::TOKEN_MINUS, "-", 0, 0, cur_file, cur_line}; }

    // Multi-character Operators
    if (c == '&') {
        advance();
        if (peek() == '&') { advance(); return {TokenType::TOKEN_LOGICAL_AND, "&&", 0, 0, cur_file, cur_line}; }
        return {TokenType::TOKEN_AND, "&", 0, 0, cur_file, cur_line};
    }
    if (c == '|') {
        advance();
        if (peek() == '|') { advance(); return {TokenType::TOKEN_LOGICAL_OR, "||", 0, 0, cur_file, cur_line}; }
        return {TokenType::TOKEN_OR_BIT, "|", 0, 0, cur_file, cur_line};
    }
    if (c == '!') {
        advance();
        if (peek() == '=') { advance(); return {TokenType::TOKEN_NEQ, "!=", 0, 0, cur_file, cur_line}; }
        return {TokenType::TOKEN_NOT, "!", 0, 0, cur_file, cur_line};
    }
    if (c == '=') {
        advance();
        if (peek() == '=') { advance(); return {TokenType::TOKEN_EQ, "==", 0, 0, cur_file, cur_line}; }
        return {TokenType::TOKEN_EQUAL, "=", 0, 0, cur_file, cur_line};
    }
    if (c == '<') {
        advance();
        if (peek() == '=') { advance(); return {TokenType::TOKEN_LE, "<=", 0, 0, cur_file, cur_line}; }
        if (peek() == '<') { advance(); return {TokenType::TOKEN_SHL, "<<", 0, 0, cur_file, cur_line}; }
        return {TokenType::TOKEN_LT, "<", 0, 0, cur_file, cur_line};
    }
    if (c == '>') {
        advance();
        if (peek() == '=') { advance(); return {TokenType::TOKEN_GE, ">=", 0, 0, cur_file, cur_line}; }
        if (peek() == '>') { advance(); return {TokenType::TOKEN_SHR, ">>", 0, 0, cur_file, cur_line}; }
        return {TokenType::TOKEN_GT, ">", 0, 0, cur_file, cur_line};
    }

    // Numeric Literals
    if (std::isdigit(c)) {
        std::string num_str;
        while (std::isdigit(peek())) num_str += advance();

        if (peek() == '\'') {
            uint32_t width = std::stoul(num_str);
            advance();
            char base = advance();
            std::string digits;
            while (std::isxdigit(peek()) || peek() == '_' || peek() == 'x' || peek() == 'X' || peek() == 'z' || peek() == 'Z') {
                if (peek() != '_') digits += peek();
                advance();
            }

            uint64_t val = 0;
            bool is_x_or_z = false;
            for (char ch : digits) {
                if (ch == 'x' || ch == 'X' || ch == 'z' || ch == 'Z') {
                    is_x_or_z = true;
                    break;
                }
            }
            if (!is_x_or_z) {
                try {
                    if (base == 'd' || base == 'D') val = std::stoull(digits);
                    else if (base == 'h' || base == 'H') val = std::stoull(digits, nullptr, 16);
                    else if (base == 'b' || base == 'B') val = std::stoull(digits, nullptr, 2);
                } catch (...) {
                    val = 0;
                }
            }
            return {TokenType::TOKEN_NUMBER, digits, val, width, cur_file, cur_line};
        }
        return {TokenType::TOKEN_NUMBER, num_str, std::stoull(num_str), 32, cur_file, cur_line};
    }

    // String Literals
    if (c == '"') {
        advance();
        std::string s;
        while (peek() != '"' && peek() != '\0') {
            if (peek() == '\\') {
                advance();
                char esc = advance();
                if (esc == 'n') s += '\n';
                else if (esc == 't') s += '\t';
                else s += esc;
            } else {
                s += advance();
            }
        }
        if (peek() == '"') advance();
        return {TokenType::TOKEN_STRING, s, 0, 0, cur_file, cur_line};
    }

    // Identifiers and Keywords
    if (std::isalpha(c) || c == '_' || c == '$') {
        std::string ident;
        while (std::isalnum(peek()) || peek() == '_' || peek() == '$') {
            ident += advance();
        }

        if (ident == "module") return {TokenType::TOKEN_MODULE, ident, 0, 0, cur_file, cur_line};
        if (ident == "endmodule") return {TokenType::TOKEN_ENDMODULE, ident, 0, 0, cur_file, cur_line};
        if (ident == "input") return {TokenType::TOKEN_INPUT, ident, 0, 0, cur_file, cur_line};
        if (ident == "output") return {TokenType::TOKEN_OUTPUT, ident, 0, 0, cur_file, cur_line};
        if (ident == "inout") return {TokenType::TOKEN_INOUT, ident, 0, 0, cur_file, cur_line};
        if (ident == "logic") return {TokenType::TOKEN_LOGIC, ident, 0, 0, cur_file, cur_line};
        if (ident == "wire") return {TokenType::TOKEN_WIRE, ident, 0, 0, cur_file, cur_line};
        if (ident == "reg") return {TokenType::TOKEN_REG, ident, 0, 0, cur_file, cur_line};
        if (ident == "parameter") return {TokenType::TOKEN_PARAMETER, ident, 0, 0, cur_file, cur_line};

void Parser::expect(TokenType t, const std::string& err) {
    if (current.type != t) {
        std::string loc = current.filename.empty() ? ("line " + std::to_string(current.line)) : (current.filename + ":" + std::to_string(current.line));
        throw std::runtime_error(loc + ": error: " + err + " (found: '" + current.text + "')");
    }
    advance();
}

std::unique_ptr<ExpressionNode> Parser::parse_primary() {
    if (current.type == TokenType::TOKEN_NUMBER) {
        uint64_t val = current.num_val;
        uint32_t w = current.bit_width;
        bool is_unk = (current.text.find('x') != std::string::npos || 
                       current.text.find('X') != std::string::npos ||
                       current.text.find('z') != std::string::npos || 
                       current.text.find('Z') != std::string::npos);
        advance();
        return std::make_unique<NumberNode>(val, w, is_unk);
    }

    if (current.type == TokenType::TOKEN_IDENTIFIER) {
        std::string name = current.text;
        advance();

        // Check for bit-select or part-select: name[msb] or name[msb:lsb]
        if (match(TokenType::TOKEN_LBRACKET)) {
            auto msb = parse_expression();
            if (match(TokenType::TOKEN_COLON)) {
                auto lsb = parse_expression();
                expect(TokenType::TOKEN_RBRACKET, "Expected ']' after part-select range");
                return std::make_unique<IndexSelectNode>(name, std::move(msb), std::move(lsb));
            } else {
                expect(TokenType::TOKEN_RBRACKET, "Expected ']' after index expression");
                return std::make_unique<IndexSelectNode>(name, std::move(msb));
            }
        }
        return std::make_unique<IdentifierNode>(name);
    }

    if (match(TokenType::TOKEN_LPAREN)) {
        auto expr = parse_expression();
        expect(TokenType::TOKEN_RPAREN, "Expected ')' after nested expression");
        return expr;
    }

    std::string loc = current.filename.empty() ? ("line " + std::to_string(current.line)) : (current.filename + ":" + std::to_string(current.line));
    throw std::runtime_error(loc + ": error: Unexpected token '" + current.text + "' in expression");
}

std::unique_ptr<StatementNode> Parser::parse_statement() {
    if (match(TokenType::TOKEN_HASH)) {
        uint64_t d = current.num_val;
        expect(TokenType::TOKEN_NUMBER, "Expected numerical delay value after '#'");
        match(TokenType::TOKEN_SEMICOLON);
        return std::make_unique<DelayStatementNode>(d);
    }

    if (match(TokenType::TOKEN_DISPLAY) || match(TokenType::TOKEN_MONITOR)) {
        bool is_mon = (lexer.get_current_location().find("monitor") != std::string::npos);
        expect(TokenType::TOKEN_LPAREN, "Expected '(' after $display / $monitor");

        bool has_lead_time = false;
        std::string fmt = "";

        if (current.type == TokenType::TOKEN_IDENTIFIER && current.text == "$time") {
            has_lead_time = true;
            advance();
            if (match(TokenType::TOKEN_COMMA)) {
                if (current.type == TokenType::TOKEN_STRING) {
                    fmt = current.text;
                    advance();
                }
            }
        } else if (current.type == TokenType::TOKEN_STRING) {
            fmt = current.text;
            advance();
        }

        std::vector<std::string> args;
        while (match(TokenType::TOKEN_COMMA)) {
            args.push_back(current.text);
            advance();
        }

        expect(TokenType::TOKEN_RPAREN, "Expected ')' closing $display / $monitor argument list");
        expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after $display / $monitor");
        return std::make_unique<DisplayStatementNode>(fmt, args, is_mon, has_lead_time);
    }

    if (match(TokenType::TOKEN_DUMPFILE)) {
        expect(TokenType::TOKEN_LPAREN, "Expected '(' after $dumpfile");
        std::string fname = current.text;
        expect(TokenType::TOKEN_STRING, "Expected string literal inside $dumpfile(\"...\")");
        expect(TokenType::TOKEN_RPAREN, "Expected ')' closing $dumpfile");
        expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after $dumpfile");
        return std::make_unique<DumpFileStatementNode>(fname);
    }

    if (match(TokenType::TOKEN_DUMPVARS)) {
        if (match(TokenType::TOKEN_LPAREN)) {
            while (current.type != TokenType::TOKEN_RPAREN && current.type != TokenType::TOKEN_EOF) {
                advance();
            }
            expect(TokenType::TOKEN_RPAREN, "Expected ')' closing $dumpvars");
        }
        expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after $dumpvars");
        return std::make_unique<DumpVarsStatementNode>();
    }

    if (match(TokenType::TOKEN_FINISH)) {
        if (match(TokenType::TOKEN_LPAREN)) expect(TokenType::TOKEN_RPAREN, "Expected ')'");
        expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after $finish");
        return std::make_unique<FinishStatementNode>();
    }

    if (match(TokenType::TOKEN_IF)) {
        expect(TokenType::TOKEN_LPAREN, "Expected '(' after 'if'");
        auto cond = parse_expression();
        expect(TokenType::TOKEN_RPAREN, "Expected ')' after 'if' condition");

        auto if_node = std::make_unique<IfStatementNode>(std::move(cond));
        if_node->then_branch = parse_statement_or_block();

        if (match(TokenType::TOKEN_ELSE)) {
            if_node->else_branch = parse_statement_or_block();
        }
        return if_node;
    }

    if (match(TokenType::TOKEN_CASE)) {
        expect(TokenType::TOKEN_LPAREN, "Expected '(' after 'case'");
        auto sel = parse_expression();
        expect(TokenType::TOKEN_RPAREN, "Expected ')' closing case selector");

        auto case_node = std::make_unique<CaseStatementNode>(std::move(sel));

        while (current.type != TokenType::TOKEN_ENDCASE && current.type != TokenType::TOKEN_EOF) {
            CaseItem item;
            if (match(TokenType::TOKEN_DEFAULT)) {
                item.is_default = true;
                match(TokenType::TOKEN_COLON);
            } else {
                item.match_expr = parse_expression();
                expect(TokenType::TOKEN_COLON, "Expected ':' after case match expression");
            }
            item.statements = parse_statement_or_block();
            case_node->items.push_back(std::move(item));
        }

        expect(TokenType::TOKEN_ENDCASE, "Expected 'endcase'");
        return case_node;
    }

    // Concatenation assignment: {cout, sum} = a + b + cin;
    if (match(TokenType::TOKEN_LBRACE)) {
        std::vector<std::string> targets;
        while (true) {
            targets.push_back(current.text);
            expect(TokenType::TOKEN_IDENTIFIER, "Expected identifier in concatenation target list");
            if (match(TokenType::TOKEN_COMMA)) continue;
            break;
        }
        expect(TokenType::TOKEN_RBRACE, "Expected '}' closing concatenation target list");
        expect(TokenType::TOKEN_EQUAL, "Expected '=' in concatenation blocking assignment");
        auto rhs = parse_expression();
        expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after concatenation assignment");
        return std::make_unique<BlockingAssignNode>(targets, std::move(rhs));
    }

    // Identifier-driven statements (assignment, memory write, or task call)
    if (current.type == TokenType::TOKEN_IDENTIFIER) {
        std::string ident = current.text;
        advance();

        // Memory array write: mem[addr] <= rhs or mem[addr] = rhs
        if (match(TokenType::TOKEN_LBRACKET)) {
            auto addr_expr = parse_expression();
            expect(TokenType::TOKEN_RBRACKET, "Expected ']' in memory assignment index");

            bool is_nba = false;
            if (match(TokenType::TOKEN_LE)) is_nba = true;
            else expect(TokenType::TOKEN_EQUAL, "Expected '=' or '<=' in memory assignment");

            auto rhs = parse_expression();
            expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after memory assignment");
            return std::make_unique<MemAssignNode>(ident, std::move(addr_expr), std::move(rhs), is_nba);
        }

        // Non-blocking assignment: target <= rhs;
        if (match(TokenType::TOKEN_LE)) {
            auto rhs = parse_expression();
            expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after non-blocking assignment");
            return std::make_unique<NonBlockingAssignNode>(ident, std::move(rhs));
        }

        // Blocking assignment: target = rhs;
        if (match(TokenType::TOKEN_EQUAL)) {
            auto rhs = parse_expression();
            expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after blocking assignment");
            return std::make_unique<BlockingAssignNode>(ident, std::move(rhs));
        }

        // Task call: task_name(); or task_name;
        if (match(TokenType::TOKEN_LPAREN)) {
            expect(TokenType::TOKEN_RPAREN, "Expected ')' in task call");
        }
        expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after statement");
        return std::make_unique<TaskCallNode>(ident);
    }

    std::string loc = current.filename.empty() ? ("line " + std::to_string(current.line)) : (current.filename + ":" + std::to_string(current.line));
    throw std::runtime_error(loc + ": error: Unexpected statement beginning with '" + current.text + "'");
}

std::vector<std::unique_ptr<StatementNode>> Parser::parse_statement_or_block() {
    std::vector<std::unique_ptr<StatementNode>> stmts;
    if (match(TokenType::TOKEN_BEGIN)) {
        while (current.type != TokenType::TOKEN_END && current.type != TokenType::TOKEN_EOF) {
            stmts.push_back(parse_statement());
        }
        expect(TokenType::TOKEN_END, "Expected 'end' closing begin block");
    } else {
        stmts.push_back(parse_statement());
    }
    return stmts;
}

std::unique_ptr<ProcessNode> Parser::parse_always_ff() {
    expect(TokenType::TOKEN_AT, "Expected '@' after always_ff");
    expect(TokenType::TOKEN_LPAREN, "Expected '(' in sensitivity list");

    auto proc = std::make_unique<ProcessNode>(ProcessType::SEQUENTIAL);

    while (true) {
        EdgeType edge = EdgeType::LEVEL;
        if (match(TokenType::TOKEN_POSEDGE)) edge = EdgeType::POSEDGE;
        else if (match(TokenType::TOKEN_NEGEDGE)) edge = EdgeType::NEGEDGE;

        std::string sig = current.text;
        expect(TokenType::TOKEN_IDENTIFIER, "Expected signal name in sensitivity list");
        proc->sensitivity_list.push_back({edge, sig});

        if (match(TokenType::TOKEN_OR) || match(TokenType::TOKEN_COMMA)) continue;
        break;
    }

    expect(TokenType::TOKEN_RPAREN, "Expected ')' closing sensitivity list");
    proc->body = parse_statement_or_block();
    return proc;
}

// Top-Level Module & File Parsing

std::vector<std::unique_ptr<ModuleNode>> Parser::parse_all() {
    std::vector<std::unique_ptr<ModuleNode>> modules;

    while (current.type != TokenType::TOKEN_EOF) {
        // Handle compilation-unit scope parameters (e.g. from `include "defs.vh"`)
        if (current.type == TokenType::TOKEN_PARAMETER || current.type == TokenType::TOKEN_LOCALPARAM) {
            advance(); // consume parameter / localparam
            
            // Optional type/width like [3:0] or logic [3:0]
            if (current.type == TokenType::TOKEN_LOGIC || current.type == TokenType::TOKEN_REG || 
                current.type == TokenType::TOKEN_WIRE || (current.type == TokenType::TOKEN_IDENTIFIER && (current.text == "int" || current.text == "byte"))) {
                advance();
            }
            if (match(TokenType::TOKEN_LBRACKET)) {
                parse_expression();
                expect(TokenType::TOKEN_COLON, "Expected ':' in parameter range");
                parse_expression();
                expect(TokenType::TOKEN_RBRACKET, "Expected ']' in parameter range");
            }

            while (true) {
                std::string pname = current.text;
                expect(TokenType::TOKEN_IDENTIFIER, "Expected parameter identifier");
                expect(TokenType::TOKEN_EQUAL, "Expected '=' in parameter assignment");
                auto val_expr = parse_expression();

                ParameterDef pdef;
                pdef.name = pname;
                pdef.default_expr = std::move(val_expr);
                global_parameters.push_back(std::move(pdef));

                if (match(TokenType::TOKEN_COMMA)) continue;
                break;
            }
            expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after parameter declaration");
            continue;
        }

        // Module parsing
        expect(TokenType::TOKEN_MODULE, "Expected 'module', 'parameter', or 'localparam' keyword");
        auto mod = std::make_unique<ModuleNode>();
        mod->defined_in_file = current.filename;
        mod->defined_on_line = current.line;
        mod->name = current.text;
        expect(TokenType::TOKEN_IDENTIFIER, "Expected module name");

        // Inherit global compilation-unit parameters into this module
        for (const auto& gp : global_parameters) {
            ParameterDef pcopy;
            pcopy.name = gp.name;
            pcopy.default_expr = gp.default_expr ? gp.default_expr->clone() : nullptr;
            mod->parameters.push_back(std::move(pcopy));
        }

        // Check for parameterized port syntax: #(parameter ...)
        if (match(TokenType::TOKEN_HASH)) {
            expect(TokenType::TOKEN_LPAREN, "Expected '(' after '#' in module header");
            while (current.type != TokenType::TOKEN_RPAREN && current.type != TokenType::TOKEN_EOF) {
                if (current.type == TokenType::TOKEN_PARAMETER || current.type == TokenType::TOKEN_LOCALPARAM) {
                    advance();
                }
                std::string pname = current.text;
                expect(TokenType::TOKEN_IDENTIFIER, "Expected parameter identifier");
                expect(TokenType::TOKEN_EQUAL, "Expected '=' in parameter definition");
                auto pexpr = parse_expression();
                mod->parameters.push_back({pname, std::move(pexpr)});

                if (match(TokenType::TOKEN_COMMA)) continue;
                else break;
            }
            expect(TokenType::TOKEN_RPAREN, "Expected ')' closing parameter list");
        }

        // Parse ANSI Ports (if present)
        if (match(TokenType::TOKEN_LPAREN)) {
            while (current.type != TokenType::TOKEN_RPAREN && current.type != TokenType::TOKEN_EOF) {
                std::string dir = "input";
                if (current.type == TokenType::TOKEN_INPUT) { dir = "input"; advance(); }
                else if (current.type == TokenType::TOKEN_OUTPUT) { dir = "output"; advance(); }
                else if (current.type == TokenType::TOKEN_INOUT) { dir = "inout"; advance(); }

                if (current.type == TokenType::TOKEN_LOGIC || current.type == TokenType::TOKEN_WIRE || current.type == TokenType::TOKEN_REG) {
                    advance();
                }

                std::unique_ptr<ExpressionNode> msb = nullptr;
                std::unique_ptr<ExpressionNode> lsb = nullptr;
                if (match(TokenType::TOKEN_LBRACKET)) {
                    msb = parse_expression();
                    expect(TokenType::TOKEN_COLON, "Expected ':' in port range");
                    lsb = parse_expression();
                    expect(TokenType::TOKEN_RBRACKET, "Expected ']' in port range");
                }

                while (true) {
                    std::string pname = current.text;
                    expect(TokenType::TOKEN_IDENTIFIER, "Expected port identifier");
                    PortNode port(pname, dir, msb ? msb->clone() : nullptr, lsb ? lsb->clone() : nullptr);
                    mod->ports.push_back(std::move(port));
                    if (match(TokenType::TOKEN_COMMA)) {
                        if (current.type == TokenType::TOKEN_INPUT || current.type == TokenType::TOKEN_OUTPUT || current.type == TokenType::TOKEN_INOUT ||
                            current.type == TokenType::TOKEN_LOGIC || current.type == TokenType::TOKEN_WIRE || current.type == TokenType::TOKEN_REG) {
                            break;
                        }
                        continue;
                    }
                    break;
                }
            }
            expect(TokenType::TOKEN_RPAREN, "Expected ')' closing port list");
        }
        expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after module header");

        // Parse module body until endmodule
        while (current.type != TokenType::TOKEN_ENDMODULE && current.type != TokenType::TOKEN_EOF) {
            // Localparam & Parameter inside module body
            if (current.type == TokenType::TOKEN_PARAMETER || current.type == TokenType::TOKEN_LOCALPARAM) {
                advance();
                if (current.type == TokenType::TOKEN_LOGIC || current.type == TokenType::TOKEN_REG || 
                    current.type == TokenType::TOKEN_WIRE) {
                    advance();
                }
                if (match(TokenType::TOKEN_LBRACKET)) {
                    parse_expression();
                    expect(TokenType::TOKEN_COLON, "Expected ':' in parameter range");
                    parse_expression();
                    expect(TokenType::TOKEN_RBRACKET, "Expected ']' in parameter range");
                }
                while (true) {
                    std::string pname = current.text;
                    expect(TokenType::TOKEN_IDENTIFIER, "Expected parameter identifier");
                    expect(TokenType::TOKEN_EQUAL, "Expected '=' in parameter assignment");
                    auto val_expr = parse_expression();
                    mod->parameters.push_back({pname, std::move(val_expr)});
                    if (match(TokenType::TOKEN_COMMA)) continue;
                    break;
                }
                expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after parameter declaration");
                continue;
            }

            // Input / Output / Inout declarations (Non-ANSI)
            if (current.type == TokenType::TOKEN_INPUT || current.type == TokenType::TOKEN_OUTPUT || current.type == TokenType::TOKEN_INOUT) {
                std::string dir = current.text;
                advance();
                if (current.type == TokenType::TOKEN_LOGIC || current.type == TokenType::TOKEN_WIRE || current.type == TokenType::TOKEN_REG) {
                    advance();
                }
                std::unique_ptr<ExpressionNode> msb = nullptr;
                std::unique_ptr<ExpressionNode> lsb = nullptr;
                if (match(TokenType::TOKEN_LBRACKET)) {
                    msb = parse_expression();
                    expect(TokenType::TOKEN_COLON, "Expected ':' in range");
                    lsb = parse_expression();
                    expect(TokenType::TOKEN_RBRACKET, "Expected ']' in range");
                }
                while (true) {
                    std::string pname = current.text;
                    expect(TokenType::TOKEN_IDENTIFIER, "Expected identifier");
                    bool found = false;
                    for (auto& p : mod->ports) {
                        if (p.name == pname) {
                            p.direction = dir;
                            if (msb) p.msb_expr = msb->clone();
                            if (lsb) p.lsb_expr = lsb->clone();
                            found = true;
                            break;
                        }
                    }
                    if (!found) {
                        mod->ports.emplace_back(pname, dir, msb ? msb->clone() : nullptr, lsb ? lsb->clone() : nullptr);
                    }
                    if (match(TokenType::TOKEN_COMMA)) continue;
                    break;
                }
                expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after port declaration");
                continue;
            }

            // Internal signals: logic, wire, reg
            if (current.type == TokenType::TOKEN_LOGIC || current.type == TokenType::TOKEN_WIRE || current.type == TokenType::TOKEN_REG) {
                advance();
                std::unique_ptr<ExpressionNode> msb = nullptr;
                std::unique_ptr<ExpressionNode> lsb = nullptr;
                if (match(TokenType::TOKEN_LBRACKET)) {
                    msb = parse_expression();
                    expect(TokenType::TOKEN_COLON, "Expected ':' in range");
                    lsb = parse_expression();
                    expect(TokenType::TOKEN_RBRACKET, "Expected ']' in range");
                }
                while (true) {
                    std::string signame = current.text;
                    expect(TokenType::TOKEN_IDENTIFIER, "Expected signal identifier");
                    PortNode sig(signame, "internal", msb ? msb->clone() : nullptr, lsb ? lsb->clone() : nullptr);

                    // Check for memory array declaration: logic [7:0] mem [0:255];
                    if (match(TokenType::TOKEN_LBRACKET)) {
                        sig.is_array = true;
                        sig.bound1_expr = parse_expression();
                        if (match(TokenType::TOKEN_COLON)) {
                            sig.bound2_expr = parse_expression();
                        }
                        expect(TokenType::TOKEN_RBRACKET, "Expected ']' in memory dimension");
                    }

                    bool is_port = false;
                    for (auto& p : mod->ports) {
                        if (p.name == signame) {
                            if (msb) p.msb_expr = msb->clone();
                            if (lsb) p.lsb_expr = lsb->clone();
                            is_port = true;
                            break;
                        }
                    }
                    if (!is_port) {
                        mod->signals.push_back(std::move(sig));
                    }
                    if (match(TokenType::TOKEN_COMMA)) continue;
                    break;
                }
                expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after signal declaration");
                continue;
            }

            // Continuous Assign
            if (match(TokenType::TOKEN_ASSIGN)) {
                std::string target = current.text;
                expect(TokenType::TOKEN_IDENTIFIER, "Expected target identifier in assign");
                expect(TokenType::TOKEN_EQUAL, "Expected '=' in assign statement");
                auto rhs = parse_expression();
                expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after assign");
                mod->continuous_assigns.push_back(std::make_unique<ContinuousAssignNode>(target, std::move(rhs)));
                continue;
            }

            // Gate Primitives (and, or, xor, nand, nor, xnor, not, buf)
            if (current.type == TokenType::TOKEN_GATE_AND  || current.type == TokenType::TOKEN_GATE_OR   ||
                current.type == TokenType::TOKEN_GATE_XOR  || current.type == TokenType::TOKEN_GATE_NAND ||
                current.type == TokenType::TOKEN_GATE_NOR  || current.type == TokenType::TOKEN_GATE_XNOR ||
                current.type == TokenType::TOKEN_GATE_NOT  || current.type == TokenType::TOKEN_GATE_BUF) {
                
                GateType gt = GateType::AND;
                if (current.type == TokenType::TOKEN_GATE_OR) gt = GateType::OR;
                else if (current.type == TokenType::TOKEN_GATE_XOR) gt = GateType::XOR;
                else if (current.type == TokenType::TOKEN_GATE_NAND) gt = GateType::NAND;
                else if (current.type == TokenType::TOKEN_GATE_NOR) gt = GateType::NOR;
                else if (current.type == TokenType::TOKEN_GATE_XNOR) gt = GateType::XNOR;
                else if (current.type == TokenType::TOKEN_GATE_NOT) gt = GateType::NOT;
                else if (current.type == TokenType::TOKEN_GATE_BUF) gt = GateType::BUF;
                advance();

                std::string inst_name = "";
                if (current.type == TokenType::TOKEN_IDENTIFIER) {
                    inst_name = current.text;
                    advance();
                } else {
                    inst_name = "gate_" + std::to_string(mod->gates.size());
                }

                expect(TokenType::TOKEN_LPAREN, "Expected '(' in gate instance");
                auto out_expr = parse_expression();
                expect(TokenType::TOKEN_COMMA, "Expected ',' after gate output");
                std::vector<std::unique_ptr<ExpressionNode>> in_exprs;
                while (true) {
                    in_exprs.push_back(parse_expression());
                    if (match(TokenType::TOKEN_COMMA)) continue;
                    break;
                }
                expect(TokenType::TOKEN_RPAREN, "Expected ')' closing gate instance");
                expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after gate instance");

                auto gate = std::make_unique<GatePrimitiveNode>(gt, std::move(out_expr), std::move(in_exprs), inst_name);
                mod->gates.push_back(std::move(gate));
                continue;
            }

            // Tasks
            if (match(TokenType::TOKEN_TASK)) {
                auto t = std::make_unique<TaskNode>();
                t->name = current.text;
                expect(TokenType::TOKEN_IDENTIFIER, "Expected task name");
                expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after task name");
                while (current.type != TokenType::TOKEN_ENDTASK && current.type != TokenType::TOKEN_EOF) {
                    t->body.push_back(parse_statement());
                }
                expect(TokenType::TOKEN_ENDTASK, "Expected 'endtask'");
                mod->tasks.push_back(std::move(t));
                continue;
            }

            // Initial block
            if (match(TokenType::TOKEN_INITIAL)) {
                auto proc = std::make_unique<ProcessNode>(ProcessType::INITIAL);
                proc->body = parse_statement_or_block();
                mod->processes.push_back(std::move(proc));
                continue;
            }

            // always_comb block
            if (match(TokenType::TOKEN_ALWAYS_COMB)) {
                auto proc = std::make_unique<ProcessNode>(ProcessType::COMBINATIONAL);
                proc->body = parse_statement_or_block();
                mod->processes.push_back(std::move(proc));
                continue;
            }

            // always_ff block
            if (match(TokenType::TOKEN_ALWAYS_FF)) {
                mod->processes.push_back(parse_always_ff());
                continue;
            }

            // always block (check for @(posedge ...))
            if (match(TokenType::TOKEN_ALWAYS)) {
                if (current.type == TokenType::TOKEN_AT) {
                    advance(); // consume '@'
                    expect(TokenType::TOKEN_LPAREN, "Expected '(' after '@'");
                    auto proc = std::make_unique<ProcessNode>(ProcessType::SEQUENTIAL);
                    while (true) {
                        EdgeType edge = EdgeType::LEVEL;
                        if (match(TokenType::TOKEN_POSEDGE)) edge = EdgeType::POSEDGE;
                        else if (match(TokenType::TOKEN_NEGEDGE)) edge = EdgeType::NEGEDGE;

                        std::string sig = current.text;
                        expect(TokenType::TOKEN_IDENTIFIER, "Expected signal name in sensitivity list");
                        proc->sensitivity_list.push_back({edge, sig});

                        if (match(TokenType::TOKEN_OR) || match(TokenType::TOKEN_COMMA)) continue;
                        break;
                    }
                    expect(TokenType::TOKEN_RPAREN, "Expected ')' closing sensitivity list");
                    proc->body = parse_statement_or_block();
                    mod->processes.push_back(std::move(proc));
                } else {
                    auto proc = std::make_unique<ProcessNode>(ProcessType::COMBINATIONAL);
                    proc->body = parse_statement_or_block();
                    mod->processes.push_back(std::move(proc));
                }
                continue;
            }

            // Module Instantiation: <mod_type> #(.PARAM(val)) inst_name (.port(sig), ...);
            if (current.type == TokenType::TOKEN_IDENTIFIER) {
                std::string mtype = current.text;
                advance();

                std::vector<std::unique_ptr<ExpressionNode>> param_overrides;
                if (match(TokenType::TOKEN_HASH)) {
                    expect(TokenType::TOKEN_LPAREN, "Expected '(' in parameter override list");
                    while (current.type != TokenType::TOKEN_RPAREN && current.type != TokenType::TOKEN_EOF) {
                        if (match(TokenType::TOKEN_DOT)) {
                            expect(TokenType::TOKEN_IDENTIFIER, "Expected parameter name after '.'");
                            expect(TokenType::TOKEN_LPAREN, "Expected '(' in named parameter override");
                            param_overrides.push_back(parse_expression());
                            expect(TokenType::TOKEN_RPAREN, "Expected ')' in named parameter override");
                        } else {
                            param_overrides.push_back(parse_expression());
                        }
                        if (match(TokenType::TOKEN_COMMA)) continue;
                        break;
                    }
                    expect(TokenType::TOKEN_RPAREN, "Expected ')' closing parameter override list");
                }

                std::string iname = current.text;
                expect(TokenType::TOKEN_IDENTIFIER, "Expected instance identifier");
                expect(TokenType::TOKEN_LPAREN, "Expected '(' in instance port connections");

                bool positional = false;
                std::vector<PortConnection> conns;
                while (current.type != TokenType::TOKEN_RPAREN && current.type != TokenType::TOKEN_EOF) {
                    if (match(TokenType::TOKEN_DOT)) {
                        std::string port_name = current.text;
                        expect(TokenType::TOKEN_IDENTIFIER, "Expected port name after '.'");
                        expect(TokenType::TOKEN_LPAREN, "Expected '(' in port connection");
                        auto expr = parse_expression();
                        expect(TokenType::TOKEN_RPAREN, "Expected ')' closing port connection");
                        conns.push_back({port_name, std::move(expr)});
                    } else {
                        positional = true;
                        auto expr = parse_expression();
                        conns.push_back({"", std::move(expr)});
                    }
                    if (match(TokenType::TOKEN_COMMA)) continue;
                    break;
                }
                expect(TokenType::TOKEN_RPAREN, "Expected ')' closing instance port connections");
                expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after instance");

                auto inst = std::make_unique<InstantiationNode>(mtype, iname, positional);
                inst->param_overrides = std::move(param_overrides);
                inst->connections = std::move(conns);
                mod->instantiations.push_back(std::move(inst));
                continue;
            }

            advance(); // Skip unexpected token safely
        }
        expect(TokenType::TOKEN_ENDMODULE, "Expected 'endmodule'");
        modules.push_back(std::move(mod));
    }

    return modules;
}

size_t Elaborator::get_or_create_net(const std::string& scoped_name, uint32_t width) {
    if (result.net_map.find(scoped_name) != result.net_map.end()) {
        return result.net_map[scoped_name];
    }
    size_t id = result.netlist.size();
    result.netlist.emplace_back(scoped_name, width);
    result.netlist.back().vcd_id = VcdDumper::generate_vcd_id(id);
    result.net_map[scoped_name] = id;
    return id;
}

uint64_t Elaborator::eval_const_expr(ExpressionNode* expr, const std::string& scope) {
    if (!expr) return 0;
    if (auto num = dynamic_cast<NumberNode*>(expr)) {
        return num->value;
    }
    if (auto id = dynamic_cast<IdentifierNode*>(expr)) {
        std::string scoped_id = scope.empty() ? id->name : (scope + "." + id->name);
        if (result.param_map.find(scoped_id) != result.param_map.end()) {
            return result.param_map.at(scoped_id);
        }
        if (result.param_map.find(id->name) != result.param_map.end()) {
            return result.param_map.at(id->name);
        }
        return 0;
    }
    if (auto bin = dynamic_cast<BinaryOpNode*>(expr)) {
        uint64_t l = eval_const_expr(bin->lhs.get(), scope);
        uint64_t r = eval_const_expr(bin->rhs.get(), scope);
        if (bin->op == "+") return l + r;
        if (bin->op == "-") return (l >= r) ? (l - r) : 0;
        if (bin->op == "*") return l * r;
        if (bin->op == "<<") return l << r;
        if (bin->op == ">>") return l >> r;
    }
    return 0;
}

NetSliceRef Elaborator::resolve_slice_ref(ExpressionNode* expr, const std::string& scope) {
    if (auto ident = dynamic_cast<IdentifierNode*>(expr)) {
        std::string full_name = scope.empty() ? ident->name : (scope + "." + ident->name);
        size_t id = result.net_map.at(full_name);
        return {id, 0, result.netlist[id].width};
    }
    if (auto idx = dynamic_cast<IndexSelectNode*>(expr)) {
        std::string full_name = scope.empty() ? idx->name : (scope + "." + idx->name);
        size_t id = result.net_map.at(full_name);
        if (!idx->is_part_select) {
            uint64_t off = eval_const_expr(idx->msb_expr.get(), scope);
            return {id, (uint32_t)off, 1};
        } else {
            uint64_t msb = eval_const_expr(idx->msb_expr.get(), scope);
            uint64_t lsb = eval_const_expr(idx->lsb_expr.get(), scope);
            uint32_t w = (msb >= lsb) ? (uint32_t)(msb - lsb + 1) : (uint32_t)(lsb - msb + 1);
            return {id, (uint32_t)lsb, w};
        }
    }
    if (auto num = dynamic_cast<NumberNode*>(expr)) {
        static size_t const_counter = 0;
        std::string const_net_name = scope + ".__const_port_" + std::to_string(const_counter++);
        size_t id = get_or_create_net(const_net_name, num->width);
        LogicVal val = num->is_unknown ? LogicVal::make_x(num->width) : LogicVal::make_known(num->width, num->value);
        result.netlist[id].set(val);
        return {id, 0, num->width};
    }

    throw std::runtime_error("Expression is not a bindable net target during elaboration");
}

void Elaborator::elaborate_module(ModuleNode* mod, const std::string& scope,
                                 const std::unordered_map<std::string, ModuleNode*>& module_defs,
                                 const std::vector<std::unique_ptr<ExpressionNode>>* overrides,
                                 const std::string& parent_scope,
                                 std::ostream* debug_out) {
    if (debug_out) {
        *debug_out << "  -> Scoping module '" << mod->name << "' as [" << scope << "]\n";
    }

    for (size_t i = 0; i < mod->parameters.size(); ++i) {
        const auto& p = mod->parameters[i];
        uint64_t val = 0;
        if (overrides && i < overrides->size()) {
            val = eval_const_expr((*overrides)[i].get(), parent_scope);
            if (debug_out) *debug_out << "     * Parameter override: " << p.name << " = " << val << "\n";
        } else {
            val = eval_const_expr(p.default_expr.get(), scope);
            if (debug_out) *debug_out << "     * Parameter default: " << p.name << " = " << val << "\n";
        }
        result.param_map[scope + "." + p.name] = val;
    }

    for (auto& p : mod->ports) {
        uint32_t w = 1;
        if (p.msb_expr && p.lsb_expr) {
            uint64_t msb = eval_const_expr(p.msb_expr.get(), scope);
            uint64_t lsb = eval_const_expr(p.lsb_expr.get(), scope);
            w = (msb >= lsb) ? (uint32_t)(msb - lsb + 1) : (uint32_t)(lsb - msb + 1);
        }
        get_or_create_net(scope + "." + p.name, w);
    }

    for (auto& s : mod->signals) {
        uint32_t w = 1;
        if (s.msb_expr && s.lsb_expr) {
            uint64_t msb = eval_const_expr(s.msb_expr.get(), scope);
            uint64_t lsb = eval_const_expr(s.lsb_expr.get(), scope);
            w = (msb >= lsb) ? (uint32_t)(msb - lsb + 1) : (uint32_t)(lsb - msb + 1);
        }
        if (s.is_array) {
            uint32_t depth = 1;
            if (s.bound1_expr && s.bound2_expr) {
                uint64_t b1 = eval_const_expr(s.bound1_expr.get(), scope);
                uint64_t b2 = eval_const_expr(s.bound2_expr.get(), scope);
                depth = (b1 >= b2) ? (uint32_t)(b1 - b2 + 1) : (uint32_t)(b2 - b1 + 1);
            } else if (s.bound1_expr) {
                depth = (uint32_t)eval_const_expr(s.bound1_expr.get(), scope);
            }
            std::string mem_scope = scope + "." + s.name;
            result.memory_map[mem_scope] = MemoryArray(mem_scope, w, depth);
            if (debug_out) {
                *debug_out << "     * Memory allocated: " << mem_scope 
                           << " [" << w << "-bit x " << depth << " words]\n";
            }
        } else {
            get_or_create_net(scope + "." + s.name, w);
        }
    }

    for (auto& ca : mod->continuous_assigns) {
        std::string scoped_target = scope + "." + ca->target_name;
        size_t target_id = result.net_map.at(scoped_target);
        result.continuous_assigns.push_back({scope, {target_id, 0, result.netlist[target_id].width}, ca->rhs.get()});
    }

    for (auto& g : mod->gates) {
        GateInstance gi;
        gi.type = g->type;
        gi.scope = scope;
        gi.out_target = resolve_slice_ref(g->out_expr.get(), scope);
        for (auto& ie : g->in_exprs) gi.in_exprs.push_back(ie.get());
        result.gates.push_back(gi);
    }

    for (auto& t : mod->tasks) {
        result.task_map[scope + "." + t->name] = t.get();
    }

    for (auto& proc : mod->processes) {
        result.processes.push_back({scope, proc.get(), 0, false});
    }

    for (auto& inst : mod->instantiations) {
        std::string child_scope = scope + "." + inst->instance_name;
        if (module_defs.find(inst->module_type) == module_defs.end()) {
            throw std::runtime_error("Undefined module type instantiated: " + inst->module_type);
        }
        ModuleNode* submod = module_defs.at(inst->module_type);

        elaborate_module(submod, child_scope, module_defs, &inst->param_overrides, scope, debug_out);

        for (size_t conn_idx = 0; conn_idx < inst->connections.size(); ++conn_idx) {
            auto& conn = inst->connections[conn_idx];
            std::string target_port_name;
            if (inst->is_positional) {
                if (conn_idx >= submod->ports.size()) {
                    throw std::runtime_error("Positional connection out of bounds in " + inst->instance_name);
                }
                target_port_name = submod->ports[conn_idx].name;
            } else {
                target_port_name = conn.port_name;
            }

            std::string dir = "input";
            for (auto& p : submod->ports) {
                if (p.name == target_port_name) {
                    dir = p.direction;
                    break;
                }
            }

            NetSliceRef parent_ref = resolve_slice_ref(conn.expr.get(), scope);
            size_t child_net_id = result.net_map.at(child_scope + "." + target_port_name);
            NetSliceRef child_ref = {child_net_id, 0, result.netlist[child_net_id].width};

            if (dir == "input") {
                result.slice_bindings.push_back({parent_ref, child_ref});
            } else {
                result.slice_bindings.push_back({child_ref, parent_ref});
            }
        }
    }
}

ElaborationResult Elaborator::elaborate(std::vector<std::unique_ptr<ModuleNode>>& modules,
                                       const std::string& top_module_name,
                                       std::ostream* debug_out) {
    if (debug_out) *debug_out << "[PHASE 3: Elaboration] Synthesizing Netlist Hierarchy...\n";
    result = ElaborationResult();

    std::unordered_map<std::string, ModuleNode*> module_defs;
    for (auto& mod : modules) {
        module_defs[mod->name] = mod.get();
    }

    ModuleNode* top = nullptr;

    if (!top_module_name.empty()) {
        if (module_defs.find(top_module_name) != module_defs.end()) {
            top = module_defs.at(top_module_name);
            if (debug_out) *debug_out << "  -> Using explicit top module: '" << top->name << "'\n";
        } else {
            throw std::runtime_error("Elaboration error: Top module '" + top_module_name + "' was not found in parsed sources.");
        }
    } else {
        std::unordered_map<std::string, bool> is_instantiated;
        for (auto& mod : modules) {
            for (auto& inst : mod->instantiations) {
                is_instantiated[inst->module_type] = true;
            }
        }
        for (auto& mod : modules) {
            if (!is_instantiated[mod->name]) {
                top = mod.get();
                break;
            }
        }
        if (!top) top = modules.front().get();
        if (debug_out) *debug_out << "  -> Auto-detected top-level root module: '" << top->name << "'\n";
    }

    elaborate_module(top, top->name, module_defs, nullptr, "", debug_out);
    return std::move(result);
}

LogicVal Evaluator::read_slice(const NetSliceRef& ref) {
    const auto& net_val = engine.netlist[ref.net_id].value;
    uint64_t mask = (ref.width >= 64) ? ~0ULL : ((1ULL << ref.width) - 1);
    uint64_t a = (net_val.aval >> ref.offset) & mask;
    uint64_t b = (net_val.bval >> ref.offset) & mask;
    return LogicVal(ref.width, a, b);
}

void Evaluator::write_slice(const NetSliceRef& ref, const LogicVal& val) {
    auto cur = engine.netlist[ref.net_id].value;
    uint64_t mask = (ref.width >= 64) ? ~0ULL : ((1ULL << ref.width) - 1);
    uint64_t clear_mask = ~(mask << ref.offset);

    uint64_t new_a = (cur.aval & clear_mask) | ((val.aval & mask) << ref.offset);
    uint64_t new_b = (cur.bval & clear_mask) | ((val.bval & mask) << ref.offset);
    engine.netlist[ref.net_id].set(LogicVal(cur.width, new_a, new_b));
}

LogicVal Evaluator::eval_expr(ExpressionNode* expr, const std::string& scope) {
    if (auto num = dynamic_cast<NumberNode*>(expr)) {
        if (num->is_unknown) return LogicVal::make_x(num->width);
        return LogicVal::make_known(num->width, num->value);
    }
    if (auto ident = dynamic_cast<IdentifierNode*>(expr)) {
        if (ident->name == "$time") return LogicVal::make_known(64, engine.sim_time);
        std::string scoped_ident = scope.empty() ? ident->name : (scope + "." + ident->name);
        if (engine.param_map.find(scoped_ident) != engine.param_map.end()) {
            return LogicVal::make_known(32, engine.param_map.at(scoped_ident));
        }
        if (engine.net_map.find(scoped_ident) != engine.net_map.end()) {
            return engine.netlist[engine.net_map.at(scoped_ident)].value;
        }
        if (engine.param_map.find(ident->name) != engine.param_map.end()) {
            return LogicVal::make_known(32, engine.param_map.at(ident->name));
        }
        return LogicVal::make_x(1);
    }
    if (auto idx_node = dynamic_cast<IndexSelectNode*>(expr)) {
        std::string scoped_ident = scope.empty() ? idx_node->name : (scope + "." + idx_node->name);
        if (engine.memory_map.find(scoped_ident) != engine.memory_map.end()) {
            const auto& mem = engine.memory_map.at(scoped_ident);
            LogicVal addr_val = eval_expr(idx_node->msb_expr.get(), scope);
            if (addr_val.is_unknown() || addr_val.aval >= mem.depth) {
                return LogicVal::make_x(mem.word_width);
            }
            return mem.words[addr_val.aval];
        }

        const auto& target_val = engine.netlist[engine.net_map.at(scoped_ident)].value;
        if (!idx_node->is_part_select) {
            LogicVal idx_val = eval_expr(idx_node->msb_expr.get(), scope);
            if (idx_val.is_unknown() || idx_val.aval >= target_val.width) return LogicVal::make_x(1);
            uint32_t sh = idx_val.aval;
            return LogicVal(1, (target_val.aval >> sh) & 1, (target_val.bval >> sh) & 1);
        } else {
            LogicVal msb_v = eval_expr(idx_node->msb_expr.get(), scope);
            LogicVal lsb_v = eval_expr(idx_node->lsb_expr.get(), scope);
            if (msb_v.is_unknown() || lsb_v.is_unknown() || msb_v.aval < lsb_v.aval) {
                uint32_t w = (msb_v.aval >= lsb_v.aval) ? (msb_v.aval - lsb_v.aval + 1) : 1;
                return LogicVal::make_x(w);
            }
            uint32_t msb = msb_v.aval;
            uint32_t lsb = lsb_v.aval;
            uint32_t slice_w = msb - lsb + 1;
            uint64_t mask = (slice_w >= 64) ? ~0ULL : ((1ULL << slice_w) - 1);
            return LogicVal(slice_w, (target_val.aval >> lsb) & mask, (target_val.bval >> lsb) & mask);
        }
    }
    if (auto un = dynamic_cast<UnaryOpNode*>(expr)) {
        LogicVal val = eval_expr(un->operand.get(), scope);
        if (un->op == '!') {
            if (val.is_unknown()) return LogicVal::make_x(1);
            return LogicVal::make_known(1, (val.aval == 0) ? 1 : 0);
        }
        if (un->op == '~') return val.bitwise_not();
    }
    if (auto bin = dynamic_cast<BinaryOpNode*>(expr)) {
        LogicVal l = eval_expr(bin->lhs.get(), scope);
        LogicVal r = eval_expr(bin->rhs.get(), scope);

        if (bin->op == "+") {
            uint32_t w = std::max(l.width, r.width);
            if (l.is_unknown() || r.is_unknown()) return LogicVal::make_x(w);
            return LogicVal::make_known(w, l.aval + r.aval);
        }
        if (bin->op == "-") {
            uint32_t w = std::max(l.width, r.width);
            if (l.is_unknown() || r.is_unknown()) return LogicVal::make_x(w);
            return LogicVal::make_known(w, l.aval - r.aval);
        }
        if (bin->op == "&")  return LogicVal::bitwise_and(l, r);
        if (bin->op == "|")  return LogicVal::bitwise_or(l, r);
        if (bin->op == "^")  return LogicVal::bitwise_xor(l, r);
        if (bin->op == "<<") return LogicVal::shift_left(l, r);
        if (bin->op == ">>") return LogicVal::shift_right(l, r);
        if (bin->op == "==") return LogicVal::logical_eq(l, r);
        if (bin->op == "!=") return LogicVal::logical_neq(l, r);
        if (bin->op == "<")  return LogicVal::rel_lt(l, r);
        if (bin->op == "<=") return LogicVal::rel_le(l, r);
        if (bin->op == ">")  return LogicVal::rel_gt(l, r);
        if (bin->op == ">=") return LogicVal::rel_ge(l, r);
        if (bin->op == "&&") {
            if ((!l.is_unknown() && l.aval == 0) || (!r.is_unknown() && r.aval == 0)) return LogicVal::make_known(1, 0);
            if (l.is_unknown() || r.is_unknown()) return LogicVal::make_x(1);
            return LogicVal::make_known(1, (l.aval != 0 && r.aval != 0) ? 1 : 0);
        }
        if (bin->op == "||") {
            if ((!l.is_unknown() && l.aval != 0) || (!r.is_unknown() && r.aval != 0)) return LogicVal::make_known(1, 1);
            if (l.is_unknown() || r.is_unknown()) return LogicVal::make_x(1);
            return LogicVal::make_known(1, (l.aval != 0 || r.aval != 0) ? 1 : 0);
        }
    }
    if (auto tern = dynamic_cast<TernaryOpNode*>(expr)) {
        LogicVal cond = eval_expr(tern->condition.get(), scope);
        if (cond.is_unknown()) {
            LogicVal t_val = eval_expr(tern->true_expr.get(), scope);
            LogicVal f_val = eval_expr(tern->false_expr.get(), scope);
            return LogicVal::make_x(std::max(t_val.width, f_val.width));
        }
        return cond.to_bool() ? eval_expr(tern->true_expr.get(), scope) : eval_expr(tern->false_expr.get(), scope);
    }
    return LogicVal::make_x(1);
}

void Evaluator::check_monitor() {
    if (!engine.monitor_session.active) return;

    bool values_changed = false;
    std::vector<LogicVal> current_values;

    for (const auto& arg_name : engine.monitor_session.args) {
        if (arg_name == "$time") {
            current_values.push_back(LogicVal::make_known(64, engine.sim_time));
        } else {
            std::string scoped_arg = engine.monitor_session.scope.empty() ? arg_name : (engine.monitor_session.scope + "." + arg_name);
            if (engine.net_map.find(scoped_arg) != engine.net_map.end()) {
                current_values.push_back(engine.netlist[engine.net_map.at(scoped_arg)].value);
            } else {
                current_values.push_back(LogicVal::make_x(1));
            }
        }
    }

    if (engine.monitor_session.prev_values.empty()) {
        values_changed = true;
    } else {
        for (size_t i = 0; i < current_values.size(); ++i) {
            if (current_values[i] != engine.monitor_session.prev_values[i]) {
                values_changed = true;
                break;
            }
        }
    }

    if (values_changed) {
        engine.monitor_session.prev_values = current_values;
        if (engine.monitor_session.has_leading_time) {
            std::cout << engine.sim_time;
        }

        size_t arg_idx = 0;
        size_t fmt_len = engine.monitor_session.format_string.size();
        for (size_t i = 0; i < fmt_len; ++i) {
            if (engine.monitor_session.format_string[i] == '%' && i + 1 < fmt_len) {
                char spec = engine.monitor_session.format_string[++i];
                if (spec == '0' && i + 1 < fmt_len) {
                    char next_spec = engine.monitor_session.format_string[i + 1];
                    if (next_spec == 't' || next_spec == 'h' || next_spec == 'd' || next_spec == 'b') {
                        spec = next_spec;
                        ++i;
                    }
                }

                if (arg_idx < current_values.size()) {
                    const auto& val = current_values[arg_idx++];
                    if (val.is_unknown()) {
                        std::cout << "x";
                    } else {
                        if (spec == 'd' || spec == 't') std::cout << val.aval;
                        else if (spec == 'b') std::cout << val.to_bin_string();
                        else if (spec == 'h') std::cout << std::hex << val.aval << std::dec;
                        else std::cout << val.aval;
                    }
                }
            } else {
                std::cout << engine.monitor_session.format_string[i];
            }
        }
        std::cout << "\n";
    }
}

void Evaluator::execute_statement(StatementNode* stmt, BoundProcess& bp) {
    if (auto if_stmt = dynamic_cast<IfStatementNode*>(stmt)) {
        LogicVal cond = eval_expr(if_stmt->condition.get(), bp.scope);
        if (cond.to_bool()) {
            for (auto& s : if_stmt->then_branch) execute_statement(s.get(), bp);
        } else {
            for (auto& s : if_stmt->else_branch) execute_statement(s.get(), bp);
        }
    }
    else if (auto case_stmt = dynamic_cast<CaseStatementNode*>(stmt)) {
        LogicVal sel_val = eval_expr(case_stmt->selector.get(), bp.scope);
        bool branch_taken = false;
        const CaseItem* default_item = nullptr;

        for (const auto& item : case_stmt->items) {
            if (item.is_default) {
                default_item = &item;
                continue;
            }
            LogicVal item_val = eval_expr(item.match_expr.get(), bp.scope);
            if (LogicVal::logical_eq(sel_val, item_val).to_bool()) {
                for (auto& s : item.statements) execute_statement(s.get(), bp);
                branch_taken = true;
                break;
            }
        }
        if (!branch_taken && default_item) {
            for (auto& s : default_item->statements) execute_statement(s.get(), bp);
        }
    }
    else if (auto tcall = dynamic_cast<TaskCallNode*>(stmt)) {
        std::string scoped_tname = bp.scope + "." + tcall->task_name;
        if (engine.task_map.find(scoped_tname) != engine.task_map.end()) {
            TaskNode* target_task = engine.task_map.at(scoped_tname);
            for (auto& s : target_task->body) {
                execute_statement(s.get(), bp);
                if (!engine.simulation_active) return;
                engine.propagate_comb();
                engine.evaluate_sequential_blocks();
                engine.commit_nba_updates();
            }
        }
    }
    else if (auto delay = dynamic_cast<DelayStatementNode*>(stmt)) {
        engine.sim_time += delay->delay;
        engine.vcd_dumper.dump_sample(engine.netlist, engine.sim_time);
        check_monitor();
    }
    else if (auto disp = dynamic_cast<DisplayStatementNode*>(stmt)) {
        if (disp->is_monitor) {
            engine.monitor_session.active = true;
            engine.monitor_session.scope = bp.scope;
            engine.monitor_session.format_string = disp->format_string;
            engine.monitor_session.args = disp->args;
            engine.monitor_session.has_leading_time = disp->has_leading_time;
            engine.monitor_session.prev_values.clear();
            check_monitor();
        } else {
            std::cout << "[CAS $display] ";
            if (disp->has_leading_time) {
                std::cout << engine.sim_time << " ";
            }
            size_t arg_idx = 0;
            size_t fmt_len = disp->format_string.size();
            for (size_t i = 0; i < fmt_len; ++i) {
                if (disp->format_string[i] == '%' && i + 1 < fmt_len) {
                    char spec = disp->format_string[++i];
                    if (spec == '0' && i + 1 < fmt_len) {
                        char next_spec = disp->format_string[i + 1];
                        if (next_spec == 't' || next_spec == 'h' || next_spec == 'd' || next_spec == 'b') {
                            spec = next_spec;
                            ++i;
                        }
                    }

                    if (arg_idx < disp->args.size()) {
                        std::string arg_name = disp->args[arg_idx++];
                        if (arg_name == "$time") {
                            std::cout << engine.sim_time;
                        } else {
                            std::string scoped_arg = bp.scope.empty() ? arg_name : (bp.scope + "." + arg_name);
                            if (engine.net_map.find(scoped_arg) != engine.net_map.end()) {
                                size_t net_id = engine.net_map.at(scoped_arg);
                                const auto& net = engine.netlist[net_id];
                                if (net.value.is_unknown()) {
                                    std::cout << "x";
                                } else {
                                    if (spec == 'd' || spec == 't') std::cout << net.value.aval;
                                    else if (spec == 'b') std::cout << net.value.to_bin_string();
                                    else if (spec == 'h') std::cout << std::hex << net.value.aval << std::dec;
                                    else std::cout << net.value.aval;
                                }
                            } else if (engine.param_map.find(scoped_arg) != engine.param_map.end()) {
                                std::cout << engine.param_map.at(scoped_arg);
                            } else if (engine.param_map.find(arg_name) != engine.param_map.end()) {
                                std::cout << engine.param_map.at(arg_name);
                            } else {
                                std::cout << "x";
                            }
                        }
                    }
                } else {
                    std::cout << disp->format_string[i];
                }
            }
            std::cout << "\n";
        }
    }
    else if (dynamic_cast<FinishStatementNode*>(stmt)) {
        std::cout << "[CAS $finish] Simulation Terminated at time " << engine.sim_time << ".\n";
        engine.simulation_active = false;
        bp.completed = true;
    }
    else if (auto nassign = dynamic_cast<NonBlockingAssignNode*>(stmt)) {
        LogicVal rhs_val = eval_expr(nassign->rhs.get(), bp.scope);
        std::string scoped_target = bp.scope + "." + nassign->target;
        size_t net_id = engine.net_map.at(scoped_target);
        engine.nba_queue.push_back({net_id, rhs_val});
    }
    else if (auto mem_assign = dynamic_cast<MemAssignNode*>(stmt)) {
        std::string scoped_mem = bp.scope + "." + mem_assign->mem_name;
        LogicVal addr_val = eval_expr(mem_assign->addr_expr.get(), bp.scope);
        LogicVal rhs_val = eval_expr(mem_assign->rhs.get(), bp.scope);

        if (!addr_val.is_unknown()) {
            uint32_t addr = addr_val.aval;
            if (mem_assign->is_non_blocking) {
                engine.nba_mem_queue.push_back({scoped_mem, addr, rhs_val});
            } else {
                if (engine.memory_map.find(scoped_mem) != engine.memory_map.end()) {
                    auto& mem = engine.memory_map.at(scoped_mem);
                    if (addr < mem.depth) mem.words[addr] = rhs_val;
                }
            }
        }
    }
    else if (auto assign = dynamic_cast<BlockingAssignNode*>(stmt)) {
        LogicVal rhs_val = eval_expr(assign->rhs.get(), bp.scope);
        if (assign->lhs_targets.size() > 1) {
            int shift = 0;
            for (int i = (int)assign->lhs_targets.size() - 1; i >= 0; --i) {
                std::string scoped_target = bp.scope + "." + assign->lhs_targets[i];
                size_t net_id = engine.net_map.at(scoped_target);
                uint32_t width = engine.netlist[net_id].width;
                uint64_t mask = (width >= 64) ? ~0ULL : ((1ULL << width) - 1);
                engine.netlist[net_id].set(LogicVal(width, (rhs_val.aval >> shift) & mask, (rhs_val.bval >> shift) & mask));
                shift += width;
            }
        } else {
            std::string scoped_target = bp.scope + "." + assign->lhs_targets[0];
            size_t net_id = engine.net_map.at(scoped_target);
            engine.netlist[net_id].set(rhs_val);
        }
    }
}

int main(int argc, char* argv[]) {
    if (argc < 2) {
        std::cerr << "Usage: cas [options] <source_files...>\n";
        return 1;
    }

    bool enable_profiling = true;
    bool verbose = true;
    bool enable_vcd = true;
    std::string top_override = "";
    std::string timescale_unit = "1ns";
    std::vector<std::string> source_files;

    for (int i = 1; i < argc; ++i) {
        if (std::strcmp(argv[i], "--no-profile") == 0) {
            enable_profiling = false;
        } else if (std::strcmp(argv[i], "--quiet") == 0) {
            verbose = false;
        } else if (std::strcmp(argv[i], "--no-vcd") == 0) {
            enable_vcd = false;
        } else if (std::strcmp(argv[i], "--top") == 0 && i + 1 < argc) {
            top_override = argv[++i];
        } else if (std::strcmp(argv[i], "--timescale") == 0 && i + 1 < argc) {
            timescale_unit = argv[++i];
        } else {
            source_files.push_back(argv[i]);
        }
    }

    if (source_files.empty()) {
        std::cerr << "[CAS ERROR] No source files provided.\n";
        return 1;
    }

    Profiler profiler(enable_profiling, verbose, enable_vcd);
    profiler.start();

    std::stringstream debug_ss;
    std::ostream* debug_ptr = verbose ? &debug_ss : nullptr;

    if (verbose) {
        debug_ss << "==================================================================\n";
        debug_ss << " CAS COMPILER & PREPROCESSOR FILE ACCESS TRACE\n";
        debug_ss << "==================================================================\n";
        debug_ss << "[PHASE 1: Ingestion & Preprocessor Hierarchy]\n";
        debug_ss << "  -> CLI Root Entry: '" << source_files[0] << "'\n";
    }

    std::vector<std::unique_ptr<ModuleNode>> all_modules;
    size_t total_bytes = 0;

    try {
        for (size_t f_idx = 0; f_idx < source_files.size(); ++f_idx) {
            const auto& file = source_files[f_idx];
            std::ifstream f(file);
            if (!f.is_open()) {
                std::cerr << "[CAS ERROR] Cannot open source file: " << file << "\n";
                return 1;
            }
            std::stringstream ss;
            ss << f.rdbuf();
            std::string content = ss.str();
            total_bytes += content.size();

            size_t slash = file.find_last_of("/\\");
            std::string base_dir = (slash == std::string::npos) ? "." : file.substr(0, slash);

            if (verbose && f_idx > 0) {
                debug_ss << "  [Source Accessed] #" << (f_idx + 1) << ": '" << file << "' (" << content.size() << " bytes)\n";
            }

            Lexer lexer(file, content, base_dir, debug_ptr);
            Parser parser(lexer);
            auto mods = parser.parse_all();
            for (auto& m : mods) {
                all_modules.push_back(std::move(m));
            }
        }

        profiler.set_source_metrics(total_bytes, all_modules.size());

        if (verbose) {
            debug_ss << "\n[PHASE 2: Parsed AST & Module Origin Attribution]\n";
            debug_ss << "  -> Discovered " << all_modules.size() << " module definition(s) across source files:\n";
            for (const auto& mod : all_modules) {
                debug_ss << "     * module '" << mod->name << "'\n"
                         << "       - Defined in : " << mod->defined_in_file << ":" << mod->defined_on_line << "\n"
                         << "       - Interface  : " << mod->ports.size() << " port(s), " 
                                                    << mod->parameters.size() << " parameter(s)\n"
                         << "       - Internal   : " << mod->signals.size() << " net(s), " 
                                                    << mod->gates.size() << " gate(s), " 
                                                    << mod->processes.size() << " process(es)\n";
            }
            debug_ss << "\n";
        }

        // ----------------------------------------------------
        // PHASE 3 & 4: Elaboration & Simulation Execution
        // ----------------------------------------------------
        Engine engine;
        engine.set_vcd_allowed(enable_vcd);
        engine.elaborate(all_modules, top_override, debug_ptr);
        engine.run(debug_ptr);

        // ----------------------------------------------------
        // PERFORMANCE & PROFILING REPORT
        // ----------------------------------------------------
        ProfileMetrics metrics = profiler.sample(engine.get_stats());

        Profiler::emit_json_telemetry(metrics, std::cout);

        std::cout << "\n---BEGIN_CAS_DEBUG_LOG---\n";
        if (verbose) {
            std::cout << debug_ss.str();
        }
        std::cout << "---END_CAS_DEBUG_LOG---\n";

    } catch (const std::exception& e) {
        std::cerr << "\n" << e.what() << "\n";
        return 2;
    }

    return 0;
}

    source_bytes_total = bytes;
    modules_parsed_total = modules;

ProfileMetrics Profiler::sample(const EngineStats& stats) {
    auto end_time = std::chrono::high_resolution_clock::now();
    double elapsed_wall_ms = std::chrono::duration<double, std::milli>(end_time - start_time).count();

    ProfileMetrics m;
    m.profiling_enabled = enabled;
    m.debug_enabled = debug_mode;
    m.vcd_enabled = vcd_mode;
    m.wallclock_ms = elapsed_wall_ms;
    m.source_bytes = source_bytes_total;
    m.modules_count = modules_parsed_total;
    m.engine_stats = stats;

    if (enabled) {
        struct rusage usage;
        getrusage(RUSAGE_SELF, &usage);

        m.user_cpu_ms = (usage.ru_utime.tv_sec * 1000.0) + (usage.ru_utime.tv_usec / 1000.0);
        m.sys_cpu_ms  = (usage.ru_stime.tv_sec * 1000.0) + (usage.ru_stime.tv_usec / 1000.0);
        m.total_cpu_ms = m.user_cpu_ms + m.sys_cpu_ms;
        m.cpu_util_pct = (elapsed_wall_ms > 0) ? (m.total_cpu_ms / elapsed_wall_ms) * 100.0 : 0.0;
        m.peak_rss_kb = usage.ru_maxrss;
    }

    return m;
}

std::string VcdDumper::generate_vcd_id(size_t id) {
    std::string res;
    size_t base = 94;
    do {
        res += (char)(33 + (id % base));
        id /= base;
    } while (id > 0);
    return res;
}

void VcdDumper::enable(const std::string& fname) {
    if (!allowed) return;
    filename = fname;
    enabled = true;
}

void VcdDumper::init_header(std::vector<Net>& netlist, uint64_t initial_time) {
    if (!enabled || !allowed || file.is_open()) return;

    file.open(filename);
    if (!file.is_open()) {
        std::cerr << "[CAS ERROR] Unable to create VCD file: " << filename << "\n";
        enabled = false;
        return;
    }

    std::time_t now = std::time(nullptr);
    file << "$date\n   " << std::asctime(std::localtime(&now)) << "$end\n";
    file << "$version\n   CAS (Cycle Accurate Simulator) v1.0.0 [Production Kernel]\n$end\n";
    file << "$timescale 1ns $end\n";

    // Assign IDs and group by hierarchical scope
    std::unordered_map<std::string, std::vector<size_t>> scopes;
    for (size_t i = 0; i < netlist.size(); ++i) {
        if (netlist[i].vcd_id.empty()) {
            netlist[i].vcd_id = generate_vcd_id(i);
        }
        std::string full_name = netlist[i].name;
        size_t dot_pos = full_name.rfind('.');
        std::string scope_name = (dot_pos == std::string::npos) ? "top" : full_name.substr(0, dot_pos);
        scopes[scope_name].push_back(i);
    }

    for (const auto& kv : scopes) {
        file << "$scope module " << kv.first << " $end\n";
        for (size_t net_id : kv.second) {
            const auto& net = netlist[net_id];
            size_t dot_pos = net.name.rfind('.');
            std::string leaf_name = (dot_pos == std::string::npos) ? net.name : net.name.substr(dot_pos + 1);
            file << "$var wire " << net.width << " " << net.vcd_id << " " << leaf_name << " $end\n";
        }
        file << "$upscope $end\n";
    }

    file << "$enddefinitions $end\n";
    file << "$dumpvars\n";
    dump_sample(netlist, initial_time, true);
    file << "$end\n";
}

void VcdDumper::dump_sample(std::vector<Net>& netlist, uint64_t current_time, bool force_dump) {
    if (!enabled || !allowed || !file.is_open()) return;

    if (!force_dump && current_time != last_dump_time) {
        file << "#" << current_time << "\n";
        last_dump_time = current_time;
    }

    for (auto& net : netlist) {
        if (force_dump || net.modified) {
            if (net.width == 1) {
                if (net.value.is_unknown()) file << "x" << net.vcd_id << "\n";
                else file << (net.value.aval & 1) << net.vcd_id << "\n";
            } else {
                file << "b" << net.value.to_bin_string() << " " << net.vcd_id << "\n";
            }
        }
    }
}

void VcdDumper::close() {
    if (file.is_open()) {
        file.close();
    }
}

// register_file.sv - Dual-port RAM array with synchronous write & asynchronous read
module register_file #(
    parameter DATA_WIDTH = 8,
    parameter ADDR_WIDTH = 4,
    parameter DEPTH      = 16
)(
    input  logic                  clk,
    input  logic                  rst_n,
    input  logic                  wr_en,
    input  logic [ADDR_WIDTH-1:0] wr_addr,
    input  logic [DATA_WIDTH-1:0] wr_data,
    input  logic [ADDR_WIDTH-1:0] rd_addr,
    output logic [DATA_WIDTH-1:0] rd_data
);
    logic [DATA_WIDTH-1:0] mem [0:DEPTH-1];

    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            // Reset clears bottom storage cell
            mem[0] <= 0;
        end else if (wr_en) begin
            mem[wr_addr] <= wr_data;
        end
    end

    // Asynchronous memory array read
    assign rd_data = mem[rd_addr];

endmodule

`include "alu_defs.vh"
`include "ripple_adder.sv"
`include "register_file.sv"

module top_mac (
    input  logic       clk,
    input  logic       rst_n,
    input  logic [2:0] opcode,
    input  logic [3:0] op_a,
    input  logic [3:0] op_b,
    input  logic       cin,
    input  logic       mem_write,
    input  logic [3:0] mem_addr,
    output logic [3:0] alu_result,
    output logic       alu_cout,
    output logic [7:0] mem_readback
);
    logic [3:0] add_sum;
    logic       add_cout;
    logic [3:0] logic_out;

    // Instantiate 4-bit Ripple Adder
    ripple_adder #(.WIDTH(4)) u_adder (
        .a(op_a),
        .b(op_b),
        .cin(cin),
        .sum(add_sum),
        .cout(add_cout)
    );

    // Instantiate Register File Storage
    register_file #(
        .DATA_WIDTH(8),
        .ADDR_WIDTH(4),
        .DEPTH(16)
    ) u_rf (
        .clk(clk),
        .rst_n(rst_n),
        .wr_en(mem_write),
        .wr_addr(mem_addr),
        .wr_data({add_cout, 3'b000, add_sum}),
        .rd_addr(mem_addr),
        .rd_data(mem_readback)
    );

    // Combinational ALU Multiplexer
    always_comb begin
        case (opcode)
            ALU_ADD: begin
                alu_result = add_sum;
                alu_cout   = add_cout;
            end
            ALU_SUB: begin
                alu_result = op_a - op_b;
                alu_cout   = (op_a >= op_b) ? 1'b1 : 1'b0;
            end
            ALU_AND: begin
                alu_result = op_a & op_b;
                alu_cout   = 1'b0;
            end
            ALU_OR: begin
                alu_result = op_a | op_b;
                alu_cout   = 1'b0;
            end
            ALU_XOR: begin
                alu_result = op_a ^ op_b;
                alu_cout   = 1'b0;
            end
            ALU_SHL: begin
                alu_result = op_a << op_b[1:0];
                alu_cout   = 1'b0;
            end
            ALU_SHR: begin
                alu_result = op_a >> op_b[1:0];
                alu_cout   = 1'b0;
            end
            default: begin
                alu_result = op_a;
                alu_cout   = 1'b0;
            end
        endcase
    end

endmodule

module dff (clk, reset,
  d, q, qb);
  input      clk;
  input      reset;
  input      d;
  output     q;
  output     qb;

  reg        q;

  assign qb = ~q;

  always @(posedge clk or posedge reset)
  begin
    if (reset) begin
      // Asynchronous reset when reset goes high
      q <= 1'b0;
    end else begin
      // Assign D to Q on positive clock edge
      q <= d;
    end
  end
endmodule

/* 
 * Random Access Memory (RAM) with
 * 1 read port and 1 write port
 */
module ram (clk_write, address_write,
  data_write, write_enable,
  clk_read, address_read, data_read);
  
  parameter D_WIDTH = 16;
  parameter A_WIDTH = 4;
  parameter A_MAX = 16; // 2^A_WIDTH

  // Write port
  input                clk_write;
  input  [A_WIDTH-1:0] address_write;
  input  [D_WIDTH-1:0] data_write;
  input                write_enable;

  // Read port
  input                clk_read;
  input  [A_WIDTH-1:0] address_read;
  output [D_WIDTH-1:0] data_read;
  
  reg    [D_WIDTH-1:0] data_read;
  
  // Memory as multi-dimensional array
  reg [D_WIDTH-1:0] memory [A_MAX-1:0];

  // Write data to memory
  always @(posedge clk_write) begin
    if (write_enable) begin
      memory[address_write] <= data_write;
    end
  end

  // Read data from memory
  always @(posedge clk_read) begin
    data_read <= memory[address_read];
  end

endmodule

module alu_core (
    input  logic [3:0] op_a,
    input  logic [3:0] op_b,
    input  logic [2:0] opcode,
    output logic [3:0] alu_out
);

always_comb begin
    case (opcode)
        3'd0: alu_out = op_a & op_b;
        3'd1: alu_out = op_a | op_b;
        3'd2: alu_out = op_a ^ op_b;
        3'd3: alu_out = ~op_a;
        3'd4: alu_out = op_a << 4'd1;
        3'd5: alu_out = op_a >> 4'd1;
        3'd6: alu_out = op_a + op_b;
        3'd7: alu_out = op_a - op_b;
        default: alu_out = 4'd0;
    endcase
end

endmodule

// alu_core.sv
module alu_core (
    input  logic [3:0] op_a,
    input  logic [3:0] op_b,
    input  logic [2:0] opcode,
    output logic [3:0] alu_out
);
    always_comb begin
        case (opcode)
            3'd0: alu_out = op_a & op_b;
            3'd1: alu_out = op_a | op_b;
            3'd2: alu_out = op_a ^ op_b;
            3'd3: alu_out = ~op_a;
            3'd4: alu_out = op_a << 4'd1;
            3'd5: alu_out = op_a >> 4'd1;
            3'd6: alu_out = op_a + op_b;
            3'd7: alu_out = op_a - op_b;
            default: alu_out = 4'd0;
        endcase
    end
endmodule

// 4. Top-level unified DUT exercising memories, dataflow, sequential logic, and submodules
module master_dut (
    input  logic       clk,
    input  logic       rst_n,
    input  logic [3:0] in_a,
    input  logic [3:0] in_b,
    input  logic [2:0] opcode,
    input  logic [1:0] mem_addr,
    input  logic       mem_we,
    output logic [3:0] adder_sum,
    output logic       adder_cout,
    output logic [3:0] alu_result,
    output logic [3:0] reg_accum,
    output logic       is_equal,
    output logic       is_less,
    output logic       bit_extracted,
    output logic [1:0] slice_extracted,
    output logic [3:0] cont_assign_out,
    output logic [3:0] mem_rdata,
    output logic       dff_q,
    output logic       dff_qb
);

    // v0.9.0 Continuous assign dataflow
    assign cont_assign_out = in_a + 4'd1;

    // v0.9.0 1D Unpacked Memory Array
    logic [3:0] ram [0:3];
    assign mem_rdata = ram[mem_addr];

    // v0.9.3 Submodule positional slice instantiation
    bit4_adder u_adder (adder_sum, adder_cout, in_a, in_b, 1'b0);

    // v0.2.0 Submodule named instantiation
    alu_core u_alu (
        .op_a(in_a),
        .op_b(in_b),
        .opcode(opcode),
        .alu_out(alu_result)
    );

    // v0.7.0 / v0.8.0 Continuous relational expressions and bit-slicing
    always_comb begin
        is_equal = (in_a == in_b) ? 1'd1 : 1'd0;
        is_less  = (in_a < in_b)  ? 1'd1 : 1'd0;
        bit_extracted   = in_a[2];
        slice_extracted = in_a[3:2];
    end

    // v0.4.0 / v0.9.2 Generic always with edge sensitivity list (clk or negedge rst_n)
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            reg_accum <= 4'd0;
            dff_q     <= 1'b0;
        end else begin
            reg_accum <= reg_accum + alu_result;
            dff_q     <= in_a[0];
            if (mem_we) begin
                ram[mem_addr] <= in_a;
            end
        end
    end

    assign dff_qb = ~dff_q;

endmodule

# CAS: Cycle-Accurate Simulator

**Current Version:** `v0.9.0`  
**License:** MIT  
**Target Standard:** IEEE 1800 SystemVerilog (subset)

CAS is an independent, ground-up discrete-event RTL simulator implemented in C++17. It operates without external compiler chains (LLVM, intermediate C++ transpilation) or third-party frontends. CAS parses SystemVerilog sources directly into an in-memory Abstract Syntax Tree (AST), elaborates arbitrary-depth hierarchical netlist graphs, and executes simulation events using an IEEE 1800-inspired stratified scheduler (Active, NBA, Combinational Convergence, and Time Wheel).

---

## Architectural Highlights

* **Continuous `assign` Dataflow Modeling:** Top-level module continuous assignments dynamically evaluated and converged within the delta-cycle loop without requiring procedural wrapper processes.
* **1D Unpacked Memory Arrays:** Support for memory structures (`logic [W-1:0] mem [0:D-1]`) with dual-rail 4-state unknown initialization, dynamic asynchronous reads (`mem[addr]`), and synchronous non-blocking writes (`mem[addr] <= data`).
* **Multi-Way Control Flow Selection (`case` / `endcase`):** Native branch selection supporting exact value matching, `default` fallbacks, and multi-statement `begin ... end` case items.
* **Vector Bit-Select & Part-Select Slicing:** Dynamic scalar extraction (`wire[idx]`) and fixed range slicing (`wire[msb:lsb]`) with dual-rail 4-state bit alignment.
* **Full ALU Operator Suite & Expression Precedence:** Operator precedence climbing supporting bitwise (`&`, `|`, `^`, `~`), logical shifts (`<<`, `>>`), relational/equality (`==`, `!=`, `<`, `<=`, `>`, `>=`), logical short-circuit gates (`&&`, `||`), and ternary conditional evaluation (`cond ? true_expr : false_expr`).
* **Recursive Multi-Level Module Elaboration:** Fully recursive elaborator traversing arbitrary submodule instantiation depths (`top -> dut -> submod -> ...`), dynamically instantiating scoped nets and binding bidirectional port interconnects.
* **Dual-Rail 4-State Bitwise Evaluation:** Exact per-bit IEEE 1800 evaluation rules for unknown bits (`0 & X = 0`, `1 | X = 1`, `X ^ X = X`, `~X = X`), preserving hardware-accurate uninitialized propagation across wide datapath buses.
* **Native 4-State Logic Engine (`0, 1, X, Z`):** Dual-rail bit-vector arithmetic (`aval` data, `bval` unknown mask). All un-driven nets, memory cells, and uninitialized registers initialize to `X` (unknown) rather than optimistic `0`, catching un-reset design bugs.
* **Self-Checking Regression Suite:** Automated verification harness covering all milestone deliverables from `v0.1.0` through `v0.9.0` with diagnostic pass/fail accounting.
* **Project-Centric Workspace Engine:** Isolated directory hierarchies (`projects/<name>/`) storing RTL designs (`design.sv`), testbenches (`testbench.sv`), metadata (`project.json`), and archived waveform traces (`waves/`).
* **Persistent Waveform History:** Timestamped VCD trace runs stored per project with in-browser selection, deletion, and immediate canvas invalidation when dump directives are omitted.
* **4-State Digital Timing Canvas:** Native HTML5 Canvas timing diagram renderer displaying clocks, single-bit logic wires, and multibit transition buses with distinct visual styles for known values vs. `X`/`Z` states.
* **Native VCD Waveform Generation:** Direct export of IEEE 1364/1800 Value Change Dump (`.vcd`) traces supporting `$dumpfile` and `$dumpvars` with hierarchical scope definitions and compact ASCII symbol mapping.
* **Stratified IEEE 1800 Scheduling (Active vs. NBA):** Evaluates non-blocking assignments (`<=`) during the active region and buffers values into an NBA update queue, committing all register transitions simultaneously before triggering combinational delta passes.
* **Edge-Triggered Sensitivity Engine:** Dynamic transition tracking (`is_posedge()`, `is_negedge()`) across hierarchical signal handles for `always_ff` blocks with asynchronous resets (`X -> 0` correctly triggers `negedge`).
* **Native Discrete-Event Time Wheel:** Min-heap priority queue (`std::priority_queue`) that advances simulation time dynamically across `#delay` intervals.

from fastapi import FastAPI, HTTPException
from fastapi.staticfiles import StaticFiles
from pydantic import BaseModel
from typing import Optional, Dict
import subprocess
import tempfile
import shutil
import json
import os
import glob
from datetime import datetime

app = FastAPI()

PROJECTS_DIR = "/app/projects"
os.makedirs(PROJECTS_DIR, exist_ok=True)

class SimRequest(BaseModel):
    files: Dict[str, str]  # filename -> file content
    entry_tb: Optional[str] = "testbench.sv"
    top_module: Optional[str] = ""
    project_name: Optional[str] = None
    enable_profiler: Optional[bool] = True
    enable_debug: Optional[bool] = True
    enable_vcd: Optional[bool] = True
    timescale: Optional[str] = "1ns"

class SaveProjectRequest(BaseModel):
    project_name: str
    files: Dict[str, str]

# ----------------- Project Management APIs ----------------- #

@app.get("/api/projects")
def list_projects():
    projects = []
    if not os.path.exists(PROJECTS_DIR):
        return {"projects": []}

    for name in os.listdir(PROJECTS_DIR):
        pdir = os.path.join(PROJECTS_DIR, name)
        if os.path.isdir(pdir):
            waves_dir = os.path.join(pdir, "waves")
            vcd_count = len(glob.glob(os.path.join(waves_dir, "*.vcd"))) if os.path.exists(waves_dir) else 0
            mtime = os.path.getmtime(pdir)
            projects.append({
                "name": name,
                "last_modified": datetime.fromtimestamp(mtime).strftime("%Y-%m-%d %H:%M:%S"),
                "vcd_count": vcd_count
            })
    projects.sort(key=lambda x: x["last_modified"], reverse=True)
    return {"projects": projects}

@app.post("/api/projects/save")
def save_project(req: SaveProjectRequest):
    pname = req.project_name.strip()
    if not pname:
        raise HTTPException(status_code=400, detail="Project name cannot be empty.")
    
    pname = "".join(c for c in pname if c.isalnum() or c in ('_', '-'))
    pdir = os.path.join(PROJECTS_DIR, pname)
    waves_dir = os.path.join(pdir, "waves")
    os.makedirs(waves_dir, exist_ok=True)

    # Clean existing non-wave source files and write updated multi-file project
    for f in os.listdir(pdir):
        if f != "waves" and f != "project.json":
            fpath = os.path.join(pdir, f)
            if os.path.isfile(fpath):
                os.remove(fpath)

    for fname, content in req.files.items():
        with open(os.path.join(pdir, fname), "w") as f:
            f.write(content)

    meta = {
        "project_name": pname,
        "saved_at": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        "files": list(req.files.keys())
    }
    with open(os.path.join(pdir, "project.json"), "w") as f:
        json.dump(meta, f, indent=2)

    return {"status": "success", "project_name": pname}

@app.get("/api/projects/{name}")
def load_project(name: str):
    pdir = os.path.join(PROJECTS_DIR, name)
    if not os.path.exists(pdir):
        raise HTTPException(status_code=404, detail="Project not found.")

    files = {}
    for fname in os.listdir(pdir):
        fpath = os.path.join(pdir, fname)
        if os.path.isfile(fpath) and fname != "project.json":
            with open(fpath, "r") as f:
                files[fname] = f.read()

    # Fallbacks if legacy project
    if "testbench.sv" not in files:
        files["testbench.sv"] = ""
    if "design.sv" not in files:
        files["design.sv"] = ""

    waves = []
    waves_dir = os.path.join(pdir, "waves")
    if os.path.exists(waves_dir):
        for wf in os.listdir(waves_dir):
            if wf.endswith(".vcd"):
                wpath = os.path.join(waves_dir, wf)
                waves.append({
                    "filename": wf,
                    "timestamp": datetime.fromtimestamp(os.path.getmtime(wpath)).strftime("%Y-%m-%d %H:%M:%S")
                })
    waves.sort(key=lambda x: x["timestamp"], reverse=True)

    return {
        "project_name": name,
        "files": files,
        "waves": waves
    }

@app.delete("/api/projects/{name}")
def delete_project(name: str):
    pdir = os.path.join(PROJECTS_DIR, name)
    if os.path.exists(pdir):
        shutil.rmtree(pdir)
        return {"status": "success"}
    raise HTTPException(status_code=404, detail="Project not found.")

@app.get("/api/projects/{name}/waves/{filename}")
def get_vcd_file(name: str, filename: str):
    wpath = os.path.join(PROJECTS_DIR, name, "waves", filename)
    if not os.path.exists(wpath):
        raise HTTPException(status_code=404, detail="VCD file not found.")
    with open(wpath, "r") as f:
        content = f.read()
    return {"filename": filename, "vcd": content}

@app.delete("/api/projects/{name}/waves/{filename}")
def delete_vcd_file(name: str, filename: str):
    wpath = os.path.join(PROJECTS_DIR, name, "waves", filename)
    if os.path.exists(wpath):
        os.remove(wpath)
        return {"status": "success"}
    raise HTTPException(status_code=404, detail="VCD file not found.")

@app.post("/api/simulate")
def simulate(req: SimRequest):
    if not req.files:
        return {
            "status": "error",
            "output": "[ERROR] No source files provided.",
            "vcd": None,
            "vcd_filename": None,
            "telemetry": None,
            "debug_log": None
        }

    with tempfile.TemporaryDirectory() as tmpdir:
        # Write ALL created files into sandbox directory so `include can resolve any of them
        for fname, content in req.files.items():
            fpath = os.path.join(tmpdir, fname)
            os.makedirs(os.path.dirname(fpath), exist_ok=True)
            with open(fpath, "w") as f:
                f.write(content)

        # Decide CLI entry files:
        # Standard Verilog practice:
        # 1. Primary entry testbench file is passed.
        # 2. If design.sv is present and NOT `included in testbench, pass it too.
        # 3. All other helper files (+ added tabs) are included via `include.
        entry_tb = req.entry_tb if (req.entry_tb in req.files) else "testbench.sv"
        files_to_run = [os.path.join(tmpdir, entry_tb)]

        tb_text = req.files.get(entry_tb, "")
        if "design.sv" in req.files:
            # If testbench already has `include "design.sv", don't pass design.sv on CLI (avoids duplicate module definitions)
            if '`include "design.sv"' not in tb_text and "`include 'design.sv'" not in tb_text:
                files_to_run.append(os.path.join(tmpdir, "design.sv"))

        cas_bin = "/app/CAS/build/cas"
        options = []
        if not req.enable_profiler:
            options.append("--no-profile")
        if not req.enable_debug:
            options.append("--quiet")
        if not req.enable_vcd:
            options.append("--no-vcd")
        if req.top_module and req.top_module.strip():
            options += ["--top", req.top_module.strip()]
        if req.timescale and req.timescale.strip():
            options += ["--timescale", req.timescale.strip()]

        run_cmd = [cas_bin] + options + files_to_run

        try:
            res = subprocess.run(run_cmd, cwd=tmpdir, capture_output=True, text=True, timeout=15)
            raw_output = res.stdout if res.returncode == 0 else (res.stderr + "\n" + res.stdout)

            telemetry_data = None
            debug_log = ""
            clean_output = raw_output

            if "---BEGIN_CAS_DEBUG_LOG---" in clean_output and "---END_CAS_DEBUG_LOG---" in clean_output:
                b1, p1 = clean_output.split("---BEGIN_CAS_DEBUG_LOG---", 1)
                debug_log, b2 = p1.split("---END_CAS_DEBUG_LOG---", 1)
                clean_output = b1 + b2
                debug_log = debug_log.strip()

            if "---BEGIN_CAS_TELEMETRY---" in clean_output and "---END_CAS_TELEMETRY---" in clean_output:
                before, part = clean_output.split("---BEGIN_CAS_TELEMETRY---", 1)
                json_str, after = part.split("---END_CAS_TELEMETRY---", 1)
                clean_output = before + after
                try:
                    telemetry_data = json.loads(json_str.strip())
                except Exception:
                    telemetry_data = None

            console_output = clean_output.strip()

            vcd_content = None
            vcd_filename = None

            for fname in os.listdir(tmpdir):
                if fname.endswith(".vcd"):
                    vcd_filename = fname
                    with open(os.path.join(tmpdir, fname), "r") as vf:
                        vcd_content = vf.read()

                    if req.project_name:
                        pdir = os.path.join(PROJECTS_DIR, req.project_name)
                        if os.path.exists(pdir):
                            pwaves = os.path.join(pdir, "waves")
                            os.makedirs(pwaves, exist_ok=True)
                            ts = datetime.now().strftime("%Y%m%d_%H%M%S")
                            saved_name = f"{ts}_{fname}"
                            shutil.copy2(os.path.join(tmpdir, fname), os.path.join(pwaves, saved_name))
                            vcd_filename = saved_name
                    break

            return {
                "status": "success" if res.returncode == 0 else "error",
                "output": console_output,
                "vcd": vcd_content,
                "vcd_filename": vcd_filename,
                "telemetry": telemetry_data,
                "debug_log": debug_log
            }
        except subprocess.TimeoutExpired as e:
            partial_out = (e.stdout or "") + "\n" + (e.stderr or "")
            return {
                "status": "error",
                "output": f"[CAS TIMEOUT (15s)]\n{partial_out}",
                "vcd": None,
                "vcd_filename": None,
                "telemetry": None,
                "debug_log": None
            }
        except Exception as e:
            return {
                "status": "error",
                "output": f"[CAS INVOCATION ERROR] {str(e)}",
                "vcd": None,
                "vcd_filename": None,
                "telemetry": None,
                "debug_log": None
            }

    // 8. 4-State Digital Timing Canvas
    function renderActiveWaveform() {
        if (!activeVcdContent) {
            document.getElementById('wave-empty').style.display = "flex";
            return;
        }
        document.getElementById('wave-empty').style.display = "none";

        const namesContainer = document.getElementById('wave-names');
        const canvas = document.getElementById('wave-canvas');
        const ctx = canvas.getContext('2d');
        namesContainer.innerHTML = '';

        const lines = activeVcdContent.split('\n');
        const idToVar = {};
        const signals = [];
        let maxTime = 0;
        let inVars = false;

        for (let line of lines) {
            line = line.trim();
            if (line.startsWith('$var')) {
                const parts = line.split(/\s+/);
                const width = parseInt(parts[2]);
                const id = parts[3];
                const name = parts[4];
                const sig = { id, name, width, changes: [] };
                idToVar[id] = sig;
                signals.push(sig);
            } else if (line.startsWith('$enddefinitions')) {
                inVars = true;
            } else if (inVars && line.startsWith('#')) {
                const t = parseInt(line.substring(1));
                if (t > maxTime) maxTime = t;
            }
        }

        if (maxTime === 0) maxTime = 50;

        let curTime = 0;
        for (let line of lines) {
            line = line.trim();
            if (!line || line.startsWith('$')) continue;
            if (line.startsWith('#')) {
                curTime = parseInt(line.substring(1));
                continue;
            }

            if (line.startsWith('b')) {
                const parts = line.substring(1).split(/\s+/);
                const val = parts[0];
                const id = parts[1];
                if (idToVar[id]) {
                    const isX = val.includes('x') || val.includes('z');
                    idToVar[id].changes.push({ time: curTime, val: isX ? 'x' : parseInt(val, 2), raw: val });
                }
            } else {
                const charVal = line[0].toLowerCase();
                const id = line.substring(1);
                if (idToVar[id]) {
                    idToVar[id].changes.push({ time: curTime, val: charVal });
                }
            }
        }

        const rowHeight = 36;
        const timeHeaderHeight = 28;
        const timeStepPx = 14;
        const totalWidth = Math.max(800, (maxTime + 10) * timeStepPx);
        const totalHeight = timeHeaderHeight + (signals.length * rowHeight);

        canvas.width = totalWidth;
        canvas.height = totalHeight;

        signals.forEach(s => {
            const div = document.createElement('div');
            div.className = 'wave-sig-row';
            div.innerHTML = `<span>${s.name}</span><span class="wave-sig-type">${s.width > 1 ? '[' + (s.width-1) + ':0]' : 'wire'}</span>`;
            namesContainer.appendChild(div);
        });

        ctx.fillStyle = '#0b0c0e';
        ctx.fillRect(0, 0, totalWidth, totalHeight);

        ctx.fillStyle = '#16171d';
        ctx.fillRect(0, 0, totalWidth, timeHeaderHeight);
        ctx.strokeStyle = '#272a33';
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(0, timeHeaderHeight);
        ctx.lineTo(totalWidth, timeHeaderHeight);
        ctx.stroke();

        ctx.font = '10px JetBrains Mono';
        ctx.fillStyle = '#8b949e';

        for (let t = 0; t <= maxTime + 5; t += 5) {
            const x = t * timeStepPx;
            ctx.beginPath();
            ctx.moveTo(x, timeHeaderHeight - 6);
            ctx.lineTo(x, totalHeight);
            ctx.strokeStyle = (t % 10 === 0) ? '#272a33' : '#1a1b22';
            ctx.stroke();
            ctx.fillText(t + 'ns', x + 3, timeHeaderHeight - 8);
        }

        signals.forEach((sig, idx) => {
            const yTop = timeHeaderHeight + (idx * rowHeight);
            const yMid = yTop + (rowHeight / 2);
            const yHigh = yTop + 7;
            const yLow = yTop + rowHeight - 7;

            ctx.strokeStyle = '#181920';
            ctx.beginPath();
            ctx.moveTo(0, yTop + rowHeight);
            ctx.lineTo(totalWidth, yTop + rowHeight);
            ctx.stroke();

            if (sig.changes.length === 0) return;

            if (sig.width === 1) {
                for (let i = 0; i < sig.changes.length; i++) {
                    const c = sig.changes[i];
                    const nextTime = (i + 1 < sig.changes.length) ? sig.changes[i+1].time : (maxTime + 5);
                    const xStart = c.time * timeStepPx;
                    const xEnd = nextTime * timeStepPx;

                    ctx.lineWidth = 2;
                    if (c.val === 'x' || c.val === 'z') {
                        ctx.strokeStyle = '#ef4444';
                        ctx.beginPath();
                        ctx.moveTo(xStart, yMid);
                        ctx.lineTo(xEnd, yMid);
                        ctx.stroke();
                    } else {
                        ctx.strokeStyle = '#10b981';
                        const yVal = (c.val === '1' || c.val === 1) ? yHigh : yLow;
                        ctx.beginPath();
                        ctx.moveTo(xStart, yVal);
                        ctx.lineTo(xEnd, yVal);
                        ctx.stroke();
                    }

                    if (i + 1 < sig.changes.length) {
                        const nextVal = sig.changes[i+1].val;
                        const yCur = (c.val === 'x' || c.val === 'z') ? yMid : ((c.val === '1' || c.val === 1) ? yHigh : yLow);
                        const yNext = (nextVal === 'x' || nextVal === 'z') ? yMid : ((nextVal === '1' || nextVal === 1) ? yHigh : yLow);
                        ctx.strokeStyle = (nextVal === 'x' || c.val === 'x') ? '#ef4444' : '#10b981';
                        ctx.beginPath();
                        ctx.moveTo(xEnd, yCur);
                        ctx.lineTo(xEnd, yNext);
                        ctx.stroke();
                    }
                }
            } else {
                for (let i = 0; i < sig.changes.length; i++) {
                    const c = sig.changes[i];
                    const nextTime = (i + 1 < sig.changes.length) ? sig.changes[i+1].time : (maxTime + 5);
                    const xStart = c.time * timeStepPx;
                    const xEnd = nextTime * timeStepPx;
                    const boxW = xEnd - xStart;

                    if (boxW <= 0) continue;

                    const isX = (c.val === 'x');
                    ctx.strokeStyle = isX ? '#ef4444' : '#38bdf8';
                    ctx.fillStyle = isX ? '#450a0a' : '#1e293b';
                    ctx.lineWidth = 1.5;

                    ctx.beginPath();
                    ctx.moveTo(xStart + 3, yHigh);
                    ctx.lineTo(xEnd - 3, yHigh);
                    ctx.lineTo(xEnd, yMid);
                    ctx.lineTo(xEnd - 3, yLow);
                    ctx.lineTo(xStart + 3, yLow);
                    ctx.lineTo(xStart, yMid);
                    ctx.closePath();
                    ctx.fill();
                    ctx.stroke();

                    ctx.fillStyle = isX ? '#fca5a5' : '#e2e8f0';
                    ctx.font = '11px JetBrains Mono';
                    const label = isX ? "X" : ("0x" + c.val.toString(16).toUpperCase());
                    if (boxW > 25) ctx.fillText(label, xStart + 8, yMid + 4);
                }
            }
        });
    }

    // 9. Fully Responsive Resizers
    const resizerMain = document.getElementById('resizer-main');
    const editorsWrapper = document.getElementById('editors-wrapper');
    const workspace = document.getElementById('workspace');

    let isDraggingV = false;
    resizerMain.addEventListener('mousedown', () => isDraggingV = true);

    const resizerH = document.getElementById('resizer-editors');
    const colTb = document.getElementById('col-tb');
    let isDraggingH = false;
    resizerH.addEventListener('mousedown', () => isDraggingH = true);

    const leftResizer = document.getElementById('left-resizer');
    const leftSidebar = document.getElementById('options-sidebar');
    let isDraggingLeft = false;
    leftResizer.addEventListener('mousedown', () => isDraggingLeft = true);

    const rightResizer = document.getElementById('right-resizer');
    const rightSidebar = document.getElementById('inspector-sidebar');
    let isDraggingRight = false;
    rightResizer.addEventListener('mousedown', () => isDraggingRight = true);

    window.addEventListener('mousemove', (e) => {
        if (isDraggingV) {
            const h = workspace.offsetHeight;
            const topH = ((e.clientY - workspace.offsetTop) / h) * 100;
            if (topH > 20 && topH < 85) {
                editorsWrapper.style.height = topH + '%';
                relayoutMonaco();
            }
        }
        if (isDraggingH) {
            const w = editorsWrapper.offsetWidth;
            const leftW = ((e.clientX - editorsWrapper.offsetLeft) / w) * 100;
            if (leftW > 20 && leftW < 80) {
                colTb.style.flex = "none";
                colTb.style.width = leftW + '%';
                relayoutMonaco();
            }
        }
        if (isDraggingLeft) {
            const newW = e.clientX;
            if (newW > 180 && newW < 450) {
                leftSidebar.style.width = newW + 'px';
                relayoutMonaco();
            }
        }
        if (isDraggingRight) {
            const newW = window.innerWidth - e.clientX;
            if (newW > 240 && newW < 600) {
                rightSidebar.style.width = newW + 'px';
                relayoutMonaco();
            }
        }
    });

    window.addEventListener('mouseup', () => {
        if (isDraggingV || isDraggingH || isDraggingLeft || isDraggingRight) {
            relayoutMonaco();
        }
        isDraggingV = false;
        isDraggingH = false;
        isDraggingLeft = false;
        isDraggingRight = false;
    });

# Release History & Version Log

All notable architectural enhancements, language grammar additions, and runtime changes for CAS are documented here in reverse chronological order.

---

## [v0.9.0] - 2026-10-26

### Added
* **Continuous `assign` Statements:**
  * Added `TOKEN_ASSIGN` (`assign`) keyword to lexer.
  * Added `ContinuousAssignNode` to the AST and module elaboration pass.
  * Evaluated within `Engine::propagate_comb()`, driving signals continuously alongside port interconnects and combinational blocks.
* **1D Unpacked Memory Arrays:**
  * Grammar support for unpacked dimension declarations: `logic [width-1:0] name [0:depth-1];`.
  * Implemented `MemoryArray` structure in `Engine` with per-word dual-rail 4-state storage.
  * Dynamic array reads via `IndexSelectNode` returning `X` for unknown/out-of-bounds addresses.
  * Added `MemAssignNode` supporting blocking (`mem[addr] = data`) and non-blocking (`mem[addr] <= data`) procedural writes.
  * Implemented `NBAMemUpdate` queue to commit memory writes synchronously in the NBA scheduling region.
* **Regression Suite Milestone Update:**
  * Extended testbench to 35 hierarchical nets, 19 port bindings, 2 continuous assigns, and 1 memory array.
  * Verified continuous assign dataflow, synchronous RAM writing, and asynchronous RAM reading.

---

## [v0.8.0] - 2026-10-26

### Added
* **Case Statement Branch Selection:**
  * Added tokens: `TOKEN_CASE` (`case`), `TOKEN_ENDCASE` (`endcase`), and `TOKEN_DEFAULT` (`default`).
  * Implemented `CaseStatementNode` and `CaseItem` supporting sequential branch testing and fallback defaults.
  * Supported single statements and multi-statement `begin ... end` compound blocks inside case branches.
* **Vector Bit-Select & Part-Select Indexing:**
  * Added `IndexSelectNode` to AST.
  * Parser handles bracketed indices during primary expression parsing (`ident[idx]` and `ident[msb:lsb]`).
  * `Engine::eval_expr()` extracts scalar bits and sub-slices with exact dual-rail `aval` / `bval` alignment, returning `X` on out-of-bounds or unknown indices.
* **Procedural Pipeline Fixes:**
  * Fixed combinational modified-flag reset order in `Engine::propagate_comb()`, preventing missed sequential edge transitions during reset assertion.
  * Updated procedural execution in `step_process()` so blocking updates trigger combinational propagation and sequential edge evaluation immediately.
* **Regression Test Suite Expansion:**
  * Expanded `projects/regression/` to 27 nets and 15 port bindings, verifying `case` branching and bit/part-select slicing with automated pass/fail verification.

---

## [v0.7.0] - 2026-10-26

### Added
* **Recursive Multi-Level Elaboration Engine:**
  * Replaced flat single-level instance walker with recursive `Engine::elaborate_module()` capable of resolving arbitrary-depth module hierarchies (`top -> dut -> submod -> ...`).
  * Hierarchical net mapping correctly handles deeply nested instance scopes and links cross-boundary port driver/receiver bindings.
* **Extended Operator Grammar & Precedence Climbing:**
  * Added tokenization and stratified parsing for bitwise operators (`&`, `|`, `^`, `~`), logical shifts (`<<`, `>>`), relational/equality (`==`, `!=`, `<`, `<=`, `>`, `>=`), logical short-circuit gates (`&&`, `||`), and ternary conditionals (`?:`).
  * Stratified expression precedence climbing eliminates parsing ambiguities in complex ALU operations.
* **Dual-Rail 4-State Bitwise Operations:**
  * Implemented exact per-bit truth tables in `LogicVal` for bitwise AND, OR, XOR, NOT, and shifts.
  * Added 4-state logical equality and relational comparison operators returning `X` on unknown inputs.
* **Procedural Conditionals in Initial Processes:**
  * Updated `Engine::execute_statement()` and `step_process()` to support procedural `if-else` branching inside testbench processes with nested `$display` and `$finish` calls.
* **Comprehensive Automated Regression Suite:**
  * Added `projects/regression/` with automated self-checking assertions covering all language features and architectural milestones from `v0.1.0` through `v0.7.0`.

---

## [v0.6.0] - 2026-10-26

### Added
* **4-State Dual-Rail Logic Engine (`0, 1, X, Z`):**
  * Replaced native unsigned integer state with dual-rail `LogicVal` (`aval` for data, `bval` for unknown/Z control mask).
  * All signals and registers now initialize to `X` by default upon elaboration, matching standard IEEE 1800 simulator behavior.
  * Implemented 4-state arithmetic poisoning: arithmetic operations on vectors with any unknown bits propagate `X` to all result bits.
  * Enhanced edge detection (`is_posedge()`, `is_negedge()`) to correctly detect transitions to and from unknown/uninitialized states (`X -> 0` triggers `negedge`, `X -> 1` triggers `posedge`).
* **Waveform & Terminal 4-State Rendering:**
  * Updated VCD dumper to format single-bit unknown scalars (`x<id>`) and multi-bit vectors containing unknown bits (`b...x... <id>`).
  * Updated Canvas timing diagram renderer to display unknown single-bit wires as mid-rail red traces and unknown buses as red-shaded `X` intervals.
  * Fixed terminal selection CSS (`user-select: text`) allowing developers to highlight and copy console simulation logs directly.

---

## [v0.5.1] - 2026-10-26

### Added
* **Project Management Architecture:**
  * Added persistent project storage under `/app/projects/<name>/` preserving `design.sv`, `testbench.sv`, `project.json`, and historical `waves/`.
  * Added REST endpoints in `server.py`: `GET/POST /api/projects`, `GET/DELETE /api/projects/{name}`, and waveform fetching/deletion routes.
  * Added Project Explorer drawer modal to create, open, and delete projects.

  * Simulations inside an active project automatically archive timestamped `.vcd` files into that project's `waves/` directory.
  * Added trace selector dropdown in the Waveform Viewer tab allowing instant playback of historical runs.
  * Implemented strict waveform canvas clearing: re-running without `$dumpfile` or `$dumpvars` immediately invalidates previous canvas renders and disables export.

## [v0.5.0] - 2026-10-26

### Added

  * Added `$dumpfile("...")` and `$dumpvars` language support to `Lexer`, `Parser`, and `AST`.
  * Implemented IEEE 1364/1800 compliant VCD header generation in `Engine`: `$date`, `$version`, `$timescale 1ns`, and hierarchical `$scope ... $var ... $upscope`.

## [v0.4.0] - 2026-10-26

### Added
* **Stratified IEEE 1800 Scheduler (Active & NBA Regions):**
  * Implemented an NBA buffer queue (`std::vector<NBAUpdate>`) in `Engine`.
  * Non-blocking assignments (`<=`) evaluate expressions during the active region without altering target nets immediately.
  * `commit_nba_updates()` synchronously commits all pending non-blocking updates and triggers combinational delta passes.
* **Sequential Logic & Edge Detection:**
  * Added `always_ff` process handling with sensitivity lists supporting `posedge` and `negedge` transitions.
  * Enhanced `Net` struct with `prev_value` history and transition queries (`is_posedge()`, `is_negedge()`).

  * Added `if (...) begin ... end else begin ... end` grammar constructs and `IfStatementNode`.
  * Added logical NOT (`!`) operator support via `UnaryOpNode`.
  * Added non-blocking assignment operator (`<=` / `TOKEN_LE`) support in `Lexer` and `Parser`.

## [v0.3.0] - 2026-10-26

### Added
* **Discrete-Event Time Wheel:**
  * Implemented an ordered min-heap priority queue (`std::priority_queue<SimEvent>`) in `Engine`.
  * Simulation time (`sim_time`) advances dynamically to scheduled timestamps rather than executing exclusively at $T=0$.
* **Cooperative Process Scheduling:**
  * Procedural processes (`initial` blocks) track execution progress using an internal Program Counter (`size_t pc`).
  * Yields execution upon reaching `#delay`, inserting a wake-up event into the time wheel at `sim_time + delay`.
* **Delay Syntax & Language Extensions:**
  * Added tokenization for the `#` operator (`TOKEN_HASH`) in `Lexer`.
  * Implemented `DelayStatementNode` in `include/ast.hpp` and integrated parsing into `Parser::parse_statement()`.
* **System Task `$time`:**
  * Added `$time` token and expression evaluation resolving directly to current simulation time.
  * Supported `%0t` and `%t` format specifiers in `$display`.

## [v0.2.0] - 2026-10-26

### Added
* **Hierarchical Module Instantiation:**
  * Added grammar support for named port connections: `module_type instance_name (.port(wire), ...);`.
  * Added `InstantiationNode` and `PortConnection` AST structures to `include/ast.hpp`.
  * Implemented dot-tokenization (`TOKEN_DOT` / `.`) in `Lexer`.
* **Hierarchical Netlist Scoping:**
  * Elaborator automatically determines the design's top-level root module.
  * Signals are scoped systematically (e.g., `tb_adder.a`, `tb_adder.u_dut.a`).
  * Continuous port-binding table (`NetBinding`) links driving parent nets to receiving child input ports, and driving child output ports back to parent nets.
* **Immediate Combinational Delta Propagation:**
  * Implemented `propagate_comb()` in the evaluation kernel to guarantee that procedural blocking assignments (`=`) immediately trigger iterative combinational convergence across module boundaries before sequential calls (such as `$display`) execute.

## [v0.1.0] - 2026-10-26

### Added
