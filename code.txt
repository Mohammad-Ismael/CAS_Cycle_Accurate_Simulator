#pragma once
#include <string>
#include <cstdint>
#include <vector>
#include <algorithm>

struct LogicVal {
    uint64_t aval; // Data word
    uint64_t bval; // Control word: 0=known, 1=unknown/Z
    uint32_t width;

    LogicVal(uint32_t w = 1, uint64_t a = 0, uint64_t b = 0)
        : aval(a), bval(b), width(w) {
        mask();
    }

    static LogicVal make_x(uint32_t w) {
        uint64_t m = (w >= 64) ? ~0ULL : ((1ULL << w) - 1);
        return LogicVal(w, m, m);
    }

    static LogicVal make_known(uint32_t w, uint64_t val) {
        return LogicVal(w, val, 0);
    }

    void mask() {
        uint64_t m = (width >= 64) ? ~0ULL : ((1ULL << width) - 1);
        aval &= m;
        bval &= m;
    }

    bool is_unknown() const {
        return bval != 0;
    }

    bool to_bool() const {
        if (is_unknown()) return false;
        return aval != 0;
    }

    bool operator==(const LogicVal& o) const {
        return (aval == o.aval) && (bval == o.bval) && (width == o.width);
    }

    bool operator!=(const LogicVal& o) const {
        return !(*this == o);
    }

    // 4-state bitwise NOT (~)
    LogicVal bitwise_not() const {
        // Known 0 -> 1, Known 1 -> 0, Unknown stays Unknown
        uint64_t new_a = (~aval) & (~bval);
        return LogicVal(width, new_a, bval);
    }

    // 4-state bitwise AND (&): 0 & X = 0, 1 & X = X
    static LogicVal bitwise_and(const LogicVal& l, const LogicVal& r) {
        uint32_t max_w = std::max(l.width, r.width);
        uint64_t new_a = l.aval & r.aval;
        // Result is unknown where an operand is unknown unless the other operand is known 0
        uint64_t l_zero = (~l.aval) & (~l.bval);
        uint64_t r_zero = (~r.aval) & (~r.bval);
        uint64_t any_zero = l_zero | r_zero;
        uint64_t new_b = (l.bval | r.bval) & (~any_zero);
        return LogicVal(max_w, new_a, new_b);
    }

    // 4-state bitwise OR (|): 1 | X = 1, 0 | X = X
    static LogicVal bitwise_or(const LogicVal& l, const LogicVal& r) {
        uint32_t max_w = std::max(l.width, r.width);
        uint64_t new_a = l.aval | r.aval;
        // Result is unknown where an operand is unknown unless the other operand is known 1
        uint64_t l_one = l.aval & (~l.bval);
        uint64_t r_one = r.aval & (~r.bval);
        uint64_t any_one = l_one | r_one;
        uint64_t new_b = (l.bval | r.bval) & (~any_one);
        return LogicVal(max_w, new_a, new_b);
    }

    // 4-state bitwise XOR (^): any unknown bit produces unknown
    static LogicVal bitwise_xor(const LogicVal& l, const LogicVal& r) {
        uint32_t max_w = std::max(l.width, r.width);
        uint64_t new_b = l.bval | r.bval;
        uint64_t new_a = (l.aval ^ r.aval) & (~new_b);
        return LogicVal(max_w, new_a, new_b);
    }

    // Logical Left Shift (<<)
    static LogicVal shift_left(const LogicVal& l, const LogicVal& r) {
        if (r.is_unknown()) return make_x(l.width);
        uint64_t sh = r.aval;
        if (sh >= l.width) return LogicVal(l.width, 0, 0);
        return LogicVal(l.width, l.aval << sh, l.bval << sh);
    }

    // Logical Right Shift (>>)
    static LogicVal shift_right(const LogicVal& l, const LogicVal& r) {
        if (r.is_unknown()) return make_x(l.width);
        uint64_t sh = r.aval;
        if (sh >= l.width) return LogicVal(l.width, 0, 0);
        return LogicVal(l.width, l.aval >> sh, l.bval >> sh);
    }

    // 4-state Equality (==)
    static LogicVal logical_eq(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval == r.aval) ? 1 : 0);
    }

    // 4-state Inequality (!=)
    static LogicVal logical_neq(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval != r.aval) ? 1 : 0);
    }

    // Relational Operators (<, <=, >, >=)
    static LogicVal rel_lt(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval < r.aval) ? 1 : 0);
    }

    static LogicVal rel_le(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval <= r.aval) ? 1 : 0);
    }

    static LogicVal rel_gt(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval > r.aval) ? 1 : 0);
    }

    static LogicVal rel_ge(const LogicVal& l, const LogicVal& r) {
        if (l.is_unknown() || r.is_unknown()) return make_x(1);
        return make_known(1, (l.aval >= r.aval) ? 1 : 0);
    }

    std::string to_bin_string() const {
        std::string s;
        for (int i = (int)width - 1; i >= 0; --i) {
            bool a = (aval >> i) & 1;
            bool b = (bval >> i) & 1;
            if (!b) {
                s += (a ? '1' : '0');
            } else {
                s += (a ? 'x' : 'z');
            }
        }
        return s;
    }
};

struct Net {
    std::string name;
    uint32_t width;
    LogicVal value;
    LogicVal prev_value;
    bool modified;
    std::string vcd_id;

    Net(std::string n, uint32_t w)
        : name(std::move(n)), width(w),
          value(LogicVal::make_x(w)),
          prev_value(LogicVal::make_x(w)),
          modified(false), vcd_id("") {}

    void set(const LogicVal& val) {
        LogicVal clamped(width, val.aval, val.bval);
        if (value != clamped) {
            prev_value = value;
            value = clamped;
            modified = true;
        }
    }

    bool is_posedge() const {
        bool curr_1 = (value.bval == 0 && (value.aval & 1) == 1);
        bool prev_not_1 = (prev_value.bval != 0) || ((prev_value.aval & 1) == 0);
        return prev_not_1 && curr_1;
    }

    bool is_negedge() const {
        bool curr_0 = (value.bval == 0 && (value.aval & 1) == 0);
        bool prev_not_0 = (prev_value.bval != 0) || ((prev_value.aval & 1) == 1);
        return prev_not_0 && curr_0;
    }
};

    if (vcd_file.is_open()) {
        vcd_file.close();

std::string Engine::generate_vcd_id(size_t id) {
    std::string res;
    size_t base = 94;
    do {
        res += (char)(33 + (id % base));
        id /= base;
    } while (id > 0);
    return res;
}

size_t Engine::get_or_create_net(const std::string& scoped_name, uint32_t width) {
    if (net_map.find(scoped_name) != net_map.end()) {
        return net_map[scoped_name];
    }
    size_t id = netlist.size();
    netlist.emplace_back(scoped_name, width);
    netlist.back().vcd_id = generate_vcd_id(id);
    net_map[scoped_name] = id;
    return id;
}

NetSliceRef Engine::resolve_slice_ref(ExpressionNode* expr, const std::string& scope) {
    if (auto ident = dynamic_cast<IdentifierNode*>(expr)) {
        std::string full_name = scope.empty() ? ident->name : (scope + "." + ident->name);
        size_t id = net_map.at(full_name);
        return {id, 0, netlist[id].width};
    }
    if (auto idx = dynamic_cast<IndexSelectNode*>(expr)) {
        std::string full_name = scope.empty() ? idx->name : (scope + "." + idx->name);
        size_t id = net_map.at(full_name);
        if (!idx->is_part_select) {
            LogicVal iv = eval_expr(idx->msb_expr.get(), scope);
            uint32_t off = iv.is_unknown() ? 0 : (uint32_t)iv.aval;
            return {id, off, 1};
        } else {
            LogicVal mv = eval_expr(idx->msb_expr.get(), scope);
            LogicVal lv = eval_expr(idx->lsb_expr.get(), scope);
            uint32_t msb = mv.is_unknown() ? 0 : (uint32_t)mv.aval;
            uint32_t lsb = lv.is_unknown() ? 0 : (uint32_t)lv.aval;
            uint32_t w = (msb >= lsb) ? (msb - lsb + 1) : (lsb - msb + 1);
            return {id, lsb, w};
        }
    }
    // Support constant literals (e.g. 1'b0, 1'b1, 0) tied to module input ports
    if (auto num = dynamic_cast<NumberNode*>(expr)) {
        static size_t const_counter = 0;
        std::string const_net_name = scope + ".__const_port_" + std::to_string(const_counter++);
        size_t id = get_or_create_net(const_net_name, num->width);
        LogicVal val = num->is_unknown ? LogicVal::make_x(num->width) : LogicVal::make_known(num->width, num->value);
        netlist[id].set(val);
        return {id, 0, num->width};
    }

    throw std::runtime_error("Expression is not a bindable net target");
}

LogicVal Engine::read_slice(const NetSliceRef& ref) {
    const auto& net_val = netlist[ref.net_id].value;
    uint64_t mask = (ref.width >= 64) ? ~0ULL : ((1ULL << ref.width) - 1);
    uint64_t a = (net_val.aval >> ref.offset) & mask;
    uint64_t b = (net_val.bval >> ref.offset) & mask;
    return LogicVal(ref.width, a, b);
}

void Engine::write_slice(const NetSliceRef& ref, const LogicVal& val) {
    auto cur = netlist[ref.net_id].value;
    uint64_t mask = (ref.width >= 64) ? ~0ULL : ((1ULL << ref.width) - 1);
    uint64_t clear_mask = ~(mask << ref.offset);

    uint64_t new_a = (cur.aval & clear_mask) | ((val.aval & mask) << ref.offset);
    uint64_t new_b = (cur.bval & clear_mask) | ((val.bval & mask) << ref.offset);
    netlist[ref.net_id].set(LogicVal(cur.width, new_a, new_b));
}

LogicVal Engine::eval_expr(ExpressionNode* expr, const std::string& scope) {
    if (auto num = dynamic_cast<NumberNode*>(expr)) {
        if (num->is_unknown) return LogicVal::make_x(num->width);
        return LogicVal::make_known(num->width, num->value);
    }
    if (auto ident = dynamic_cast<IdentifierNode*>(expr)) {
        if (ident->name == "$time") return LogicVal::make_known(64, sim_time);
        std::string scoped_ident = scope.empty() ? ident->name : (scope + "." + ident->name);
        return netlist[net_map.at(scoped_ident)].value;
    }
    if (auto idx_node = dynamic_cast<IndexSelectNode*>(expr)) {
        std::string scoped_ident = scope.empty() ? idx_node->name : (scope + "." + idx_node->name);
        if (memory_map.find(scoped_ident) != memory_map.end()) {
            const auto& mem = memory_map.at(scoped_ident);
            LogicVal addr_val = eval_expr(idx_node->msb_expr.get(), scope);
            if (addr_val.is_unknown() || addr_val.aval >= mem.depth) {
                return LogicVal::make_x(mem.word_width);
            }
            return mem.words[addr_val.aval];
        }

        const auto& target_val = netlist[net_map.at(scoped_ident)].value;
        if (!idx_node->is_part_select) {
            LogicVal idx_val = eval_expr(idx_node->msb_expr.get(), scope);
            if (idx_val.is_unknown() || idx_val.aval >= target_val.width) return LogicVal::make_x(1);
            uint32_t sh = idx_val.aval;
            return LogicVal(1, (target_val.aval >> sh) & 1, (target_val.bval >> sh) & 1);
        } else {
            LogicVal msb_v = eval_expr(idx_node->msb_expr.get(), scope);
            LogicVal lsb_v = eval_expr(idx_node->lsb_expr.get(), scope);
            if (msb_v.is_unknown() || lsb_v.is_unknown() || msb_v.aval < lsb_v.aval) {
                uint32_t w = (msb_v.aval >= lsb_v.aval) ? (msb_v.aval - lsb_v.aval + 1) : 1;
                return LogicVal::make_x(w);
            }
            uint32_t msb = msb_v.aval;
            uint32_t lsb = lsb_v.aval;
            uint32_t slice_w = msb - lsb + 1;
            uint64_t mask = (slice_w >= 64) ? ~0ULL : ((1ULL << slice_w) - 1);
            return LogicVal(slice_w, (target_val.aval >> lsb) & mask, (target_val.bval >> lsb) & mask);
        }
    }
    if (auto un = dynamic_cast<UnaryOpNode*>(expr)) {
        LogicVal val = eval_expr(un->operand.get(), scope);
        if (un->op == '!') {
            if (val.is_unknown()) return LogicVal::make_x(1);
            return LogicVal::make_known(1, (val.aval == 0) ? 1 : 0);
        }
        if (un->op == '~') return val.bitwise_not();
    }
    if (auto bin = dynamic_cast<BinaryOpNode*>(expr)) {
        LogicVal l = eval_expr(bin->lhs.get(), scope);
        LogicVal r = eval_expr(bin->rhs.get(), scope);

        if (bin->op == "+") {
            uint32_t w = std::max(l.width, r.width);
            if (l.is_unknown() || r.is_unknown()) return LogicVal::make_x(w);
            return LogicVal::make_known(w, l.aval + r.aval);
        }
        if (bin->op == "-") {
            uint32_t w = std::max(l.width, r.width);
            if (l.is_unknown() || r.is_unknown()) return LogicVal::make_x(w);
            return LogicVal::make_known(w, l.aval - r.aval);
        }
        if (bin->op == "&")  return LogicVal::bitwise_and(l, r);
        if (bin->op == "|")  return LogicVal::bitwise_or(l, r);
        if (bin->op == "^")  return LogicVal::bitwise_xor(l, r);
        if (bin->op == "<<") return LogicVal::shift_left(l, r);
        if (bin->op == ">>") return LogicVal::shift_right(l, r);
        if (bin->op == "==") return LogicVal::logical_eq(l, r);
        if (bin->op == "!=") return LogicVal::logical_neq(l, r);
        if (bin->op == "<")  return LogicVal::rel_lt(l, r);
        if (bin->op == "<=") return LogicVal::rel_le(l, r);
        if (bin->op == ">")  return LogicVal::rel_gt(l, r);
        if (bin->op == ">=") return LogicVal::rel_ge(l, r);
        if (bin->op == "&&") {
            if ((!l.is_unknown() && l.aval == 0) || (!r.is_unknown() && r.aval == 0)) return LogicVal::make_known(1, 0);
            if (l.is_unknown() || r.is_unknown()) return LogicVal::make_x(1);
            return LogicVal::make_known(1, (l.aval != 0 && r.aval != 0) ? 1 : 0);
        }
        if (bin->op == "||") {
            if ((!l.is_unknown() && l.aval != 0) || (!r.is_unknown() && r.aval != 0)) return LogicVal::make_known(1, 1);
            if (l.is_unknown() || r.is_unknown()) return LogicVal::make_x(1);
            return LogicVal::make_known(1, (l.aval != 0 || r.aval != 0) ? 1 : 0);
        }
    }
    if (auto tern = dynamic_cast<TernaryOpNode*>(expr)) {
        LogicVal cond = eval_expr(tern->condition.get(), scope);
        if (cond.is_unknown()) {
            LogicVal t_val = eval_expr(tern->true_expr.get(), scope);
            LogicVal f_val = eval_expr(tern->false_expr.get(), scope);
            return LogicVal::make_x(std::max(t_val.width, f_val.width));
        }
        return cond.to_bool() ? eval_expr(tern->true_expr.get(), scope) : eval_expr(tern->false_expr.get(), scope);
    }
    return LogicVal::make_x(1);
}

void Engine::init_vcd_header() {
    if (!vcd_enabled || vcd_file.is_open()) return;

    vcd_file.open(vcd_filename);
    if (!vcd_file.is_open()) {
        std::cerr << "[CAS ERROR] Unable to create VCD file: " << vcd_filename << "\n";
        vcd_enabled = false;
        return;
    }

    std::time_t now = std::time(nullptr);
    vcd_file << "$date\n   " << std::asctime(std::localtime(&now)) << "$end\n";
    vcd_file << "$version\n   CAS (Cycle Accurate Simulator) v1.0.0 [Production Kernel]\n$end\n";
    vcd_file << "$timescale 1ns $end\n";

    std::unordered_map<std::string, std::vector<size_t>> scopes;
    for (size_t i = 0; i < netlist.size(); ++i) {
        std::string full_name = netlist[i].name;
        size_t dot_pos = full_name.rfind('.');
        std::string scope_name = (dot_pos == std::string::npos) ? "top" : full_name.substr(0, dot_pos);
        scopes[scope_name].push_back(i);
    }

    for (const auto& kv : scopes) {
        vcd_file << "$scope module " << kv.first << " $end\n";
        for (size_t net_id : kv.second) {
            const auto& net = netlist[net_id];
            size_t dot_pos = net.name.rfind('.');
            std::string leaf_name = (dot_pos == std::string::npos) ? net.name : net.name.substr(dot_pos + 1);
            vcd_file << "$var wire " << net.width << " " << net.vcd_id << " " << leaf_name << " $end\n";
        }
        vcd_file << "$upscope $end\n";
    }

    vcd_file << "$enddefinitions $end\n";
    vcd_file << "$dumpvars\n";
    dump_vcd_sample(true);
    vcd_file << "$end\n";
}

void Engine::dump_vcd_sample(bool initial_dump) {
    if (!vcd_enabled || !vcd_file.is_open()) return;

    if (!initial_dump && sim_time != last_dump_time) {
        vcd_file << "#" << sim_time << "\n";
        last_dump_time = sim_time;
    }

    for (auto& net : netlist) {
        if (initial_dump || net.modified) {
            if (net.width == 1) {
                if (net.value.is_unknown()) vcd_file << "x" << net.vcd_id << "\n";
                else vcd_file << (net.value.aval & 1) << net.vcd_id << "\n";
            } else {
                vcd_file << "b" << net.value.to_bin_string() << " " << net.vcd_id << "\n";
            }
        }
    }
}

void Engine::check_monitor() {
    if (!monitor_session.active) return;

    bool values_changed = false;
    std::vector<LogicVal> current_values;

    for (const auto& arg_name : monitor_session.args) {
        if (arg_name == "$time") {
            current_values.push_back(LogicVal::make_known(64, sim_time));
        } else {
            std::string scoped_arg = monitor_session.scope + "." + arg_name;
            if (net_map.find(scoped_arg) != net_map.end()) {
                current_values.push_back(netlist[net_map.at(scoped_arg)].value);
            } else {
                current_values.push_back(LogicVal::make_x(1));
            }
        }
    }

    if (monitor_session.prev_values.empty()) {
        values_changed = true;
    } else {
        for (size_t i = 0; i < current_values.size(); ++i) {
            if (current_values[i] != monitor_session.prev_values[i]) {
                values_changed = true;
                break;
            }
        }
    }

    if (values_changed) {
        monitor_session.prev_values = current_values;
        if (monitor_session.has_leading_time) {
            std::cout << sim_time;
        }

        size_t arg_idx = 0;
        for (size_t i = 0; i < monitor_session.format_string.size(); ++i) {
            if (monitor_session.format_string[i] == '%' && i + 1 < monitor_session.format_string.size()) {
                char spec = monitor_session.format_string[++i];
                if (spec == '0' && i + 1 < monitor_session.format_string.size() && monitor_session.format_string[i + 1] == 't') {
                    spec = 't'; i++;
                } else if (spec == '0' && i + 1 < monitor_session.format_string.size() && (monitor_session.format_string[i + 1] == 'h' || monitor_session.format_string[i + 1] == 'd' || monitor_session.format_string[i + 1] == 'b')) {
                    spec = monitor_session.format_string[++i];
                }

                if (arg_idx < current_values.size()) {
                    const auto& val = current_values[arg_idx++];
                    if (val.is_unknown()) {
                        std::cout << "x";
                    } else {
                        if (spec == 'd' || spec == 't') std::cout << val.aval;
                        else if (spec == 'b') std::cout << val.to_bin_string();
                        else if (spec == 'h') std::cout << std::hex << val.aval << std::dec;
                    }
                }
            } else {
                std::cout << monitor_session.format_string[i];
            }
        }
        std::cout << "\n";
    }
}

void Engine::propagate_comb() {
    bool changed = true;
    size_t iters = 0;
    while (changed && iters < 100) {
        changed = false;

        // 1. Module port interconnect slice bindings
        for (auto& bind : slice_bindings) {
            LogicVal src_val = read_slice(bind.src);
            LogicVal dst_val = read_slice(bind.dst);
            if (src_val != dst_val) {
                write_slice(bind.dst, src_val);
                changed = true;
            }
        }

        // 2. Built-in Gate Primitives
        for (auto& g : gates) {
            LogicVal res = eval_expr(g.in_exprs[0], g.scope);
            for (size_t i = 1; i < g.in_exprs.size(); ++i) {
                LogicVal next_in = eval_expr(g.in_exprs[i], g.scope);
                if (g.type == GateType::AND) res = LogicVal::bitwise_and(res, next_in);
                else if (g.type == GateType::OR) res = LogicVal::bitwise_or(res, next_in);
                else if (g.type == GateType::XOR) res = LogicVal::bitwise_xor(res, next_in);
                else if (g.type == GateType::NAND) res = LogicVal::bitwise_and(res, next_in);
                else if (g.type == GateType::NOR) res = LogicVal::bitwise_or(res, next_in);
                else if (g.type == GateType::XNOR) res = LogicVal::bitwise_xor(res, next_in);
            }
            if (g.type == GateType::NAND || g.type == GateType::NOR || g.type == GateType::XNOR || g.type == GateType::NOT) {
                res = res.bitwise_not();
            }
            LogicVal cur_out = read_slice(g.out_target);
            if (cur_out != res) {
                write_slice(g.out_target, res);
                changed = true;
            }
        }

        // 3. Continuous assign statements
        for (auto& ca : continuous_assigns) {
            LogicVal val = eval_expr(ca.rhs, ca.scope);
            LogicVal cur = read_slice(ca.target);
            if (cur != val) {
                write_slice(ca.target, val);
                changed = true;
            }
        }

        // 4. always_comb blocks
        for (auto& bp : processes) {
            if (bp.proc->type != ProcessType::COMBINATIONAL) continue;
            for (auto& stmt : bp.proc->body) {
                execute_statement(stmt.get(), bp);
            }
        }

        for (const auto& n : netlist) {
            if (n.modified) {
                changed = true;
                break;
            }
        }
        iters++;
    }
}

void Engine::execute_statement(StatementNode* stmt, BoundProcess& bp) {
    if (auto if_stmt = dynamic_cast<IfStatementNode*>(stmt)) {
        LogicVal cond = eval_expr(if_stmt->condition.get(), bp.scope);
        if (cond.to_bool()) {
            for (auto& s : if_stmt->then_branch) execute_statement(s.get(), bp);
        } else {
            for (auto& s : if_stmt->else_branch) execute_statement(s.get(), bp);
        }
    }
    else if (auto case_stmt = dynamic_cast<CaseStatementNode*>(stmt)) {
        LogicVal sel_val = eval_expr(case_stmt->selector.get(), bp.scope);
        bool branch_taken = false;
        const CaseItem* default_item = nullptr;

        for (const auto& item : case_stmt->items) {
            if (item.is_default) {
                default_item = &item;
                continue;
            }
            LogicVal item_val = eval_expr(item.match_expr.get(), bp.scope);
            if (LogicVal::logical_eq(sel_val, item_val).to_bool()) {
                for (auto& s : item.statements) execute_statement(s.get(), bp);
                branch_taken = true;
                break;
            }
        }
        if (!branch_taken && default_item) {
            for (auto& s : default_item->statements) execute_statement(s.get(), bp);
        }
    }
    else if (auto tcall = dynamic_cast<TaskCallNode*>(stmt)) {
        std::string scoped_tname = bp.scope + "." + tcall->task_name;
        if (task_map.find(scoped_tname) != task_map.end()) {
            TaskNode* target_task = task_map.at(scoped_tname);
            for (auto& s : target_task->body) {
                execute_statement(s.get(), bp);
                if (!simulation_active) return;
            }
        }
    }
    else if (auto disp = dynamic_cast<DisplayStatementNode*>(stmt)) {
        if (disp->is_monitor) {
            monitor_session.active = true;
            monitor_session.scope = bp.scope;
            monitor_session.format_string = disp->format_string;
            monitor_session.args = disp->args;
            monitor_session.has_leading_time = disp->has_leading_time;
            monitor_session.prev_values.clear();
            check_monitor();
        } else {
            std::cout << "[CAS $display] ";
            if (disp->has_leading_time) {
                std::cout << sim_time << " ";
            }
            size_t arg_idx = 0;
            for (size_t i = 0; i < disp->format_string.size(); ++i) {
                if (disp->format_string[i] == '%' && i + 1 < disp->format_string.size()) {
                    char spec = disp->format_string[++i];
                    if (spec == '0' && i + 1 < disp->format_string.size() && disp->format_string[i + 1] == 't') {
                        spec = 't'; i++;
                    } else if (spec == '0' && i + 1 < disp->format_string.size() && (disp->format_string[i + 1] == 'h' || disp->format_string[i + 1] == 'd' || disp->format_string[i + 1] == 'b')) {
                        spec = disp->format_string[++i];
                    }

                    if (arg_idx < disp->args.size()) {
                        std::string arg_name = disp->args[arg_idx++];
                        if (arg_name == "$time") {
                            std::cout << sim_time;
                        } else {
                            std::string scoped_arg = bp.scope + "." + arg_name;
                            size_t net_id = net_map.at(scoped_arg);
                            const auto& net = netlist[net_id];
                            if (net.value.is_unknown()) {
                                std::cout << "x";
                            } else {
                                if (spec == 'd' || spec == 't') std::cout << net.value.aval;
                                else if (spec == 'b') std::cout << net.value.to_bin_string();
                                else if (spec == 'h') std::cout << std::hex << net.value.aval << std::dec;
                            }
                        }
                    }
                } else {
                    std::cout << disp->format_string[i];
                }
            }
            std::cout << "\n";
        }
    }
    else if (dynamic_cast<FinishStatementNode*>(stmt)) {
        std::cout << "[CAS $finish] Simulation Terminated at time " << sim_time << ".\n";
        simulation_active = false;
        bp.completed = true;
    }
    else if (auto nassign = dynamic_cast<NonBlockingAssignNode*>(stmt)) {
        LogicVal rhs_val = eval_expr(nassign->rhs.get(), bp.scope);
        std::string scoped_target = bp.scope + "." + nassign->target;
        size_t net_id = net_map.at(scoped_target);
        nba_queue.push_back({net_id, rhs_val});
    }
    else if (auto mem_assign = dynamic_cast<MemAssignNode*>(stmt)) {
        std::string scoped_mem = bp.scope + "." + mem_assign->mem_name;
        LogicVal addr_val = eval_expr(mem_assign->addr_expr.get(), bp.scope);
        LogicVal rhs_val = eval_expr(mem_assign->rhs.get(), bp.scope);

        if (!addr_val.is_unknown()) {
            uint32_t addr = addr_val.aval;
            if (mem_assign->is_non_blocking) {
                nba_mem_queue.push_back({scoped_mem, addr, rhs_val});
            } else {
                if (memory_map.find(scoped_mem) != memory_map.end()) {
                    auto& mem = memory_map.at(scoped_mem);
                    if (addr < mem.depth) mem.words[addr] = rhs_val;
                }
            }
        }
    }
    else if (auto assign = dynamic_cast<BlockingAssignNode*>(stmt)) {
        LogicVal rhs_val = eval_expr(assign->rhs.get(), bp.scope);
        if (assign->lhs_targets.size() > 1) {
            int shift = 0;
            for (int i = (int)assign->lhs_targets.size() - 1; i >= 0; --i) {
                std::string scoped_target = bp.scope + "." + assign->lhs_targets[i];
                size_t net_id = net_map.at(scoped_target);
                uint32_t width = netlist[net_id].width;
                uint64_t mask = (width >= 64) ? ~0ULL : ((1ULL << width) - 1);
                netlist[net_id].set(LogicVal(width, (rhs_val.aval >> shift) & mask, (rhs_val.bval >> shift) & mask));
                shift += width;
            }
        } else {
            std::string scoped_target = bp.scope + "." + assign->lhs_targets[0];
            size_t net_id = net_map.at(scoped_target);
            netlist[net_id].set(rhs_val);
        }
    }
}

void Engine::evaluate_sequential_blocks() {
    for (auto& bp : processes) {
        if (bp.proc->type != ProcessType::SEQUENTIAL) continue;

        bool triggered = false;
        for (auto& sens : bp.proc->sensitivity_list) {
            std::string scoped_sig = bp.scope + "." + sens.signal_name;
            size_t net_id = net_map.at(scoped_sig);
            const auto& net = netlist[net_id];

            if (sens.edge == EdgeType::POSEDGE && net.is_posedge()) { triggered = true; break; }
            else if (sens.edge == EdgeType::NEGEDGE && net.is_negedge()) { triggered = true; break; }
        }

        if (triggered) {
            for (auto& stmt : bp.proc->body) execute_statement(stmt.get(), bp);
        }
    }
}

void Engine::commit_nba_updates() {
    if (nba_queue.empty() && nba_mem_queue.empty()) return;

    for (const auto& update : nba_queue) netlist[update.net_id].set(update.val);
    nba_queue.clear();

    for (const auto& mem_up : nba_mem_queue) {
        if (memory_map.find(mem_up.mem_scoped_name) != memory_map.end()) {
            auto& mem = memory_map.at(mem_up.mem_scoped_name);
            if (mem_up.addr < mem.depth) mem.words[mem_up.addr] = mem_up.val;
        }
    }
    nba_mem_queue.clear();
    propagate_comb();
}

void Engine::elaborate_module(ModuleNode* mod, const std::string& scope,
                              const std::unordered_map<std::string, ModuleNode*>& module_defs) {
    for (auto& p : mod->ports) {
        uint32_t w = (p.msb >= p.lsb) ? (p.msb - p.lsb + 1) : 1;
        get_or_create_net(scope + "." + p.name, w);
    }

    for (auto& s : mod->signals) {
        uint32_t w = (s.msb >= s.lsb) ? (s.msb - s.lsb + 1) : 1;
        if (s.is_array) {
            std::string mem_scope = scope + "." + s.name;
            memory_map[mem_scope] = MemoryArray(mem_scope, w, s.array_depth);
        } else {
            get_or_create_net(scope + "." + s.name, w);
        }
    }

    for (auto& ca : mod->continuous_assigns) {
        std::string scoped_target = scope + "." + ca->target_name;
        size_t target_id = net_map.at(scoped_target);
        continuous_assigns.push_back({scope, {target_id, 0, netlist[target_id].width}, ca->rhs.get()});
    }

    for (auto& g : mod->gates) {
        GateInstance gi;
        gi.type = g->type;
        gi.scope = scope;
        gi.out_target = resolve_slice_ref(g->out_expr.get(), scope);
        for (auto& ie : g->in_exprs) gi.in_exprs.push_back(ie.get());
        gates.push_back(gi);
    }

    for (auto& t : mod->tasks) {
        task_map[scope + "." + t->name] = t.get();
    }

    for (auto& proc : mod->processes) {
        processes.push_back({scope, proc.get(), 0, false});
    }

    for (auto& inst : mod->instantiations) {
        std::string child_scope = scope + "." + inst->instance_name;
        if (module_defs.find(inst->module_type) == module_defs.end()) {
            throw std::runtime_error("Undefined module type instantiated: " + inst->module_type);
        }
        ModuleNode* submod = module_defs.at(inst->module_type);

        elaborate_module(submod, child_scope, module_defs);

        for (size_t conn_idx = 0; conn_idx < inst->connections.size(); ++conn_idx) {
            auto& conn = inst->connections[conn_idx];
            std::string target_port_name;
            if (inst->is_positional) {
                if (conn_idx >= submod->ports.size()) {
                    throw std::runtime_error("Positional connection out of bounds in " + inst->instance_name);
                }
                target_port_name = submod->ports[conn_idx].name;
            } else {
                target_port_name = conn.port_name;
            }

            std::string dir = "input";
            for (auto& p : submod->ports) {
                if (p.name == target_port_name) {
                    dir = p.direction;
                    break;
                }
            }

            NetSliceRef parent_ref = resolve_slice_ref(conn.expr.get(), scope);
            size_t child_net_id = net_map.at(child_scope + "." + target_port_name);
            NetSliceRef child_ref = {child_net_id, 0, netlist[child_net_id].width};

            if (dir == "input") {
                slice_bindings.push_back({parent_ref, child_ref});
            } else {
                slice_bindings.push_back({child_ref, parent_ref});
            }
        }
    }
}

void Engine::elaborate(std::vector<std::unique_ptr<ModuleNode>>& modules) {
    std::unordered_map<std::string, ModuleNode*> module_defs;
    for (auto& mod : modules) {
        module_defs[mod->name] = mod.get();
    }

    std::unordered_map<std::string, bool> is_instantiated;
    for (auto& mod : modules) {
        for (auto& inst : mod->instantiations) {
            is_instantiated[inst->module_type] = true;
        }
    }

    ModuleNode* top = nullptr;
    for (auto& mod : modules) {
        if (!is_instantiated[mod->name]) {
            top = mod.get();
            break;
        }
    }
    if (!top) top = modules.front().get();

    elaborate_module(top, top->name, module_defs);
}

void Engine::step_process(size_t pid) {
    auto& bp = processes[pid];
    if (bp.completed) return;

    while (bp.pc < bp.proc->body.size()) {
        auto stmt = bp.proc->body[bp.pc].get();

        if (auto delay = dynamic_cast<DelayStatementNode*>(stmt)) {
            bp.pc++;
            uint64_t wake_time = sim_time + delay->delay;
            time_wheel.push({wake_time, (uint32_t)pid, event_order_counter++});
            return;
        }

        if (auto df = dynamic_cast<DumpFileStatementNode*>(stmt)) {
            vcd_filename = df->filename;
            vcd_enabled = true;
            bp.pc++;
            continue;
        }

        if (dynamic_cast<DumpVarsStatementNode*>(stmt)) {
            init_vcd_header();
            bp.pc++;
            continue;
        }

        if (auto if_stmt = dynamic_cast<IfStatementNode*>(stmt)) {
            execute_statement(if_stmt, bp);
            if (!simulation_active) return;
            bp.pc++;
            continue;
        }

        if (auto case_stmt = dynamic_cast<CaseStatementNode*>(stmt)) {
            execute_statement(case_stmt, bp);
            if (!simulation_active) return;
            bp.pc++;
            continue;
        }

        execute_statement(stmt, bp);
        if (!simulation_active) return;

        propagate_comb();
        evaluate_sequential_blocks();
        commit_nba_updates();
        dump_vcd_sample();
        check_monitor();

        for (auto& n : netlist) n.modified = false;
        bp.pc++;
    }

    bp.completed = true;
}

void Engine::run() {
    std::cout << "CAS (Cycle Accurate Simulator) v1.0.0 [Production Kernel] Initiated\n";
    std::cout << "Hierarchical Nets Elaborated: " << netlist.size() << "\n";
    std::cout << "Slice Interconnects Bound: " << slice_bindings.size() << "\n";
    std::cout << "Gate Primitives Instantiated: " << gates.size() << "\n";
    std::cout << "Continuous Assigns Bound: " << continuous_assigns.size() << "\n";
    std::cout << "--------------------------------------------------------\n";

    propagate_comb();

    for (size_t i = 0; i < processes.size(); ++i) {
        if (processes[i].proc->type == ProcessType::INITIAL) {
            step_process(i);
        }
    }

    while (simulation_active && !time_wheel.empty()) {
        SimEvent ev = time_wheel.top();
        time_wheel.pop();

        sim_time = ev.time;
        step_process(ev.process_id);
    }

    if (vcd_enabled && vcd_file.is_open()) {
        vcd_file.close();
        std::cout << "[CAS VCD] Trace successfully dumped to: " << vcd_filename << "\n";
    }

    std::cout << "--------------------------------------------------------\n";
    std::cout << "Simulation finished at sim_time = " << sim_time << ".\n";
}

#include "../include/lexer.hpp"
#include <cctype>
#include <stdexcept>

Lexer::Lexer(std::string source) : src(std::move(source)) {}

char Lexer::peek() const {
    if (cursor >= src.size()) return '\0';
    return src[cursor];
}

char Lexer::advance() {
    if (cursor >= src.size()) return '\0';
    char c = src[cursor++];
    if (c == '\n') line++;
    return c;
}

void Lexer::skip_whitespace() {
    while (true) {
        char c = peek();
        if (std::isspace(c)) {
            advance();
        } else if (c == '/' && cursor + 1 < src.size() && src[cursor + 1] == '/') {
            while (peek() != '\n' && peek() != '\0') advance();
        } else if (c == '/' && cursor + 1 < src.size() && src[cursor + 1] == '*') {
            advance(); advance();
            while (!(peek() == '*' && cursor + 1 < src.size() && src[cursor + 1] == '/')) {
                if (peek() == '\0') break;
                advance();
            }
            if (peek() != '\0') { advance(); advance(); }
        } else {
            break;
        }
    }
}

Token Lexer::next_token() {
    skip_whitespace();
    if (cursor >= src.size()) return {TokenType::TOKEN_EOF, "", 0, 0, line};

    char c = peek();

    // Delimiters
    if (c == '(') { advance(); return {TokenType::TOKEN_LPAREN, "(", 0, 0, line}; }
    if (c == ')') { advance(); return {TokenType::TOKEN_RPAREN, ")", 0, 0, line}; }
    if (c == '[') { advance(); return {TokenType::TOKEN_LBRACKET, "[", 0, 0, line}; }
    if (c == ']') { advance(); return {TokenType::TOKEN_RBRACKET, "]", 0, 0, line}; }
    if (c == '{') { advance(); return {TokenType::TOKEN_LBRACE, "{", 0, 0, line}; }
    if (c == '}') { advance(); return {TokenType::TOKEN_RBRACE, "}", 0, 0, line}; }
    if (c == ';') { advance(); return {TokenType::TOKEN_SEMICOLON, ";", 0, 0, line}; }
    if (c == ',') { advance(); return {TokenType::TOKEN_COMMA, ",", 0, 0, line}; }
    if (c == '.') { advance(); return {TokenType::TOKEN_DOT, ".", 0, 0, line}; }
    if (c == '#') { advance(); return {TokenType::TOKEN_HASH, "#", 0, 0, line}; }
    if (c == '@') { advance(); return {TokenType::TOKEN_AT, "@", 0, 0, line}; }
    if (c == '?') { advance(); return {TokenType::TOKEN_QUESTION, "?", 0, 0, line}; }
    if (c == ':') { advance(); return {TokenType::TOKEN_COLON, ":", 0, 0, line}; }
    if (c == '~') { advance(); return {TokenType::TOKEN_TILDE, "~", 0, 0, line}; }
    if (c == '^') { advance(); return {TokenType::TOKEN_XOR, "^", 0, 0, line}; }
    if (c == '+') { advance(); return {TokenType::TOKEN_PLUS, "+", 0, 0, line}; }
    if (c == '-') { advance(); return {TokenType::TOKEN_MINUS, "-", 0, 0, line}; }

    // Multi-character Operators
    if (c == '&') {
        advance();
        if (peek() == '&') { advance(); return {TokenType::TOKEN_LOGICAL_AND, "&&", 0, 0, line}; }
        return {TokenType::TOKEN_AND, "&", 0, 0, line};
    }
    if (c == '|') {
        advance();
        if (peek() == '|') { advance(); return {TokenType::TOKEN_LOGICAL_OR, "||", 0, 0, line}; }
        return {TokenType::TOKEN_OR_BIT, "|", 0, 0, line};
    }
    if (c == '!') {
        advance();
        if (peek() == '=') { advance(); return {TokenType::TOKEN_NEQ, "!=", 0, 0, line}; }
        return {TokenType::TOKEN_NOT, "!", 0, 0, line};
    }
    if (c == '=') {
        advance();
        if (peek() == '=') { advance(); return {TokenType::TOKEN_EQ, "==", 0, 0, line}; }
        return {TokenType::TOKEN_EQUAL, "=", 0, 0, line};
    }
    if (c == '<') {
        advance();
        if (peek() == '=') { advance(); return {TokenType::TOKEN_LE, "<=", 0, 0, line}; }
        if (peek() == '<') { advance(); return {TokenType::TOKEN_SHL, "<<", 0, 0, line}; }
        return {TokenType::TOKEN_LT, "<", 0, 0, line};
    }
    if (c == '>') {
        advance();
        if (peek() == '=') { advance(); return {TokenType::TOKEN_GE, ">=", 0, 0, line}; }
        if (peek() == '>') { advance(); return {TokenType::TOKEN_SHR, ">>", 0, 0, line}; }
        return {TokenType::TOKEN_GT, ">", 0, 0, line};
    }

    // Numeric Literals
    if (std::isdigit(c)) {
        std::string num_str;
        while (std::isdigit(peek())) num_str += advance();

        if (peek() == '\'') {
            uint32_t width = std::stoul(num_str);
            advance(); // consume '\''
            char base = advance();
            std::string digits;
            while (std::isxdigit(peek()) || peek() == '_' || peek() == 'x' || peek() == 'X' || peek() == 'z' || peek() == 'Z') {
                if (peek() != '_') digits += peek();
                advance();
            }

            uint64_t val = 0;
            bool is_x_or_z = false;
            for (char ch : digits) {
                if (ch == 'x' || ch == 'X' || ch == 'z' || ch == 'Z') {
                    is_x_or_z = true;
                    break;
                }
            }
            if (!is_x_or_z) {
                try {
                    if (base == 'd' || base == 'D') val = std::stoull(digits);
                    else if (base == 'h' || base == 'H') val = std::stoull(digits, nullptr, 16);
                    else if (base == 'b' || base == 'B') val = std::stoull(digits, nullptr, 2);
                } catch (...) {
                    val = 0;
                }
            }
            return {TokenType::TOKEN_NUMBER, digits, val, width, line};
        }
        return {TokenType::TOKEN_NUMBER, num_str, std::stoull(num_str), 32, line};
    }

    // String Literals
    if (c == '"') {
        advance();
        std::string s;
        while (peek() != '"' && peek() != '\0') {
            if (peek() == '\\') {
                advance();
                char esc = advance();
                if (esc == 'n') s += '\n';
                else if (esc == 't') s += '\t';
                else s += esc;
            } else {
                s += advance();
            }
        }
        if (peek() == '"') advance();
        return {TokenType::TOKEN_STRING, s, 0, 0, line};
    }

    // Identifiers and Keywords
    if (std::isalpha(c) || c == '_' || c == '$') {
        std::string ident;
        while (std::isalnum(peek()) || peek() == '_' || peek() == '$') {
            ident += advance();
        }

        if (ident == "module") return {TokenType::TOKEN_MODULE, ident, 0, 0, line};
        if (ident == "endmodule") return {TokenType::TOKEN_ENDMODULE, ident, 0, 0, line};
        if (ident == "input") return {TokenType::TOKEN_INPUT, ident, 0, 0, line};
        if (ident == "output") return {TokenType::TOKEN_OUTPUT, ident, 0, 0, line};
        if (ident == "inout") return {TokenType::TOKEN_INOUT, ident, 0, 0, line};
        if (ident == "logic") return {TokenType::TOKEN_LOGIC, ident, 0, 0, line};
        if (ident == "wire") return {TokenType::TOKEN_WIRE, ident, 0, 0, line};
        if (ident == "reg") return {TokenType::TOKEN_REG, ident, 0, 0, line};
        if (ident == "assign") return {TokenType::TOKEN_ASSIGN, ident, 0, 0, line};
        if (ident == "always") return {TokenType::TOKEN_ALWAYS, ident, 0, 0, line};
        if (ident == "always_comb") return {TokenType::TOKEN_ALWAYS_COMB, ident, 0, 0, line};
        if (ident == "always_ff") return {TokenType::TOKEN_ALWAYS_FF, ident, 0, 0, line};
        if (ident == "task") return {TokenType::TOKEN_TASK, ident, 0, 0, line};
        if (ident == "endtask") return {TokenType::TOKEN_ENDTASK, ident, 0, 0, line};

        // Sensitivity 'or' keyword (also acts as gate 'or')
        if (ident == "or") return {TokenType::TOKEN_OR, ident, 0, 0, line};

        // Gate Primitives
        if (ident == "and") return {TokenType::TOKEN_GATE_AND, ident, 0, 0, line};
        if (ident == "xor") return {TokenType::TOKEN_GATE_XOR, ident, 0, 0, line};
        if (ident == "nand") return {TokenType::TOKEN_GATE_NAND, ident, 0, 0, line};
        if (ident == "nor") return {TokenType::TOKEN_GATE_NOR, ident, 0, 0, line};
        if (ident == "xnor") return {TokenType::TOKEN_GATE_XNOR, ident, 0, 0, line};
        if (ident == "not") return {TokenType::TOKEN_GATE_NOT, ident, 0, 0, line};
        if (ident == "buf") return {TokenType::TOKEN_GATE_BUF, ident, 0, 0, line};

        if (ident == "posedge") return {TokenType::TOKEN_POSEDGE, ident, 0, 0, line};
        if (ident == "negedge") return {TokenType::TOKEN_NEGEDGE, ident, 0, 0, line};
        if (ident == "if") return {TokenType::TOKEN_IF, ident, 0, 0, line};
        if (ident == "else") return {TokenType::TOKEN_ELSE, ident, 0, 0, line};
        if (ident == "case") return {TokenType::TOKEN_CASE, ident, 0, 0, line};
        if (ident == "endcase") return {TokenType::TOKEN_ENDCASE, ident, 0, 0, line};
        if (ident == "default") return {TokenType::TOKEN_DEFAULT, ident, 0, 0, line};
        if (ident == "begin") return {TokenType::TOKEN_BEGIN, ident, 0, 0, line};
        if (ident == "end") return {TokenType::TOKEN_END, ident, 0, 0, line};
        if (ident == "initial") return {TokenType::TOKEN_INITIAL, ident, 0, 0, line};
        if (ident == "$display") return {TokenType::TOKEN_DISPLAY, ident, 0, 0, line};
        if (ident == "$monitor") return {TokenType::TOKEN_MONITOR, ident, 0, 0, line};
        if (ident == "$finish") return {TokenType::TOKEN_FINISH, ident, 0, 0, line};
        if (ident == "$stop") return {TokenType::TOKEN_STOP, ident, 0, 0, line};
        if (ident == "$dumpfile") return {TokenType::TOKEN_DUMPFILE, ident, 0, 0, line};
        if (ident == "$dumpvars") return {TokenType::TOKEN_DUMPVARS, ident, 0, 0, line};

        return {TokenType::TOKEN_IDENTIFIER, ident, 0, 0, line};
    }

    throw std::runtime_error(std::string("Unknown lexical token encountered: '") + c + "' on line " + std::to_string(line));
}

int main(int argc, char* argv[]) {
    if (argc < 2) {
        std::cerr << "Usage: cas <source_files...>\n";
        return 1;
    }

    std::string combined_src;
    for (int i = 1; i < argc; ++i) {
        std::ifstream f(argv[i]);
        if (!f.is_open()) {
            std::cerr << "Error: Cannot open " << argv[i] << "\n";
            return 1;
        }
        std::stringstream ss;
        ss << f.rdbuf();
        combined_src += ss.str() + "\n";
    }

    try {
        Lexer lexer(combined_src);
        Parser parser(lexer);
        auto modules = parser.parse_all();

        Engine engine;
        engine.elaborate(modules);
        engine.run();
    } catch (const std::exception& e) {
        std::cerr << "FATAL SIMULATOR ERROR: " << e.what() << "\n";
        return 2;
    }

    return 0;
}

#include "../include/parser.hpp"
#include <stdexcept>
#include <iostream>

Parser::Parser(Lexer& lex) : lexer(lex) {
    advance();
}

void Parser::advance() {
    current = lexer.next_token();
}

bool Parser::match(TokenType t) {
    if (current.type == t) {
        advance();
        return true;
    }
    return false;
}

void Parser::expect(TokenType t, const std::string& err) {
    if (current.type != t) {
        throw std::runtime_error(err + " on line " + std::to_string(current.line) + " (found: '" + current.text + "')");
    }
    advance();
}

std::unique_ptr<ExpressionNode> Parser::parse_primary() {
    if (current.type == TokenType::TOKEN_NUMBER) {
        bool is_unk = false;
        for (char ch : current.text) {
            if (ch == 'x' || ch == 'X' || ch == 'z' || ch == 'Z') {
                is_unk = true;
                break;
            }
        }
        auto node = std::make_unique<NumberNode>(current.num_val, current.bit_width, is_unk);
        advance();
        return node;
    }
    if (current.type == TokenType::TOKEN_IDENTIFIER) {
        std::string ident_name = current.text;
        advance();

        if (match(TokenType::TOKEN_LBRACKET)) {
            auto first_expr = parse_expression();
            if (match(TokenType::TOKEN_COLON)) {
                auto second_expr = parse_expression();
                expect(TokenType::TOKEN_RBRACKET, "Expected ']' after part-select slice");
                return std::make_unique<IndexSelectNode>(ident_name, std::move(first_expr), std::move(second_expr));
            } else {
                expect(TokenType::TOKEN_RBRACKET, "Expected ']' after index");
                return std::make_unique<IndexSelectNode>(ident_name, std::move(first_expr));
            }
        }
        return std::make_unique<IdentifierNode>(ident_name);
    }
    if (match(TokenType::TOKEN_LPAREN)) {
        auto expr = parse_expression();
        expect(TokenType::TOKEN_RPAREN, "Expected ')' closing expression");
        return expr;
    }
    throw std::runtime_error("Unexpected token in expression evaluation on line " + std::to_string(current.line) + " (token: '" + current.text + "')");
}

std::unique_ptr<ExpressionNode> Parser::parse_unary() {
    if (match(TokenType::TOKEN_NOT)) {
        return std::make_unique<UnaryOpNode>('!', parse_unary());
    }
    if (match(TokenType::TOKEN_TILDE)) {
        return std::make_unique<UnaryOpNode>('~', parse_unary());
    }
    return parse_primary();
}

std::unique_ptr<ExpressionNode> Parser::parse_shift() {
    auto lhs = parse_unary();
    while (current.type == TokenType::TOKEN_SHL || current.type == TokenType::TOKEN_SHR) {
        std::string op = (current.type == TokenType::TOKEN_SHL) ? "<<" : ">>";
        advance();
        auto rhs = parse_unary();
        lhs = std::make_unique<BinaryOpNode>(op, std::move(lhs), std::move(rhs));
    }
    return lhs;
}

std::unique_ptr<ExpressionNode> Parser::parse_additive() {
    auto lhs = parse_shift();
    while (current.type == TokenType::TOKEN_PLUS || current.type == TokenType::TOKEN_MINUS) {
        std::string op = (current.type == TokenType::TOKEN_PLUS) ? "+" : "-";
        advance();
        auto rhs = parse_shift();
        lhs = std::make_unique<BinaryOpNode>(op, std::move(lhs), std::move(rhs));
    }
    return lhs;
}

std::unique_ptr<ExpressionNode> Parser::parse_relational() {
    auto lhs = parse_additive();
    while (current.type == TokenType::TOKEN_LT || current.type == TokenType::TOKEN_LE ||
           current.type == TokenType::TOKEN_GT || current.type == TokenType::TOKEN_GE) {
        std::string op;
        if (current.type == TokenType::TOKEN_LT) op = "<";
        else if (current.type == TokenType::TOKEN_LE) op = "<=";
        else if (current.type == TokenType::TOKEN_GT) op = ">";
        else if (current.type == TokenType::TOKEN_GE) op = ">=";
        advance();
        auto rhs = parse_additive();
        lhs = std::make_unique<BinaryOpNode>(op, std::move(lhs), std::move(rhs));
    }
    return lhs;
}

std::unique_ptr<ExpressionNode> Parser::parse_equality() {
    auto lhs = parse_relational();
    while (current.type == TokenType::TOKEN_EQ || current.type == TokenType::TOKEN_NEQ) {
        std::string op = (current.type == TokenType::TOKEN_EQ) ? "==" : "!=";
        advance();
        auto rhs = parse_relational();
        lhs = std::make_unique<BinaryOpNode>(op, std::move(lhs), std::move(rhs));
    }
    return lhs;
}

std::unique_ptr<ExpressionNode> Parser::parse_bitwise_and() {
    auto lhs = parse_equality();
    while (match(TokenType::TOKEN_AND)) {
        auto rhs = parse_equality();
        lhs = std::make_unique<BinaryOpNode>("&", std::move(lhs), std::move(rhs));
    }
    return lhs;
}

std::unique_ptr<ExpressionNode> Parser::parse_bitwise_xor() {
    auto lhs = parse_bitwise_and();
    while (match(TokenType::TOKEN_XOR)) {
        auto rhs = parse_bitwise_and();
        lhs = std::make_unique<BinaryOpNode>("^", std::move(lhs), std::move(rhs));
    }
    return lhs;
}

std::unique_ptr<ExpressionNode> Parser::parse_bitwise_or() {
    auto lhs = parse_bitwise_xor();
    while (match(TokenType::TOKEN_OR_BIT)) {
        auto rhs = parse_bitwise_xor();
        lhs = std::make_unique<BinaryOpNode>("|", std::move(lhs), std::move(rhs));
    }
    return lhs;
}

std::unique_ptr<ExpressionNode> Parser::parse_logical_and() {
    auto lhs = parse_bitwise_or();
    while (match(TokenType::TOKEN_LOGICAL_AND)) {
        auto rhs = parse_bitwise_or();
        lhs = std::make_unique<BinaryOpNode>("&&", std::move(lhs), std::move(rhs));
    }
    return lhs;
}

std::unique_ptr<ExpressionNode> Parser::parse_logical_or() {
    auto lhs = parse_logical_and();
    while (match(TokenType::TOKEN_LOGICAL_OR)) {
        auto rhs = parse_logical_and();
        lhs = std::make_unique<BinaryOpNode>("||", std::move(lhs), std::move(rhs));
    }
    return lhs;
}

std::unique_ptr<ExpressionNode> Parser::parse_ternary() {
    auto cond = parse_logical_or();
    if (match(TokenType::TOKEN_QUESTION)) {
        auto true_expr = parse_expression();
        expect(TokenType::TOKEN_COLON, "Expected ':' in ternary conditional operator");
        auto false_expr = parse_ternary();
        return std::make_unique<TernaryOpNode>(std::move(cond), std::move(true_expr), std::move(false_expr));
    }
    return cond;
}

std::unique_ptr<ExpressionNode> Parser::parse_expression() {
    return parse_ternary();
}

std::unique_ptr<StatementNode> Parser::parse_statement() {
    // 1. Delay
    if (match(TokenType::TOKEN_HASH)) {
        if (current.type != TokenType::TOKEN_NUMBER) {
            throw std::runtime_error("Expected numeric delay value after '#' on line " + std::to_string(current.line));
        }
        uint64_t delay_val = current.num_val;
        advance();
        match(TokenType::TOKEN_SEMICOLON);
        return std::make_unique<DelayStatementNode>(delay_val);
    }

    // 2. If-Else Construct
    if (match(TokenType::TOKEN_IF)) {
        expect(TokenType::TOKEN_LPAREN, "Expected '(' after 'if'");
        auto cond = parse_expression();
        expect(TokenType::TOKEN_RPAREN, "Expected ')' after if-condition");

        auto if_stmt = std::make_unique<IfStatementNode>(std::move(cond));

        if (match(TokenType::TOKEN_BEGIN)) {
            while (current.type != TokenType::TOKEN_END && current.type != TokenType::TOKEN_EOF) {
                if_stmt->then_branch.push_back(parse_statement());
            }
            expect(TokenType::TOKEN_END, "Expected 'end' closing if-block");
        } else {
            if_stmt->then_branch.push_back(parse_statement());
        }

        if (match(TokenType::TOKEN_ELSE)) {
            if (match(TokenType::TOKEN_BEGIN)) {
                while (current.type != TokenType::TOKEN_END && current.type != TokenType::TOKEN_EOF) {
                    if_stmt->else_branch.push_back(parse_statement());
                }
                expect(TokenType::TOKEN_END, "Expected 'end' closing else-block");
            } else {
                if_stmt->else_branch.push_back(parse_statement());
            }
        }
        return if_stmt;
    }

    // 3. Case Statement
    if (match(TokenType::TOKEN_CASE)) {
        expect(TokenType::TOKEN_LPAREN, "Expected '(' after 'case'");
        auto selector = parse_expression();
        expect(TokenType::TOKEN_RPAREN, "Expected ')' after case selector");

        auto case_stmt = std::make_unique<CaseStatementNode>(std::move(selector));

        while (current.type != TokenType::TOKEN_ENDCASE && current.type != TokenType::TOKEN_EOF) {
            CaseItem item;
            if (match(TokenType::TOKEN_DEFAULT)) {
                item.is_default = true;
                item.match_expr = nullptr;
                match(TokenType::TOKEN_COLON);
            } else {
                item.is_default = false;
                item.match_expr = parse_expression();
                expect(TokenType::TOKEN_COLON, "Expected ':' after case item value");
            }

            if (match(TokenType::TOKEN_BEGIN)) {
                while (current.type != TokenType::TOKEN_END && current.type != TokenType::TOKEN_EOF) {
                    item.statements.push_back(parse_statement());
                }
                expect(TokenType::TOKEN_END, "Expected 'end' closing case branch block");
            } else {
                item.statements.push_back(parse_statement());
            }

            case_stmt->items.push_back(std::move(item));
        }

        expect(TokenType::TOKEN_ENDCASE, "Expected 'endcase'");
        return case_stmt;
    }

    // 4. $display and $monitor (Supports both "$display(fmt, ...)" and "$monitor($time, fmt, ...)")
    if (current.type == TokenType::TOKEN_DISPLAY || current.type == TokenType::TOKEN_MONITOR) {
        bool is_mon = (current.type == TokenType::TOKEN_MONITOR);
        advance();
        expect(TokenType::TOKEN_LPAREN, "Expected '(' after display/monitor");

        bool has_lead_time = false;
        std::string fmt = "";
        std::vector<std::string> args;

        // Check if first argument is $time or an identifier before the format string
        if (current.type == TokenType::TOKEN_IDENTIFIER && current.text == "$time") {
            has_lead_time = true;
            advance();
            expect(TokenType::TOKEN_COMMA, "Expected ',' after $time");
        }

        if (current.type == TokenType::TOKEN_STRING) {
            fmt = current.text;
            advance();
        }

        while (match(TokenType::TOKEN_COMMA)) {
            args.push_back(current.text);
            expect(TokenType::TOKEN_IDENTIFIER, "Expected identifier in argument list");
        }

        expect(TokenType::TOKEN_RPAREN, "Expected ')'");
        expect(TokenType::TOKEN_SEMICOLON, "Expected ';'");
        return std::make_unique<DisplayStatementNode>(fmt, args, is_mon, has_lead_time);
    }

    // 5. $finish and $stop
    if (match(TokenType::TOKEN_FINISH) || match(TokenType::TOKEN_STOP)) {
        if (match(TokenType::TOKEN_LPAREN)) expect(TokenType::TOKEN_RPAREN, "Expected ')'");
        expect(TokenType::TOKEN_SEMICOLON, "Expected ';'");
        return std::make_unique<FinishStatementNode>();
    }

    // 6. $dumpfile
    if (match(TokenType::TOKEN_DUMPFILE)) {
        expect(TokenType::TOKEN_LPAREN, "Expected '(' after $dumpfile");
        std::string fname = current.text;
        expect(TokenType::TOKEN_STRING, "Expected filename string for $dumpfile");
        expect(TokenType::TOKEN_RPAREN, "Expected ')'");
        expect(TokenType::TOKEN_SEMICOLON, "Expected ';'");
        return std::make_unique<DumpFileStatementNode>(fname);
    }

    // 7. $dumpvars
    if (match(TokenType::TOKEN_DUMPVARS)) {
        if (match(TokenType::TOKEN_LPAREN)) {
            while (current.type != TokenType::TOKEN_RPAREN && current.type != TokenType::TOKEN_EOF) {
                advance();
            }
            expect(TokenType::TOKEN_RPAREN, "Expected ')'");
        }
        expect(TokenType::TOKEN_SEMICOLON, "Expected ';'");
        return std::make_unique<DumpVarsStatementNode>();
    }

    // 8. Statements beginning with an identifier
    if (current.type == TokenType::TOKEN_IDENTIFIER) {
        std::string lhs_name = current.text;
        advance();

        if (match(TokenType::TOKEN_SEMICOLON)) {
            return std::make_unique<TaskCallNode>(lhs_name);
        }

        if (match(TokenType::TOKEN_LBRACKET)) {
            auto addr_expr = parse_expression();
            expect(TokenType::TOKEN_RBRACKET, "Expected ']' closing array index on LHS");

            bool is_non_blocking = false;
            if (match(TokenType::TOKEN_LE)) {
                is_non_blocking = true;
            } else {
                expect(TokenType::TOKEN_EQUAL, "Expected '=' or '<=' in array assignment");
            }
            auto rhs_expr = parse_expression();
            expect(TokenType::TOKEN_SEMICOLON, "Expected ';' terminating array assignment");
            return std::make_unique<MemAssignNode>(lhs_name, std::move(addr_expr), std::move(rhs_expr), is_non_blocking);
        }

        std::vector<std::string> targets = { lhs_name };
        if (match(TokenType::TOKEN_LE)) {
            auto expr = parse_expression();
            expect(TokenType::TOKEN_SEMICOLON, "Expected ';' terminating non-blocking assignment");
            return std::make_unique<NonBlockingAssignNode>(targets[0], std::move(expr));
        } else {
            expect(TokenType::TOKEN_EQUAL, "Expected '=' or '<=' in assignment");
            auto expr = parse_expression();
            expect(TokenType::TOKEN_SEMICOLON, "Expected ';' terminating blocking assignment");
            return std::make_unique<BlockingAssignNode>(targets, std::move(expr));
        }
    }

    // 9. Concatenation Assignment: {a, b} = expr;
    if (match(TokenType::TOKEN_LBRACE)) {
        std::vector<std::string> targets;
        while (true) {
            targets.push_back(current.text);
            expect(TokenType::TOKEN_IDENTIFIER, "Expected identifier inside concatenation");
            if (match(TokenType::TOKEN_COMMA)) continue;
            break;
        }
        expect(TokenType::TOKEN_RBRACE, "Expected '}' in concatenation assignment");
        expect(TokenType::TOKEN_EQUAL, "Expected '=' in concatenation assignment");
        auto expr = parse_expression();
        expect(TokenType::TOKEN_SEMICOLON, "Expected ';' terminating blocking concatenation assignment");
        return std::make_unique<BlockingAssignNode>(targets, std::move(expr));
    }

    throw std::runtime_error("Unexpected statement beginning with '" + current.text + "' on line " + std::to_string(current.line));
}

std::vector<std::unique_ptr<ModuleNode>> Parser::parse_all() {
    std::vector<std::unique_ptr<ModuleNode>> modules;

    while (current.type != TokenType::TOKEN_EOF) {
        expect(TokenType::TOKEN_MODULE, "Expected 'module' keyword");
        auto mod = std::make_unique<ModuleNode>();
        mod->name = current.text;
        expect(TokenType::TOKEN_IDENTIFIER, "Expected module name");

        // Ports List: Handles ANSI (input Cin, x, y, output s, Cout) and Non-ANSI
        if (match(TokenType::TOKEN_LPAREN)) {
            std::string current_dir = "unspecified";
            uint32_t current_msb = 0, current_lsb = 0;

            while (current.type != TokenType::TOKEN_RPAREN && current.type != TokenType::TOKEN_EOF) {
                if (current.type == TokenType::TOKEN_INPUT) {
                    current_dir = "input";
                    current_msb = 0; current_lsb = 0;
                    advance();
                } else if (current.type == TokenType::TOKEN_OUTPUT) {
                    current_dir = "output";
                    current_msb = 0; current_lsb = 0;
                    advance();
                } else if (current.type == TokenType::TOKEN_INOUT) {
                    current_dir = "inout";
                    current_msb = 0; current_lsb = 0;
                    advance();
                }

                if (current.type == TokenType::TOKEN_LOGIC ||
                    current.type == TokenType::TOKEN_WIRE  ||
                    current.type == TokenType::TOKEN_REG) {
                    advance();
                }

                if (match(TokenType::TOKEN_LBRACKET)) {
                    current_msb = current.num_val; advance();
                    expect(TokenType::TOKEN_COLON, "Expected ':' in bus width");
                    current_lsb = current.num_val; advance();
                    expect(TokenType::TOKEN_RBRACKET, "Expected ']' in bus width");
                }

                PortNode port;
                port.direction = current_dir;
                port.msb = current_msb;
                port.lsb = current_lsb;
                port.name = current.text;
                expect(TokenType::TOKEN_IDENTIFIER, "Expected port identifier");
                mod->ports.push_back(port);

                if (!match(TokenType::TOKEN_COMMA)) break;
            }
            expect(TokenType::TOKEN_RPAREN, "Expected ')' closing port list");
        }
        expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after module header");

        // Module Body
        while (current.type != TokenType::TOKEN_ENDMODULE && current.type != TokenType::TOKEN_EOF) {
            // 1. Port declarations in body: input, output, inout
            if (current.type == TokenType::TOKEN_INPUT ||
                current.type == TokenType::TOKEN_OUTPUT ||
                current.type == TokenType::TOKEN_INOUT) {

                std::string dir = (current.type == TokenType::TOKEN_INPUT) ? "input" :
                                  (current.type == TokenType::TOKEN_OUTPUT) ? "output" : "inout";
                advance();

                if (current.type == TokenType::TOKEN_WIRE ||
                    current.type == TokenType::TOKEN_REG ||
                    current.type == TokenType::TOKEN_LOGIC) {
                    advance();
                }

                uint32_t msb = 0, lsb = 0;
                if (match(TokenType::TOKEN_LBRACKET)) {
                    msb = current.num_val; advance();
                    expect(TokenType::TOKEN_COLON, "Expected ':' in bus width");
                    lsb = current.num_val; advance();
                    expect(TokenType::TOKEN_RBRACKET, "Expected ']' in bus width");
                }

                while (true) {
                    std::string sig_name = current.text;
                    expect(TokenType::TOKEN_IDENTIFIER, "Expected port name in body declaration");

                    bool found = false;
                    for (auto& p : mod->ports) {
                        if (p.name == sig_name) {
                            p.direction = dir;
                            p.msb = msb;
                            p.lsb = lsb;
                            found = true;
                            break;
                        }
                    }
                    if (!found) {
                        PortNode p;
                        p.name = sig_name;
                        p.direction = dir;
                        p.msb = msb;
                        p.lsb = lsb;
                        mod->ports.push_back(p);
                    }

                    if (match(TokenType::TOKEN_COMMA)) continue;
                    break;
                }
                expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after port declaration");
            }
            // 2. Signals: logic, wire, reg
            else if (current.type == TokenType::TOKEN_LOGIC ||
                     current.type == TokenType::TOKEN_WIRE  ||
                     current.type == TokenType::TOKEN_REG) {
                advance();

                uint32_t msb = 0, lsb = 0;
                if (match(TokenType::TOKEN_LBRACKET)) {
                    msb = current.num_val; advance();
                    expect(TokenType::TOKEN_COLON, "Expected ':' in bus width");
                    lsb = current.num_val; advance();
                    expect(TokenType::TOKEN_RBRACKET, "Expected ']' in bus width");
                }

                while (true) {
                    PortNode sig;
                    sig.direction = "internal";
                    sig.msb = msb;
                    sig.lsb = lsb;
                    sig.name = current.text;
                    expect(TokenType::TOKEN_IDENTIFIER, "Expected signal name");

                    bool already_port = false;
                    for (auto& p : mod->ports) {
                        if (p.name == sig.name) {
                            already_port = true;
                            if (msb > 0) { p.msb = msb; p.lsb = lsb; }
                            break;
                        }
                    }

                    if (match(TokenType::TOKEN_LBRACKET)) {
                        uint32_t bound1 = current.num_val; advance();
                        if (match(TokenType::TOKEN_COLON)) {
                            uint32_t bound2 = current.num_val; advance();
                            expect(TokenType::TOKEN_RBRACKET, "Expected ']' closing array dimension");
                            sig.is_array = true;
                            sig.array_depth = (bound2 >= bound1) ? (bound2 - bound1 + 1) : (bound1 - bound2 + 1);
                        } else {
                            expect(TokenType::TOKEN_RBRACKET, "Expected ']' closing array dimension");
                            sig.is_array = true;
                            sig.array_depth = bound1;
                        }
                    }

                    if (!already_port) {
                        mod->signals.push_back(sig);
                    }

                    if (match(TokenType::TOKEN_COMMA)) continue;
                    break;
                }
                expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after signal declaration");
            }
            // 3. Built-in Gate Primitives
            else if (current.type == TokenType::TOKEN_GATE_AND  ||
                     current.type == TokenType::TOKEN_GATE_OR   ||
                     current.type == TokenType::TOKEN_OR        ||
                     current.type == TokenType::TOKEN_GATE_XOR  ||
                     current.type == TokenType::TOKEN_GATE_NAND ||
                     current.type == TokenType::TOKEN_GATE_NOR  ||
                     current.type == TokenType::TOKEN_GATE_XNOR ||
                     current.type == TokenType::TOKEN_GATE_NOT  ||
                     current.type == TokenType::TOKEN_GATE_BUF) {

                auto gate = std::make_unique<GatePrimitiveNode>();
                if (current.type == TokenType::TOKEN_GATE_AND) gate->type = GateType::AND;
                else if (current.type == TokenType::TOKEN_GATE_OR || current.type == TokenType::TOKEN_OR) gate->type = GateType::OR;
                else if (current.type == TokenType::TOKEN_GATE_XOR) gate->type = GateType::XOR;
                else if (current.type == TokenType::TOKEN_GATE_NAND) gate->type = GateType::NAND;
                else if (current.type == TokenType::TOKEN_GATE_NOR) gate->type = GateType::NOR;
                else if (current.type == TokenType::TOKEN_GATE_XNOR) gate->type = GateType::XNOR;
                else if (current.type == TokenType::TOKEN_GATE_NOT) gate->type = GateType::NOT;
                else if (current.type == TokenType::TOKEN_GATE_BUF) gate->type = GateType::BUF;
                advance();

                if (current.type == TokenType::TOKEN_IDENTIFIER) {
                    gate->instance_name = current.text;
                    advance();
                } else {
                    gate->instance_name = "gate_" + std::to_string(mod->gates.size());
                }

                expect(TokenType::TOKEN_LPAREN, "Expected '(' in gate connection list");
                gate->out_expr = parse_expression();

                while (match(TokenType::TOKEN_COMMA)) {
                    gate->in_exprs.push_back(parse_expression());
                }
                expect(TokenType::TOKEN_RPAREN, "Expected ')' closing gate connections");
                expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after gate instantiation");
                mod->gates.push_back(std::move(gate));
            }
            // 4. Continuous assign
            else if (match(TokenType::TOKEN_ASSIGN)) {
                std::string target = current.text;
                expect(TokenType::TOKEN_IDENTIFIER, "Expected target identifier in continuous assign");
                expect(TokenType::TOKEN_EQUAL, "Expected '=' in continuous assign");
                auto rhs = parse_expression();
                expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after continuous assign");
                mod->continuous_assigns.push_back(std::make_unique<ContinuousAssignNode>(target, std::move(rhs)));
            }
            // 5. Tasks
            else if (match(TokenType::TOKEN_TASK)) {
                auto task = std::make_unique<TaskNode>();
                task->name = current.text;
                expect(TokenType::TOKEN_IDENTIFIER, "Expected task name");
                match(TokenType::TOKEN_SEMICOLON);

                while (current.type != TokenType::TOKEN_ENDTASK && current.type != TokenType::TOKEN_EOF) {
                    task->body.push_back(parse_statement());
                }
                expect(TokenType::TOKEN_ENDTASK, "Expected 'endtask'");
                mod->tasks.push_back(std::move(task));
            }
            // 6. always_comb
            else if (match(TokenType::TOKEN_ALWAYS_COMB)) {
                auto proc = std::make_unique<ProcessNode>(ProcessType::COMBINATIONAL);
                expect(TokenType::TOKEN_BEGIN, "Expected 'begin' block");
                while (current.type != TokenType::TOKEN_END && current.type != TokenType::TOKEN_EOF) {
                    proc->body.push_back(parse_statement());
                }
                expect(TokenType::TOKEN_END, "Expected 'end'");
                mod->processes.push_back(std::move(proc));
            }
            // 7. always_ff or generic always
            else if (match(TokenType::TOKEN_ALWAYS_FF) || match(TokenType::TOKEN_ALWAYS)) {
                auto proc = std::make_unique<ProcessNode>(ProcessType::SEQUENTIAL);
                expect(TokenType::TOKEN_AT, "Expected '@' in always block sensitivity list");
                expect(TokenType::TOKEN_LPAREN, "Expected '(' in sensitivity list");

                bool has_edge = false;
                while (current.type != TokenType::TOKEN_RPAREN && current.type != TokenType::TOKEN_EOF) {
                    while (current.type == TokenType::TOKEN_OR ||
                           current.type == TokenType::TOKEN_GATE_OR ||
                           current.type == TokenType::TOKEN_OR_BIT ||
                           current.type == TokenType::TOKEN_COMMA) {
                        advance();
                    }

                    if (current.type == TokenType::TOKEN_RPAREN) break;

                    EdgeType edge = EdgeType::LEVEL;
                    if (match(TokenType::TOKEN_POSEDGE)) {
                        edge = EdgeType::POSEDGE;
                        has_edge = true;
                    } else if (match(TokenType::TOKEN_NEGEDGE)) {
                        edge = EdgeType::NEGEDGE;
                        has_edge = true;
                    }

                    std::string sig = current.text;
                    expect(TokenType::TOKEN_IDENTIFIER, "Expected signal name in sensitivity list");
                    proc->sensitivity_list.push_back({edge, sig});

                    while (current.type == TokenType::TOKEN_OR ||
                           current.type == TokenType::TOKEN_GATE_OR ||
                           current.type == TokenType::TOKEN_OR_BIT ||
                           current.type == TokenType::TOKEN_COMMA) {
                        advance();
                    }
                }
                expect(TokenType::TOKEN_RPAREN, "Expected ')' closing sensitivity list");

                if (!has_edge) {
                    proc->type = ProcessType::COMBINATIONAL;
                }

                if (match(TokenType::TOKEN_BEGIN)) {
                    while (current.type != TokenType::TOKEN_END && current.type != TokenType::TOKEN_EOF) {
                        proc->body.push_back(parse_statement());
                    }
                    expect(TokenType::TOKEN_END, "Expected 'end'");
                } else {
                    proc->body.push_back(parse_statement());
                }
                mod->processes.push_back(std::move(proc));
            }
            // 8. initial
            else if (match(TokenType::TOKEN_INITIAL)) {
                auto proc = std::make_unique<ProcessNode>(ProcessType::INITIAL);
                if (match(TokenType::TOKEN_BEGIN)) {
                    while (current.type != TokenType::TOKEN_END && current.type != TokenType::TOKEN_EOF) {
                        proc->body.push_back(parse_statement());
                    }
                    expect(TokenType::TOKEN_END, "Expected 'end'");
                } else {
                    proc->body.push_back(parse_statement());
                }
                mod->processes.push_back(std::move(proc));
            }
            // 9. Module Instantiations
            else if (current.type == TokenType::TOKEN_IDENTIFIER) {
                std::string mod_type = current.text;
                advance();
                std::string inst_name = current.text;
                expect(TokenType::TOKEN_IDENTIFIER, "Expected instance name");
                expect(TokenType::TOKEN_LPAREN, "Expected '(' for instance port map");

                bool is_pos = (current.type != TokenType::TOKEN_DOT);
                auto inst = std::make_unique<InstantiationNode>(mod_type, inst_name, is_pos);

                while (current.type != TokenType::TOKEN_RPAREN && current.type != TokenType::TOKEN_EOF) {
                    PortConnection conn;
                    if (!is_pos) {
                        expect(TokenType::TOKEN_DOT, "Expected '.' before port name");
                        conn.port_name = current.text;
                        expect(TokenType::TOKEN_IDENTIFIER, "Expected port name");
                        expect(TokenType::TOKEN_LPAREN, "Expected '(' after port name");
                        conn.expr = parse_expression();
                        expect(TokenType::TOKEN_RPAREN, "Expected ')' closing port expression");
                    } else {
                        conn.port_name = "";
                        conn.expr = parse_expression();
                    }
                    inst->connections.push_back(std::move(conn));
                    if (!match(TokenType::TOKEN_COMMA)) break;
                }

                expect(TokenType::TOKEN_RPAREN, "Expected ')' closing instantiation");
                expect(TokenType::TOKEN_SEMICOLON, "Expected ';' after instantiation");
                mod->instantiations.push_back(std::move(inst));
            }
            else {
                advance();
            }
        }
        expect(TokenType::TOKEN_ENDMODULE, "Expected 'endmodule'");
        modules.push_back(std::move(mod));
    }

    return modules;
}

// Design
// D flip-flop
module dff (clk, reset,
  d, q, qb);
  input      clk;
  input      reset;
  input      d;
  output     q;
  output     qb;

  reg        q;

  assign qb = ~q;

  always @(posedge clk or posedge reset)
  begin
    if (reset) begin
      // Asynchronous reset when reset goes high
      q <= 1'b0;
    end else begin
      // Assign D to Q on positive clock edge
      q <= d;
    end
  end

module alu_core (
    input  logic [3:0] op_a,
    input  logic [3:0] op_b,
    input  logic [2:0] opcode,
    output logic [3:0] alu_out
);

always_comb begin
    case (opcode)
        3'd0: alu_out = op_a & op_b;
        3'd1: alu_out = op_a | op_b;
        3'd2: alu_out = op_a ^ op_b;
        3'd3: alu_out = ~op_a;
        3'd4: alu_out = op_a << 4'd1;
        3'd5: alu_out = op_a >> 4'd1;
        3'd6: alu_out = op_a + op_b;
        3'd7: alu_out = op_a - op_b;
        default: alu_out = 4'd0;
    endcase
end

endmodule

module regression_dut (
    input  logic       clk,
    input  logic       rst_n,
    input  logic [3:0] a,
    input  logic [3:0] b,
    input  logic [2:0] opcode,
    input  logic [1:0] mem_addr,
    input  logic       mem_we,
    output logic [3:0] alu_result,
    output logic [3:0] reg_accum,
    output logic       is_equal,
    output logic       is_less,
    output logic       bit_extracted,
    output logic [1:0] slice_extracted,
    output logic [3:0] cont_assign_out,
    output logic [3:0] mem_rdata
);

    // Continuous assign statement (v0.9.0 dataflow)
    assign cont_assign_out = a + 4'd1;

    // 1D Unpacked Memory Array: 4 words x 4 bits (v0.9.0)
    logic [3:0] ram [0:3];

    // Asynchronous memory read
    assign mem_rdata = ram[mem_addr];

    alu_core u_core (
        .op_a(a),
        .op_b(b),
        .opcode(opcode),
        .alu_out(alu_result)
    );

    always_comb begin
        is_equal = (a == b) ? 1'd1 : 1'd0;
        is_less  = (a < b)  ? 1'd1 : 1'd0;

        bit_extracted   = a[2];
        slice_extracted = a[3:2];
    end

    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            reg_accum <= 4'd0;
        end else begin
            reg_accum <= reg_accum + alu_result;
            // Synchronous memory write
            if (mem_we) begin
                ram[mem_addr] <= a;
            end
        end
    end

endmodule

# CAS: Cycle-Accurate Simulator

**Current Version:** `v0.9.0`  
**License:** MIT  
**Target Standard:** IEEE 1800 SystemVerilog (subset)

CAS is an independent, ground-up discrete-event RTL simulator implemented in C++17. It operates without external compiler chains (LLVM, intermediate C++ transpilation) or third-party frontends. CAS parses SystemVerilog sources directly into an in-memory Abstract Syntax Tree (AST), elaborates arbitrary-depth hierarchical netlist graphs, and executes simulation events using an IEEE 1800-inspired stratified scheduler (Active, NBA, Combinational Convergence, and Time Wheel).

---

## Architectural Highlights

* **Continuous `assign` Dataflow Modeling:** Top-level module continuous assignments dynamically evaluated and converged within the delta-cycle loop without requiring procedural wrapper processes.
* **1D Unpacked Memory Arrays:** Support for memory structures (`logic [W-1:0] mem [0:D-1]`) with dual-rail 4-state unknown initialization, dynamic asynchronous reads (`mem[addr]`), and synchronous non-blocking writes (`mem[addr] <= data`).
* **Multi-Way Control Flow Selection (`case` / `endcase`):** Native branch selection supporting exact value matching, `default` fallbacks, and multi-statement `begin ... end` case items.
* **Vector Bit-Select & Part-Select Slicing:** Dynamic scalar extraction (`wire[idx]`) and fixed range slicing (`wire[msb:lsb]`) with dual-rail 4-state bit alignment.
* **Full ALU Operator Suite & Expression Precedence:** Operator precedence climbing supporting bitwise (`&`, `|`, `^`, `~`), logical shifts (`<<`, `>>`), relational/equality (`==`, `!=`, `<`, `<=`, `>`, `>=`), logical short-circuit gates (`&&`, `||`), and ternary conditional evaluation (`cond ? true_expr : false_expr`).
* **Recursive Multi-Level Module Elaboration:** Fully recursive elaborator traversing arbitrary submodule instantiation depths (`top -> dut -> submod -> ...`), dynamically instantiating scoped nets and binding bidirectional port interconnects.
* **Dual-Rail 4-State Bitwise Evaluation:** Exact per-bit IEEE 1800 evaluation rules for unknown bits (`0 & X = 0`, `1 | X = 1`, `X ^ X = X`, `~X = X`), preserving hardware-accurate uninitialized propagation across wide datapath buses.
* **Native 4-State Logic Engine (`0, 1, X, Z`):** Dual-rail bit-vector arithmetic (`aval` data, `bval` unknown mask). All un-driven nets, memory cells, and uninitialized registers initialize to `X` (unknown) rather than optimistic `0`, catching un-reset design bugs.
* **Self-Checking Regression Suite:** Automated verification harness covering all milestone deliverables from `v0.1.0` through `v0.9.0` with diagnostic pass/fail accounting.
* **Project-Centric Workspace Engine:** Isolated directory hierarchies (`projects/<name>/`) storing RTL designs (`design.sv`), testbenches (`testbench.sv`), metadata (`project.json`), and archived waveform traces (`waves/`).
* **Persistent Waveform History:** Timestamped VCD trace runs stored per project with in-browser selection, deletion, and immediate canvas invalidation when dump directives are omitted.
* **4-State Digital Timing Canvas:** Native HTML5 Canvas timing diagram renderer displaying clocks, single-bit logic wires, and multibit transition buses with distinct visual styles for known values vs. `X`/`Z` states.
* **Native VCD Waveform Generation:** Direct export of IEEE 1364/1800 Value Change Dump (`.vcd`) traces supporting `$dumpfile` and `$dumpvars` with hierarchical scope definitions and compact ASCII symbol mapping.
* **Stratified IEEE 1800 Scheduling (Active vs. NBA):** Evaluates non-blocking assignments (`<=`) during the active region and buffers values into an NBA update queue, committing all register transitions simultaneously before triggering combinational delta passes.
* **Edge-Triggered Sensitivity Engine:** Dynamic transition tracking (`is_posedge()`, `is_negedge()`) across hierarchical signal handles for `always_ff` blocks with asynchronous resets (`X -> 0` correctly triggers `negedge`).
* **Native Discrete-Event Time Wheel:** Min-heap priority queue (`std::priority_queue`) that advances simulation time dynamically across `#delay` intervals.

app = FastAPI()

PROJECTS_DIR = "/app/projects"

# ----------------- Project Management APIs ----------------- #

@app.get("/api/projects")
def list_projects():
    projects = []
    if not os.path.exists(PROJECTS_DIR):
        return {"projects": []}

    for name in os.listdir(PROJECTS_DIR):
        pdir = os.path.join(PROJECTS_DIR, name)
        if os.path.isdir(pdir):
            waves_dir = os.path.join(pdir, "waves")
            vcd_count = len(glob.glob(os.path.join(waves_dir, "*.vcd"))) if os.path.exists(waves_dir) else 0
            mtime = os.path.getmtime(pdir)
            projects.append({
                "name": name,
                "last_modified": datetime.fromtimestamp(mtime).strftime("%Y-%m-%d %H:%M:%S"),
                "vcd_count": vcd_count
            })
    projects.sort(key=lambda x: x["last_modified"], reverse=True)
    return {"projects": projects}

@app.post("/api/projects/save")
def save_project(req: SaveProjectRequest):
    pname = req.project_name.strip()
    if not pname:
        raise HTTPException(status_code=400, detail="Project name cannot be empty.")
    
    # Sanitize name
    pname = "".join(c for c in pname if c.isalnum() or c in ('_', '-'))
    pdir = os.path.join(PROJECTS_DIR, pname)
    waves_dir = os.path.join(pdir, "waves")
    os.makedirs(waves_dir, exist_ok=True)

    with open(os.path.join(pdir, "design.sv"), "w") as f:
        f.write(req.design_code)

    with open(os.path.join(pdir, "testbench.sv"), "w") as f:
        f.write(req.tb_code)

    meta = {
        "project_name": pname,
        "saved_at": datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    }
    with open(os.path.join(pdir, "project.json"), "w") as f:
        json.dump(meta, f, indent=2)

    return {"status": "success", "project_name": pname}

@app.get("/api/projects/{name}")
def load_project(name: str):
    pdir = os.path.join(PROJECTS_DIR, name)
    if not os.path.exists(pdir):
        raise HTTPException(status_code=404, detail="Project not found.")

    design_code = ""
    tb_code = ""
    d_path = os.path.join(pdir, "design.sv")
    t_path = os.path.join(pdir, "testbench.sv")

    if os.path.exists(d_path):
        with open(d_path, "r") as f:
            design_code = f.read()

    if os.path.exists(t_path):
        with open(t_path, "r") as f:
            tb_code = f.read()

    waves = []
    waves_dir = os.path.join(pdir, "waves")
    if os.path.exists(waves_dir):
        for wf in os.listdir(waves_dir):
            if wf.endswith(".vcd"):
                wpath = os.path.join(waves_dir, wf)
                waves.append({
                    "filename": wf,
                    "timestamp": datetime.fromtimestamp(os.path.getmtime(wpath)).strftime("%Y-%m-%d %H:%M:%S")
                })
    waves.sort(key=lambda x: x["timestamp"], reverse=True)

    return {
        "project_name": name,
        "design_code": design_code,
        "tb_code": tb_code,
        "waves": waves
    }

@app.post("/api/simulate")
def simulate(req: SimRequest):
    if not req.design_code.strip() and not req.tb_code.strip():
        return {"status": "error", "output": "[ERROR] Both RTL and Testbench are empty.", "vcd": None, "vcd_filename": None}

    with tempfile.TemporaryDirectory() as tmpdir:
        files_to_run = []

        if req.tb_code.strip():
            tb_path = os.path.join(tmpdir, "testbench.sv")
            with open(tb_path, "w") as f:
                f.write(req.tb_code)
            files_to_run.append(tb_path)

        if req.design_code.strip():
            dut_path = os.path.join(tmpdir, "design.sv")
            with open(dut_path, "w") as f:
                f.write(req.design_code)
            files_to_run.append(dut_path)

        cas_bin = "/app/CAS/build/cas"
        run_cmd = [cas_bin] + files_to_run

        try:
            res = subprocess.run(run_cmd, cwd=tmpdir, capture_output=True, text=True, timeout=10)
            output_msg = res.stdout if res.returncode == 0 else (res.stderr + "\n" + res.stdout)
            vcd_content = None
            vcd_filename = None

            for fname in os.listdir(tmpdir):
                if fname.endswith(".vcd"):
                    vcd_filename = fname
                    with open(os.path.join(tmpdir, fname), "r") as vf:
                        vcd_content = vf.read()

                    # If project is active, persist to project's waves/ folder with a timestamped prefix
                    if req.project_name:
                        pdir = os.path.join(PROJECTS_DIR, req.project_name)
                        if os.path.exists(pdir):
                            pwaves = os.path.join(pdir, "waves")
                            os.makedirs(pwaves, exist_ok=True)
                            ts = datetime.now().strftime("%Y%m%d_%H%M%S")
                            saved_name = f"{ts}_{fname}"
                            shutil.copy2(os.path.join(tmpdir, fname), os.path.join(pwaves, saved_name))
                            vcd_filename = saved_name
                    break

            return {
                "status": "success" if res.returncode == 0 else "error",
                "output": output_msg,
                "vcd": vcd_content,
                "vcd_filename": vcd_filename
            }
        except Exception as e:
            return {"status": "error", "output": f"[CAS INVOCATION ERROR] {str(e)}", "vcd": None, "vcd_filename": None}

            document.getElementById('active-project-label').textContent = activeProject;

        document.getElementById('active-project-label').textContent = activeProject;
        rtlEditor.setValue(data.design_code);
        tbEditor.setValue(data.tb_code);

        refreshWaveDropdown(data.waves);

    async function deleteProject(e, name) {

            await fetch(`/api/projects/${name}`, { method: 'DELETE' });
            if (activeProject === name) {
                newProject();

    async function executeSimulation() {
        const term = document.getElementById('panel-terminal');
        const btnRun = document.getElementById('btn-run');
        const bulb = document.getElementById('status-bulb');
        const statusMsg = document.getElementById('status-msg');

        btnRun.disabled = true;
        bulb.className = "status-indicator busy";
        statusMsg.textContent = "Simulating...";
        term.textContent = "[CAS] Executing simulation...\n";

        try {
            const res = await fetch('/api/simulate', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    design_code: rtlEditor.getValue(),
                    tb_code: tbEditor.getValue(),
                    top_module: "tb_counter",
                    project_name: activeProject
                })
            });

            const data = await res.json();
            term.textContent = data.output;

            if (data.status === "success") {
                bulb.className = "status-indicator ready";
                statusMsg.textContent = "Simulation Converged";
            } else {
                bulb.className = "status-indicator err";
                statusMsg.textContent = "Simulation Failed";
            }

            if (data.vcd) {
                activeVcdContent = data.vcd;
                activeVcdFilename = data.vcd_filename;
                document.getElementById('panel-vcd-raw').textContent = data.vcd;
                document.getElementById('btn-export-vcd').disabled = false;
                renderActiveWaveform();

                // Refresh wave dropdown if inside an active project
                if (activeProject) {
                    const pres = await fetch(`/api/projects/${activeProject}`);
                    const pdata = await pres.json();
                    refreshWaveDropdown(pdata.waves);
                    document.getElementById('wave-select').value = data.vcd_filename;
                }
            } else {
                // Strict clear if no dump occurred in this simulation
                clearWaveformState();
            }
        } catch (e) {
            term.textContent = "[UI FETCH ERROR] " + e.message;
            bulb.className = "status-indicator err";
            statusMsg.textContent = "Invocation Error";
        } finally {
            btnRun.disabled = false;
        }
    }

    // 6. 4-State Digital Waveform Canvas Rendering
    function renderActiveWaveform() {
        if (!activeVcdContent) {
            document.getElementById('wave-empty').style.display = "flex";
            return;
        }
        document.getElementById('wave-empty').style.display = "none";

        const namesContainer = document.getElementById('wave-names');
        const canvas = document.getElementById('wave-canvas');
        const ctx = canvas.getContext('2d');
        namesContainer.innerHTML = '';

        const lines = activeVcdContent.split('\n');
        const idToVar = {};
        const signals = [];
        let maxTime = 0;
        let inVars = false;

        for (let line of lines) {
            line = line.trim();
            if (line.startsWith('$var')) {
                const parts = line.split(/\s+/);
                const width = parseInt(parts[2]);
                const id = parts[3];
                const name = parts[4];
                const sig = { id, name, width, changes: [] };
                idToVar[id] = sig;
                signals.push(sig);
            } else if (line.startsWith('$enddefinitions')) {
                inVars = true;
            } else if (inVars && line.startsWith('#')) {
                const t = parseInt(line.substring(1));
                if (t > maxTime) maxTime = t;
            }
        }

        if (maxTime === 0) maxTime = 50;

        let curTime = 0;
        for (let line of lines) {
            line = line.trim();
            if (!line || line.startsWith('$')) continue;
            if (line.startsWith('#')) {
                curTime = parseInt(line.substring(1));
                continue;
            }

            if (line.startsWith('b')) {
                const parts = line.substring(1).split(/\s+/);
                const val = parts[0];
                const id = parts[1];
                if (idToVar[id]) {
                    const isX = val.includes('x') || val.includes('z');
                    idToVar[id].changes.push({ time: curTime, val: isX ? 'x' : parseInt(val, 2), raw: val });
                }
            } else {
                const charVal = line[0].toLowerCase();
                const id = line.substring(1);
                if (idToVar[id]) {
                    idToVar[id].changes.push({ time: curTime, val: charVal });
                }
            }
        }

        const rowHeight = 36;
        const timeHeaderHeight = 28;
        const timeStepPx = 14;
        const totalWidth = Math.max(800, (maxTime + 10) * timeStepPx);
        const totalHeight = timeHeaderHeight + (signals.length * rowHeight);

        canvas.width = totalWidth;
        canvas.height = totalHeight;

        signals.forEach(s => {
            const div = document.createElement('div');
            div.className = 'wave-sig-row';
            div.innerHTML = `<span>${s.name}</span><span class="wave-sig-type">${s.width > 1 ? '[' + (s.width-1) + ':0]' : 'wire'}</span>`;
            namesContainer.appendChild(div);
        });

        ctx.fillStyle = '#0b0c0e';
        ctx.fillRect(0, 0, totalWidth, totalHeight);

        // Ruler bar
        ctx.fillStyle = '#16171d';
        ctx.fillRect(0, 0, totalWidth, timeHeaderHeight);
        ctx.strokeStyle = '#272a33';
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.moveTo(0, timeHeaderHeight);
        ctx.lineTo(totalWidth, timeHeaderHeight);
        ctx.stroke();

        ctx.font = '10px JetBrains Mono';
        ctx.fillStyle = '#8b949e';

        for (let t = 0; t <= maxTime + 5; t += 5) {
            const x = t * timeStepPx;
            ctx.beginPath();
            ctx.moveTo(x, timeHeaderHeight - 6);
            ctx.lineTo(x, totalHeight);
            ctx.strokeStyle = (t % 10 === 0) ? '#272a33' : '#1a1b22';
            ctx.stroke();
            ctx.fillText(t + 'ns', x + 3, timeHeaderHeight - 8);
        }

        // Draw Waves
        signals.forEach((sig, idx) => {
            const yTop = timeHeaderHeight + (idx * rowHeight);
            const yMid = yTop + (rowHeight / 2);
            const yHigh = yTop + 7;
            const yLow = yTop + rowHeight - 7;

            ctx.strokeStyle = '#181920';
            ctx.beginPath();
            ctx.moveTo(0, yTop + rowHeight);
            ctx.lineTo(totalWidth, yTop + rowHeight);
            ctx.stroke();

            if (sig.changes.length === 0) return;

            if (sig.width === 1) {
                // 1-Bit Scalar Digital Wire with 4-State handling
                for (let i = 0; i < sig.changes.length; i++) {
                    const c = sig.changes[i];
                    const nextTime = (i + 1 < sig.changes.length) ? sig.changes[i+1].time : (maxTime + 5);
                    const xStart = c.time * timeStepPx;
                    const xEnd = nextTime * timeStepPx;

                    ctx.lineWidth = 2;
                    if (c.val === 'x' || c.val === 'z') {
                        ctx.strokeStyle = '#ef4444'; // Red for Unknown / Z
                        ctx.beginPath();
                        ctx.moveTo(xStart, yMid);
                        ctx.lineTo(xEnd, yMid);
                        ctx.stroke();
                    } else {
                        ctx.strokeStyle = '#10b981'; // Emerald for Known
                        const yVal = (c.val === '1' || c.val === 1) ? yHigh : yLow;
                        ctx.beginPath();
                        ctx.moveTo(xStart, yVal);
                        ctx.lineTo(xEnd, yVal);
                        ctx.stroke();
                    }

                    // Vertical transition line
                    if (i + 1 < sig.changes.length) {
                        const nextVal = sig.changes[i+1].val;
                        const yCur = (c.val === 'x' || c.val === 'z') ? yMid : ((c.val === '1' || c.val === 1) ? yHigh : yLow);
                        const yNext = (nextVal === 'x' || nextVal === 'z') ? yMid : ((nextVal === '1' || nextVal === 1) ? yHigh : yLow);
                        ctx.strokeStyle = (nextVal === 'x' || c.val === 'x') ? '#ef4444' : '#10b981';
                        ctx.beginPath();
                        ctx.moveTo(xEnd, yCur);
                        ctx.lineTo(xEnd, yNext);
                        ctx.stroke();
                    }
                }
            } else {
                // Multi-Bit Bus Box with 4-state Unknown support
                for (let i = 0; i < sig.changes.length; i++) {
                    const c = sig.changes[i];
                    const nextTime = (i + 1 < sig.changes.length) ? sig.changes[i+1].time : (maxTime + 5);
                    const xStart = c.time * timeStepPx;
                    const xEnd = nextTime * timeStepPx;
                    const boxW = xEnd - xStart;

                    if (boxW <= 0) continue;

                    const isX = (c.val === 'x');
                    ctx.strokeStyle = isX ? '#ef4444' : '#38bdf8';
                    ctx.fillStyle = isX ? '#450a0a' : '#1e293b';
                    ctx.lineWidth = 1.5;

                    ctx.beginPath();
                    ctx.moveTo(xStart + 3, yHigh);
                    ctx.lineTo(xEnd - 3, yHigh);
                    ctx.lineTo(xEnd, yMid);
                    ctx.lineTo(xEnd - 3, yLow);
                    ctx.lineTo(xStart + 3, yLow);
                    ctx.lineTo(xStart, yMid);
                    ctx.closePath();
                    ctx.fill();
                    ctx.stroke();

                    ctx.fillStyle = isX ? '#fca5a5' : '#e2e8f0';
                    ctx.font = '11px JetBrains Mono';
                    const label = isX ? "X" : ("0x" + c.val.toString(16).toUpperCase());
                    if (boxW > 25) ctx.fillText(label, xStart + 8, yMid + 4);
                }
            }
        });
    }

    // 7. Draggable Splitters
    const resizerMain = document.getElementById('resizer-main');
    const editorsWrapper = document.getElementById('editors-wrapper');
    const workspace = document.getElementById('workspace');

    let isDraggingV = false;
    resizerMain.addEventListener('mousedown', () => isDraggingV = true);

    const resizerH = document.getElementById('resizer-editors');
    const colTb = document.getElementById('col-tb');
    let isDraggingH = false;
    resizerH.addEventListener('mousedown', () => isDraggingH = true);

    window.addEventListener('mousemove', (e) => {
        if (isDraggingV) {
            const h = workspace.offsetHeight;
            const topH = ((e.clientY - workspace.offsetTop) / h) * 100;
            if (topH > 20 && topH < 85) editorsWrapper.style.height = topH + '%';
        }
        if (isDraggingH) {
            const w = editorsWrapper.offsetWidth;
            const leftW = ((e.clientX - editorsWrapper.offsetLeft) / w) * 100;
            if (leftW > 20 && leftW < 80) {
                colTb.style.flex = "none";
                colTb.style.width = leftW + '%';
            }
        }
    });

    window.addEventListener('mouseup', () => {
        isDraggingV = false;
        isDraggingH = false;
    });
</script>

</body>
</html>
