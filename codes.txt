from fastapi import FastAPI, HTTPException
from fastapi.staticfiles import StaticFiles
from pydantic import BaseModel
from typing import Optional, Dict
import subprocess
import tempfile
import shutil
import json
import os
import glob
import re
from datetime import datetime

app = FastAPI()

PROJECTS_DIR = "/app/projects"
os.makedirs(PROJECTS_DIR, exist_ok=True)

class SimRequest(BaseModel):
    files: Dict[str, str]
    entry_tb: Optional[str] = "testbench.sv"
    top_module: Optional[str] = ""
    project_name: Optional[str] = None
    enable_profiler: Optional[bool] = True
    enable_debug: Optional[bool] = True
    enable_vcd: Optional[bool] = True
    vcd_filename: Optional[str] = "trace.vcd"
    vcd_levels: Optional[int] = 0
    enable_display: Optional[bool] = True
    enable_monitor: Optional[bool] = True
    enable_finish_msg: Optional[bool] = True
    timescale: Optional[str] = "1ns"

class SaveProjectRequest(BaseModel):
    project_name: str
    files: Dict[str, str]
    settings: Optional[Dict] = {}
    status: Optional[Dict] = {}

class RenameProjectRequest(BaseModel):
    old_name: str
    new_name: str

# ----------------- Project Management APIs ----------------- #

@app.get("/api/projects")
def list_projects():
    projects = []
    if not os.path.exists(PROJECTS_DIR):
        return {"projects": []}

    for name in os.listdir(PROJECTS_DIR):
        pdir = os.path.join(PROJECTS_DIR, name)
        if os.path.isdir(pdir):
            waves_dir = os.path.join(pdir, "waves")
            vcd_count = len(glob.glob(os.path.join(waves_dir, "*.vcd"))) if os.path.exists(waves_dir) else 0
            mtime = os.path.getmtime(pdir)
            projects.append({
                "name": name,
                "last_modified": datetime.fromtimestamp(mtime).strftime("%Y-%m-%d %H:%M:%S"),
                "vcd_count": vcd_count
            })
    projects.sort(key=lambda x: x["last_modified"], reverse=True)
    return {"projects": projects}

@app.post("/api/projects/save")
def save_project(req: SaveProjectRequest):
    pname = req.project_name.strip()
    if not pname:
        raise HTTPException(status_code=400, detail="Project name cannot be empty.")
    
    pname = "".join(c for c in pname if c.isalnum() or c in ('_', '-'))
    pdir = os.path.join(PROJECTS_DIR, pname)
    waves_dir = os.path.join(pdir, "waves")
    os.makedirs(waves_dir, exist_ok=True)

    for f in os.listdir(pdir):
        if f != "waves" and f != "project.json":
            fpath = os.path.join(pdir, f)
            if os.path.isfile(fpath):
                os.remove(fpath)

    for fname, content in req.files.items():
        with open(os.path.join(pdir, fname), "w") as f:
            f.write(content)

    meta = {
        "project_name": pname,
        "saved_at": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
        "files": list(req.files.keys()),
        "settings": req.settings or {},
        "status": req.status or {}
    }
    with open(os.path.join(pdir, "project.json"), "w") as f:
        json.dump(meta, f, indent=2)

    return {"status": "success", "project_name": pname}

@app.post("/api/projects/rename")
def rename_project(req: RenameProjectRequest):
    old_pname = req.old_name.strip()
    new_pname = req.new_name.strip()
    
    if not old_pname or not new_pname:
        raise HTTPException(status_code=400, detail="Project names cannot be empty.")

    old_clean = "".join(c for c in old_pname if c.isalnum() or c in ('_', '-'))
    new_clean = "".join(c for c in new_pname if c.isalnum() or c in ('_', '-'))

    old_dir = os.path.join(PROJECTS_DIR, old_clean)
    new_dir = os.path.join(PROJECTS_DIR, new_clean)

    if not os.path.exists(old_dir):
        raise HTTPException(status_code=404, detail="Source project does not exist.")
    if os.path.exists(new_dir) and old_clean != new_clean:
        raise HTTPException(status_code=400, detail="A project with the new name already exists.")

    if old_clean != new_clean:
        shutil.move(old_dir, new_dir)

    # Update metadata file with new name
    meta_path = os.path.join(new_dir, "project.json")
    meta = {}
    if os.path.exists(meta_path):
        try:
            with open(meta_path, "r") as f:
                meta = json.load(f)
        except Exception:
            meta = {}
    
    meta["project_name"] = new_clean
    meta["saved_at"] = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    with open(meta_path, "w") as f:
        json.dump(meta, f, indent=2)

    return {"status": "success", "old_name": old_clean, "new_name": new_clean}

@app.get("/api/projects/{name}")
def load_project(name: str):
    pdir = os.path.join(PROJECTS_DIR, name)
    if not os.path.exists(pdir):
        raise HTTPException(status_code=404, detail="Project not found.")

    files = {}
    for fname in os.listdir(pdir):
        fpath = os.path.join(pdir, fname)
        if os.path.isfile(fpath) and fname != "project.json":
            with open(fpath, "r") as f:
                files[fname] = f.read()

    if "testbench.sv" not in files:
        files["testbench.sv"] = ""
    if "design.sv" not in files:
        files["design.sv"] = ""

    waves = []
    waves_dir = os.path.join(pdir, "waves")
    if os.path.exists(waves_dir):
        for wf in os.listdir(waves_dir):
            if wf.endswith(".vcd"):
                wpath = os.path.join(waves_dir, wf)
                waves.append({
                    "filename": wf,
                    "timestamp": datetime.fromtimestamp(os.path.getmtime(wpath)).strftime("%Y-%m-%d %H:%M:%S")
                })
    waves.sort(key=lambda x: x["timestamp"], reverse=True)

    settings = {}
    status = {}
    meta_path = os.path.join(pdir, "project.json")
    if os.path.exists(meta_path):
        try:
            with open(meta_path, "r") as f:
                meta = json.load(f)
                settings = meta.get("settings", {})
                status = meta.get("status", {})
        except Exception as e:
            print(f"[ERROR] Failed to read project.json for {name}: {e}")
            settings = {}
            status = {}

    return {
        "project_name": name,
        "files": files,
        "waves": waves,
        "settings": settings,
        "status": status
    }

@app.delete("/api/projects/{name}")
def delete_project(name: str):
    pdir = os.path.join(PROJECTS_DIR, name)
    if os.path.exists(pdir):
        shutil.rmtree(pdir)
        return {"status": "success"}
    raise HTTPException(status_code=404, detail="Project not found.")

@app.get("/api/projects/{name}/waves/{filename}")
def get_vcd_file(name: str, filename: str):
    wpath = os.path.join(PROJECTS_DIR, name, "waves", filename)
    if not os.path.exists(wpath):
        raise HTTPException(status_code=404, detail="VCD file not found.")
    with open(wpath, "r") as f:
        content = f.read()
    return {"filename": filename, "vcd": content}

@app.delete("/api/projects/{name}/waves/{filename}")
def delete_vcd_file(name: str, filename: str):
    wpath = os.path.join(PROJECTS_DIR, name, "waves", filename)
    if os.path.exists(wpath):
        os.remove(wpath)
        return {"status": "success"}
    raise HTTPException(status_code=404, detail="VCD file not found.")

@app.post("/api/simulate")
def simulate(req: SimRequest):
    if not req.files:
        return {
            "status": "error",
            "output": "[ERROR] No source files provided.",
            "vcd": None,
            "vcd_filename": None,
            "telemetry": None,
            "debug_log": None
        }

    entry_tb = req.entry_tb if (req.entry_tb in req.files) else "testbench.sv"
    processed_files = dict(req.files)

    # Inspect source files for manual $dumpfile / $dumpvars
    dump_regex = re.compile(r'\$(dumpfile|dumpvars)\b')
    hardcoded_locations = []

    for fname, content in req.files.items():
        lines = content.split('\n')
        for idx, line in enumerate(lines, 1):
            clean_line = line.strip()
            if clean_line.startswith('//') or clean_line.startswith('/*'):
                continue
            if dump_regex.search(clean_line):
                hardcoded_locations.append(f"{fname}:{idx}")

    has_hardcoded_vcd = len(hardcoded_locations) > 0

    # Rule 1: If UI toggle is checked AND hardcoded lines exist -> abort
    if req.enable_vcd and has_hardcoded_vcd:
        locs_str = ", ".join(hardcoded_locations)
        abort_msg = (
            f"[CAS VCD CONFLICT ERROR] Hardcoded waveform dump routine detected at {locs_str}.\n"
            f"Action Required:\n"
            f"  1. Delete or comment out the manual $dumpfile/$dumpvars lines from your testbench, OR\n"
            f"  2. Uncheck 'Enable VCD Wave Dump' in the left sidebar to allow manual dumping."
        )
        return {
            "status": "error",
            "output": abort_msg,
            "vcd": None,
            "vcd_filename": None,
            "telemetry": None,
            "debug_log": None
        }

    # Rule 2: If UI toggle is checked AND NO hardcoded lines exist -> auto-inject VCD block
    if req.enable_vcd and not has_hardcoded_vcd and entry_tb in processed_files:
        tb_src = processed_files[entry_tb]
        endmod_pos = tb_src.rfind("endmodule")
        if endmod_pos != -1:
            raw_vcd_name = (req.vcd_filename or "trace.vcd").strip()
            if not raw_vcd_name.endswith(".vcd"):
                raw_vcd_name += ".vcd"
            
            levels = req.vcd_levels if (req.vcd_levels is not None and req.vcd_levels >= 0) else 0

            injected_block = (
                f"\n// [CAS Auto-Generated VCD Waveform Injection]\n"
                f"initial begin\n"
                f"    $dumpfile(\"{raw_vcd_name}\");\n"
                f"    $dumpvars({levels});\n"
                f"end\n\n"
            )
            processed_files[entry_tb] = tb_src[:endmod_pos] + injected_block + tb_src[endmod_pos:]

    with tempfile.TemporaryDirectory() as tmpdir:
        for fname, content in processed_files.items():
            fpath = os.path.join(tmpdir, fname)
            os.makedirs(os.path.dirname(fpath), exist_ok=True)
            with open(fpath, "w") as f:
                f.write(content)

        files_to_run = [os.path.join(tmpdir, entry_tb)]

        tb_text = processed_files.get(entry_tb, "")
        if "design.sv" in processed_files:
            if '`include "design.sv"' not in tb_text and "`include 'design.sv'" not in tb_text:
                files_to_run.append(os.path.join(tmpdir, "design.sv"))

        cas_bin = "/app/CAS/build/cas"
        options = []
        if not req.enable_profiler:
            options.append("--no-profile")
        if not req.enable_debug:
            options.append("--quiet")

        # Disable VCD only if user explicitly unchecked the toggle AND didn't provide hardcoded VCD code
        if not req.enable_vcd and not has_hardcoded_vcd:
            options.append("--no-vcd")

        if not req.enable_display:
            options.append("--no-display")
        if not req.enable_monitor:
            options.append("--no-monitor")
        if not req.enable_finish_msg:
            options.append("--no-finish-msg")
        if req.top_module and req.top_module.strip():
            options += ["--top", req.top_module.strip()]
        if req.timescale and req.timescale.strip():
            options += ["--timescale", req.timescale.strip()]

        run_cmd = [cas_bin] + options + files_to_run

        try:
            res = subprocess.run(run_cmd, cwd=tmpdir, capture_output=True, text=True, timeout=15)
            raw_output = res.stdout if res.returncode == 0 else (res.stderr + "\n" + res.stdout)
            normalized_output = raw_output.replace(tmpdir + "/", "").replace(tmpdir + "\\", "")

            telemetry_data = None
            debug_log = ""
            clean_output = normalized_output

            if "---BEGIN_CAS_DEBUG_LOG---" in clean_output and "---END_CAS_DEBUG_LOG---" in clean_output:
                b1, p1 = clean_output.split("---BEGIN_CAS_DEBUG_LOG---", 1)
                debug_log, b2 = p1.split("---END_CAS_DEBUG_LOG---", 1)
                clean_output = b1 + b2
                debug_log = debug_log.strip()

            if "---BEGIN_CAS_TELEMETRY---" in clean_output and "---END_CAS_TELEMETRY---" in clean_output:
                before, part = clean_output.split("---BEGIN_CAS_TELEMETRY---", 1)
                json_str, after = part.split("---END_CAS_TELEMETRY---", 1)
                clean_output = before + after
                try:
                    telemetry_data = json.loads(json_str.strip())
                except Exception:
                    telemetry_data = None

            console_output = clean_output.strip()

            vcd_content = None
            vcd_filename = None

            for fname in os.listdir(tmpdir):
                if fname.endswith(".vcd"):
                    vcd_filename = fname
                    with open(os.path.join(tmpdir, fname), "r") as vf:
                        vcd_content = vf.read()

                    if req.project_name:
                        pdir = os.path.join(PROJECTS_DIR, req.project_name)
                        if os.path.exists(pdir):
                            pwaves = os.path.join(pdir, "waves")
                            os.makedirs(pwaves, exist_ok=True)
                            dest_path = os.path.join(pwaves, fname)
                            shutil.copy2(os.path.join(tmpdir, fname), dest_path)
                            vcd_filename = fname
                    break

            return {
                "status": "success" if res.returncode == 0 else "error",
                "output": console_output,
                "vcd": vcd_content,
                "vcd_filename": vcd_filename,
                "telemetry": telemetry_data,
                "debug_log": debug_log
            }
        except subprocess.TimeoutExpired as e:
            partial_out = (e.stdout or "") + "\n" + (e.stderr or "")
            cleaned = partial_out.replace(tmpdir + "/", "").replace(tmpdir + "\\", "")
            return {
                "status": "error",
                "output": f"[CAS TIMEOUT (15s)]\n{cleaned}",
                "vcd": None,
                "vcd_filename": None,
                "telemetry": None,
                "debug_log": None
            }
        except Exception as e:
            return {
                "status": "error",
                "output": f"[CAS INVOCATION ERROR] {str(e)}",
                "vcd": None,
                "vcd_filename": None,
                "telemetry": None,
                "debug_log": None
            }

window.StudioProjects = {
    activeProject: null,

    DEFAULT_SETTINGS: {
        top_module: "",
        timescale: "1ns",
        enable_profiler: false,
        enable_debug: false,
        enable_vcd: true,
        vcd_filename: "trace.vcd",
        vcd_levels: 0,
        enable_display: true,
        enable_monitor: true,
        enable_finish_msg: true
    },

    newProject() {
        if (confirm("Start a new scratchpad? Any unsaved edits will be discarded.")) {
            this.activeProject = null;
            document.getElementById('active-project-label').textContent = "[Unsaved Scratchpad]";
            if (window.WaveApp) window.WaveApp.clear();

            for (const key in window.StudioEditor.projectFiles) {
                delete window.StudioEditor.projectFiles[key];
            }
            window.StudioEditor.projectFiles["testbench.sv"] = "";
            window.StudioEditor.projectFiles["design.sv"] = "";
            window.StudioEditor.activeTbFile = "testbench.sv";
            window.StudioEditor.activeRtlFile = "design.sv";
            window.StudioEditor.tbEditor.setValue("");
            window.StudioEditor.rtlEditor.setValue("");
            window.StudioEditor.renderTabs();

            // Reset sidebar to default clean state
            this.applySettings(this.DEFAULT_SETTINGS);
            this.refreshWaveDropdown([]);
        }
    },

    getCurrentSettings() {
        return {
            top_module: document.getElementById('cfg-top-module').value.trim(),
            timescale: document.getElementById('cfg-timescale').value,
            enable_profiler: document.getElementById('chk-profiler').checked,
            enable_debug: document.getElementById('chk-debug-trace').checked,
            enable_vcd: document.getElementById('chk-vcd-enable').checked,
            vcd_filename: document.getElementById('cfg-vcd-filename').value.trim() || "trace.vcd",
            vcd_levels: parseInt(document.getElementById('cfg-vcd-levels').value, 10) || 0,
            enable_display: document.getElementById('chk-opt-display').checked,
            enable_monitor: document.getElementById('chk-opt-monitor').checked,
            enable_finish_msg: document.getElementById('chk-opt-finish').checked
        };
    },

    applySettings(s) {
        const cfg = Object.assign({}, this.DEFAULT_SETTINGS, s || {});

        const topInput = document.getElementById('cfg-top-module');
        if (topInput) topInput.value = cfg.top_module || "";

        const tsSelect = document.getElementById('cfg-timescale');
        if (tsSelect) {
            tsSelect.value = cfg.timescale || "1ns";
            const tsLbl = document.getElementById('lbl-timescale');
            if (tsLbl) tsLbl.textContent = cfg.timescale || "1ns";
        }

        const chkProf = document.getElementById('chk-profiler');
        if (chkProf) chkProf.checked = Boolean(cfg.enable_profiler);

        const chkDebug = document.getElementById('chk-debug-trace');
        if (chkDebug) chkDebug.checked = Boolean(cfg.enable_debug);

        const chkVcd = document.getElementById('chk-vcd-enable');
        if (chkVcd) {
            chkVcd.checked = Boolean(cfg.enable_vcd);
            if (typeof window.toggleVcdSettingsDrawer === 'function') {
                window.toggleVcdSettingsDrawer(chkVcd.checked);
            }
        }

        const vcdFileInput = document.getElementById('cfg-vcd-filename');
        if (vcdFileInput) {
            vcdFileInput.value = cfg.vcd_filename || "trace.vcd";
        }

        const vcdLvlSelect = document.getElementById('cfg-vcd-levels');
        if (vcdLvlSelect) {
            vcdLvlSelect.value = String(cfg.vcd_levels !== undefined ? cfg.vcd_levels : 0);
        }

        const chkDisp = document.getElementById('chk-opt-display');
        if (chkDisp) chkDisp.checked = Boolean(cfg.enable_display);

        const chkMon = document.getElementById('chk-opt-monitor');
        if (chkMon) chkMon.checked = Boolean(cfg.enable_monitor);

        const chkFin = document.getElementById('chk-opt-finish');
        if (chkFin) chkFin.checked = Boolean(cfg.enable_finish_msg);
    },

    async load(name) {
        this.closeModal();
        const res = await fetch(`/api/projects/${name}`);
        if (!res.ok) {
            alert("Failed to load project: " + name);
            return;
        }
        const data = await res.json();

        this.activeProject = data.project_name;
        document.getElementById('active-project-label').textContent = this.activeProject;

        // 1. Reset and load project source files
        for (const key in window.StudioEditor.projectFiles) {
            delete window.StudioEditor.projectFiles[key];
        }
        Object.assign(window.StudioEditor.projectFiles, data.files);

        // 2. Load and apply project settings immediately
        this.applySettings(data.settings);

        // 3. Restore active editor tabs
        const status = data.status || {};
        const targetTb = (status.active_tb_file && data.files[status.active_tb_file]) 
            ? status.active_tb_file 
            : (data.files["testbench.sv"] ? "testbench.sv" : Object.keys(data.files)[0]);

        const targetRtl = (status.active_rtl_file && data.files[status.active_rtl_file]) 
            ? status.active_rtl_file 
            : (data.files["design.sv"] ? "design.sv" : Object.keys(data.files)[0]);

        window.StudioEditor.activeTbFile = targetTb;
        window.StudioEditor.activeRtlFile = targetRtl;

        window.StudioEditor.tbEditor.setValue(window.StudioEditor.projectFiles[targetTb] || "");
        window.StudioEditor.rtlEditor.setValue(window.StudioEditor.projectFiles[targetRtl] || "");
        window.StudioEditor.renderTabs();

        // 4. Populate waves dropdown and load selected trace
        this.refreshWaveDropdown(data.waves);
        if (window.WaveApp) window.WaveApp.clear();

        if (data.waves && data.waves.length > 0) {
            const waveToLoad = (status.selected_wave && data.waves.some(w => w.filename === status.selected_wave))
                ? status.selected_wave
                : data.waves[0].filename;
            this.loadVcd(waveToLoad);
            const sel = document.getElementById('wave-select');
            if (sel) sel.value = waveToLoad;
        }
    },

    async save() {
        let name = this.activeProject;
        if (!name) {
            name = prompt("Enter project name:");
            if (!name) return;
        }

        window.StudioEditor.projectFiles[window.StudioEditor.activeTbFile] = window.StudioEditor.tbEditor.getValue();
        window.StudioEditor.projectFiles[window.StudioEditor.activeRtlFile] = window.StudioEditor.rtlEditor.getValue();

        const currentSettings = this.getCurrentSettings();
        const currentStatus = {
            active_tb_file: window.StudioEditor.activeTbFile,
            active_rtl_file: window.StudioEditor.activeRtlFile,
            selected_wave: document.getElementById('wave-select') ? document.getElementById('wave-select').value : ""
        };

        const res = await fetch('/api/projects/save', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                project_name: name,
                files: window.StudioEditor.projectFiles,
                settings: currentSettings,
                status: currentStatus
            })
        });

        const data = await res.json();
        if (data.status === "success") {
            this.activeProject = data.project_name;
            document.getElementById('active-project-label').textContent = this.activeProject;
            window.StudioConsole.render(`[CAS] Project '${this.activeProject}' saved cleanly with all files & settings.\n`);
        } else {
            alert("Error saving project: " + (data.detail || "Unknown error"));
        }
    },

    async renamePrompt(e, currentName) {
        if (e) e.stopPropagation();
        const newName = prompt(`Enter new name for project '${currentName}':`, currentName);
        if (!newName || !newName.trim() || newName.trim() === currentName) return;

        const res = await fetch('/api/projects/rename', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                old_name: currentName,
                new_name: newName.trim()
            })
        });

        const data = await res.json();
        if (res.ok && data.status === "success") {
            if (this.activeProject === currentName) {
                this.activeProject = data.new_name;
                document.getElementById('active-project-label').textContent = this.activeProject;
            }
            this.openModal();
        } else {
            alert("Rename failed: " + (data.detail || "Unknown error"));
        }
    },

    async openModal() {
        const modal = document.getElementById('projects-modal');
        const list = document.getElementById('projects-list');
        list.innerHTML = "Loading...";
        modal.classList.add('open');

        const res = await fetch('/api/projects');
        const data = await res.json();
        list.innerHTML = "";

        if (data.projects.length === 0) {
            list.innerHTML = "<div style='color: var(--text-muted); font-size: 12px;'>No projects saved yet.</div>";
            return;
        }

        data.projects.forEach(p => {
            const row = document.createElement('div');
            row.className = 'project-item';
            row.innerHTML = `
                <div style="flex: 1; overflow: hidden; cursor: pointer;">
                    <div class="project-item-title">${p.name}</div>
                    <div class="project-item-meta">Modified: ${p.last_modified} | ${p.vcd_count} waveform(s)</div>
                </div>
                <div style="display: flex; gap: 6px;">
                    <button class="btn btn-secondary" style="font-size: 11px; padding: 3px 8px;" onclick="StudioProjects.renamePrompt(event, '${p.name}')">Rename</button>
                    <button class="btn btn-secondary" style="color: var(--accent-red); font-size: 11px; padding: 3px 8px;" onclick="StudioProjects.delete(event, '${p.name}')">Delete</button>
                </div>
            `;
            row.querySelector('div:first-child').onclick = () => this.load(p.name);
            list.appendChild(row);
        });
    },

    closeModal() {
        document.getElementById('projects-modal').classList.remove('open');
    },

    async load(name) {
        this.closeModal();
        const res = await fetch(`/api/projects/${name}`);
        const data = await res.json();

        this.activeProject = data.project_name;
        document.getElementById('active-project-label').textContent = this.activeProject;

        for (const key in window.StudioEditor.projectFiles) {
            delete window.StudioEditor.projectFiles[key];
        }
        Object.assign(window.StudioEditor.projectFiles, data.files);

        // Restore active editor files if recorded
        const status = data.status || {};
        const targetTb = (status.active_tb_file && data.files[status.active_tb_file]) 
            ? status.active_tb_file 
            : (data.files["testbench.sv"] ? "testbench.sv" : Object.keys(data.files)[0]);

        const targetRtl = (status.active_rtl_file && data.files[status.active_rtl_file]) 
            ? status.active_rtl_file 
            : (data.files["design.sv"] ? "design.sv" : Object.keys(data.files)[0]);

        window.StudioEditor.activeTbFile = targetTb;
        window.StudioEditor.activeRtlFile = targetRtl;

        window.StudioEditor.tbEditor.setValue(window.StudioEditor.projectFiles[targetTb] || "");
        window.StudioEditor.rtlEditor.setValue(window.StudioEditor.projectFiles[targetRtl] || "");
        window.StudioEditor.renderTabs();

        // Restore compiler & simulator settings
        this.applySettings(data.settings);

        this.refreshWaveDropdown(data.waves);
        if (window.WaveApp) window.WaveApp.clear();

        if (data.waves.length > 0) {
            const waveToLoad = (status.selected_wave && data.waves.some(w => w.filename === status.selected_wave))
                ? status.selected_wave
                : data.waves[0].filename;
            this.loadVcd(waveToLoad);
            const sel = document.getElementById('wave-select');
            if (sel) sel.value = waveToLoad;
        }
    },

    async delete(e, name) {
        if (e) e.stopPropagation();
        if (confirm(`Are you sure you want to permanently delete project '${name}' and all its files?`)) {
            const res = await fetch(`/api/projects/${name}`, { method: 'DELETE' });
            if (res.ok) {
                if (this.activeProject === name) {
                    this.newProject();
                }
                this.openModal();
            } else {
                alert("Failed to delete project.");
            }
        }
    },

    refreshWaveDropdown(waves) {
        const sel = document.getElementById('wave-select');
        if (!sel) return;
        sel.innerHTML = "";
        if (!waves || waves.length === 0) {
            sel.innerHTML = '<option value="">(No traces saved)</option>';
            return;
        }
        waves.forEach(w => {
            const opt = document.createElement('option');
            opt.value = w.filename;
            opt.textContent = `${w.filename} (${w.timestamp})`;
            sel.appendChild(opt);
        });
    },

    async loadVcd(filename) {
        if (!this.activeProject || !filename) return;
        try {
            const res = await fetch(`/api/projects/${this.activeProject}/waves/${filename}`);
            if (!res.ok) return;
            const data = await res.json();
            
            if (window.WaveApp) {
                window.WaveApp.loadTrace(data.vcd, filename);
            }
        } catch (e) {
            console.error("Failed to load VCD file:", e);
        }
    },

    async deleteSelectedWave() {
        const sel = document.getElementById('wave-select');
        if (!sel) return;
        const fname = sel.value;
        if (!fname || !this.activeProject) return;

        if (confirm(`Delete waveform '${fname}'?`)) {
            const res = await fetch(`/api/projects/${this.activeProject}/waves/${fname}`, { method: 'DELETE' });
            if (res.ok) {
                // Fetch the updated list of waves without touching project files or editor buffers
                const pres = await fetch(`/api/projects/${this.activeProject}`);
                if (pres.ok) {
                    const pdata = await pres.json();
                    this.refreshWaveDropdown(pdata.waves || []);

                    // If deleted wave was the active one, clear canvas or load the next available wave
                    if (window.WaveApp && window.WaveApp.rawVcdFilename === fname) {
                        if (pdata.waves && pdata.waves.length > 0) {
                            this.loadVcd(pdata.waves[0].filename);
                            sel.value = pdata.waves[0].filename;
                        } else {
                            window.WaveApp.clear();
                        }
                    }
                }
            } else {
                alert("Failed to delete waveform file.");
            }
        }
    }
};

window.downloadActiveVcd = () => {
    if (!window.WaveApp || !window.WaveApp.rawVcdContent) return;
    const blob = new Blob([window.WaveApp.rawVcdContent], { type: 'text/plain' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = window.WaveApp.rawVcdFilename || "trace.vcd";
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
};

    async run() {
        const btnRun = document.getElementById('btn-run');
        const bulb = document.getElementById('status-bulb');
        const statusMsg = document.getElementById('status-msg');

        const topModule = document.getElementById('cfg-top-module').value;
        const timescale = document.getElementById('cfg-timescale').value;
        const enableProfiler = document.getElementById('chk-profiler').checked;
        const enableDebug = document.getElementById('chk-debug-trace').checked;
        const enableVcd = document.getElementById('chk-vcd-enable').checked;

        // Custom VCD injection parameters
        const vcdFilename = document.getElementById('cfg-vcd-filename').value;
        const vcdLevels = parseInt(document.getElementById('cfg-vcd-levels').value, 10) || 0;

        const enableDisplay = document.getElementById('chk-opt-display').checked;
        const enableMonitor = document.getElementById('chk-opt-monitor').checked;
        const enableFinish = document.getElementById('chk-opt-finish').checked;

        document.getElementById('lbl-timescale').textContent = timescale;

        window.StudioEditor.projectFiles[window.StudioEditor.activeTbFile] = window.StudioEditor.tbEditor.getValue();
        window.StudioEditor.projectFiles[window.StudioEditor.activeRtlFile] = window.StudioEditor.rtlEditor.getValue();

        btnRun.disabled = true;
        bulb.className = "status-indicator busy";
        statusMsg.textContent = "Simulating...";
        window.StudioConsole.render("[CAS] Executing simulation...\n");

        try {
            const res = await fetch('/api/simulate', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    files: window.StudioEditor.projectFiles,
                    entry_tb: window.StudioEditor.activeTbFile,
                    top_module: topModule,
                    project_name: window.StudioProjects.activeProject,
                    enable_profiler: enableProfiler,
                    enable_debug: enableDebug,
                    enable_vcd: enableVcd,
                    vcd_filename: vcdFilename,
                    vcd_levels: vcdLevels,
                    enable_display: enableDisplay,
                    enable_monitor: enableMonitor,
                    enable_finish_msg: enableFinish,
                    timescale: timescale
                })
            });

            const data = await res.json();
            window.StudioConsole.render(data.output);

            if (data.status === "success") {
                bulb.className = "status-indicator ready";
                statusMsg.textContent = "Simulation Converged";
                document.getElementById('debug-status-pill').textContent = "Converged";
                document.getElementById('debug-status-pill').style.background = "#064e3b";
                document.getElementById('debug-status-pill').style.color = "#34d399";
            } else {
                bulb.className = "status-indicator err";
                statusMsg.textContent = "Simulation Failed";
                document.getElementById('debug-status-pill').textContent = "Failed";
                document.getElementById('debug-status-pill').style.background = "#450a0a";
                document.getElementById('debug-status-pill').style.color = "#ef4444";
            }

            if (enableDebug && data.debug_log && data.debug_log.trim().length > 0) {
                document.getElementById('debug-trace-log').textContent = data.debug_log;
            } else {
                document.getElementById('debug-trace-log').textContent = "// Compiler Phase Trace disabled in Settings sidebar.";
            }

            if (enableProfiler && data.telemetry && data.telemetry.profiling_enabled) {
                const t = data.telemetry;
                document.getElementById('prof-wall').textContent = t.wallclock_ms.toFixed(2) + " ms";
                document.getElementById('prof-user').textContent = t.user_cpu_ms.toFixed(2) + " ms";
                document.getElementById('prof-sys').textContent = t.sys_cpu_ms.toFixed(2) + " ms";
                document.getElementById('prof-eff').textContent = t.cpu_util_pct.toFixed(1) + "% CPU";
                document.getElementById('prof-rss').textContent = (t.peak_rss_kb / 1024.0).toFixed(2) + " MB";
                document.getElementById('prof-src-bytes').textContent = t.source_bytes + " B";
                document.getElementById('prof-modules').textContent = t.modules_count;

                document.getElementById('prof-nets').textContent = t.nets_count + " / " + t.nets_cap;
                document.getElementById('prof-procs').textContent = t.procs_count;
                document.getElementById('prof-slices').textContent = t.slices_count;
                document.getElementById('prof-rams').textContent = t.rams_count;
                document.getElementById('prof-load').textContent = t.net_map_load.toFixed(2);

                document.getElementById('prof-events').textContent = t.events_dispatched;
                document.getElementById('prof-deltas').textContent = t.delta_cycles;
                document.getElementById('prof-nbas').textContent = t.nba_commits;
            } else {
                document.getElementById('prof-wall').textContent = "Bypassed";
                document.getElementById('prof-user').textContent = "--";
                document.getElementById('prof-sys').textContent = "--";
                document.getElementById('prof-eff').textContent = "Disabled";
                document.getElementById('prof-rss').textContent = "--";
                document.getElementById('prof-src-bytes').textContent = "--";
                document.getElementById('prof-modules').textContent = "--";
                document.getElementById('prof-nets').textContent = "--";
                document.getElementById('prof-procs').textContent = "--";
                document.getElementById('prof-slices').textContent = "--";
                document.getElementById('prof-rams').textContent = "--";
                document.getElementById('prof-load').textContent = "--";
                document.getElementById('prof-events').textContent = "--";
                document.getElementById('prof-deltas').textContent = "--";
                document.getElementById('prof-nbas').textContent = "--";
            }

            if (data.vcd && enableVcd) {
                window.WaveApp.rawVcdContent = data.vcd;
                window.WaveApp.rawVcdFilename = data.vcd_filename;
                document.getElementById('btn-export-vcd').disabled = false;
                window.WaveApp.loadTrace(data.vcd, data.vcd_filename);

                if (window.StudioProjects.activeProject) {
                    const pres = await fetch(`/api/projects/${window.StudioProjects.activeProject}`);
                    const pdata = await pres.json();
                    window.StudioProjects.refreshWaveDropdown(pdata.waves);
                    document.getElementById('wave-select').value = data.vcd_filename;
                }
            } else if (!enableVcd) {
                const sel = document.getElementById('wave-select');
                if (sel.value && sel.value !== "") {
                    window.StudioProjects.loadVcd(sel.value);
                } else if (window.StudioProjects.activeProject) {

                    const pdata = await pres.json();
                    window.StudioProjects.refreshWaveDropdown(pdata.waves);

                    else window.WaveApp.clear();
                } else {
                    window.WaveApp.clear();
                }
            } else {
                window.WaveApp.clear();
            }
        } catch (e) {
            window.StudioConsole.render("[UI FETCH ERROR] " + e.message);
            bulb.className = "status-indicator err";
            statusMsg.textContent = "Invocation Error";
        } finally {
            btnRun.disabled = false;
        }
    },

    initResizers() {
        const resizerMain = document.getElementById('resizer-main');
        const editorsWrapper = document.getElementById('editors-wrapper');
        const workspace = document.getElementById('workspace');
        let isDraggingV = false;
        resizerMain.addEventListener('mousedown', () => isDraggingV = true);

        const resizerH = document.getElementById('resizer-editors');
        const colTb = document.getElementById('col-tb');
        let isDraggingH = false;
        resizerH.addEventListener('mousedown', () => isDraggingH = true);

        const leftResizer = document.getElementById('left-resizer');
        const leftSidebar = document.getElementById('options-sidebar');
        let isDraggingLeft = false;
        leftResizer.addEventListener('mousedown', () => isDraggingLeft = true);

        const rightResizer = document.getElementById('right-resizer');
        const rightSidebar = document.getElementById('inspector-sidebar');
        let isDraggingRight = false;
        rightResizer.addEventListener('mousedown', () => isDraggingRight = true);

        window.addEventListener('mousemove', (e) => {
            if (isDraggingV) {
                const h = workspace.offsetHeight;
                const topH = ((e.clientY - workspace.offsetTop) / h) * 100;
                if (topH > 20 && topH < 85) {
                    editorsWrapper.style.height = topH + '%';
                    window.StudioEditor.relayout();
                }
            }
            if (isDraggingH) {
                const w = editorsWrapper.offsetWidth;
                const leftW = ((e.clientX - editorsWrapper.offsetLeft) / w) * 100;
                if (leftW > 20 && leftW < 80) {
                    colTb.style.flex = "none";
                    colTb.style.width = leftW + '%';
                    window.StudioEditor.relayout();
                }
            }
            if (isDraggingLeft) {
                const newW = e.clientX;
                if (newW > 180 && newW < 450) {
                    leftSidebar.style.width = newW + 'px';
                    window.StudioEditor.relayout();
                }
            }
            if (isDraggingRight) {
                const newW = window.innerWidth - e.clientX;
                if (newW > 240 && newW < 600) {
                    rightSidebar.style.width = newW + 'px';
                    window.StudioEditor.relayout();
                }
            }
        });

        window.addEventListener('mouseup', () => {
            if (isDraggingV || isDraggingH || isDraggingLeft || isDraggingRight) {
                window.StudioEditor.relayout();
            }
            isDraggingV = false;
            isDraggingH = false;
            isDraggingLeft = false;
            isDraggingRight = false;
        });
    }
};
